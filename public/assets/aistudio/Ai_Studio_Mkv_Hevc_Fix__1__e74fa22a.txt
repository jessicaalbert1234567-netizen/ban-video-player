/**
 * Universal Stereo Downmix & PCM Normalizer AudioProcessor.
 *
 * Safely converts ANY channel configuration (Mono 1.0, Stereo 2.0, 2.1, 3.0, Quad 4.0, 5.0,
 * 5.1 Surround, 6.1, 7.1 Surround) and ANY PCM sample encoding (16-bit, Float/fltp, 24-bit,
 * 32-bit, 8-bit) into universal 16-bit Stereo PCM (C.ENCODING_PCM_16BIT, 2 channels).
 */
@UnstableApi
class UniversalStereoAudioProcessor : BaseAudioProcessor() {

    override fun onConfigure(inputAudioFormat: AudioProcessor.AudioFormat): AudioProcessor.AudioFormat {
        val encoding = inputAudioFormat.encoding
        val channels = inputAudioFormat.channelCount
        val sampleRate = inputAudioFormat.sampleRate

        if (channels <= 0 || sampleRate <= 0) {
            return AudioProcessor.AudioFormat.NOT_SET
        }

        val isSupportedEncoding = encoding == C.ENCODING_PCM_16BIT ||
                encoding == C.ENCODING_PCM_FLOAT ||
                encoding == C.ENCODING_PCM_24BIT ||
                encoding == C.ENCODING_PCM_32BIT ||
                encoding == C.ENCODING_PCM_8BIT

        if (!isSupportedEncoding) {
            throw AudioProcessor.UnhandledAudioFormatException(inputAudioFormat)
        }

        // Standard 16-bit stereo passes through directly with zero overhead
        if (channels == 2 && encoding == C.ENCODING_PCM_16BIT) {
            return AudioProcessor.AudioFormat.NOT_SET
        }

        return AudioProcessor.AudioFormat(sampleRate, 2, C.ENCODING_PCM_16BIT)
    }

    override fun queueInput(inputBuffer: ByteBuffer) {
        val remaining = inputBuffer.remaining()
        if (remaining == 0) return

        val channels = inputAudioFormat.channelCount
        val encoding = inputAudioFormat.encoding
        val bytesPerSample = when (encoding) {
            C.ENCODING_PCM_8BIT -> 1
            C.ENCODING_PCM_16BIT -> 2
            C.ENCODING_PCM_24BIT -> 3
            C.ENCODING_PCM_32BIT, C.ENCODING_PCM_FLOAT -> 4
            else -> 2
        }
        val frameSize = channels * bytesPerSample
        if (frameSize <= 0) {
            inputBuffer.position(inputBuffer.limit())
            return
        }

        val frameCount = remaining / frameSize
        if (frameCount <= 0) {
            inputBuffer.position(inputBuffer.limit())
            return
        }

        val src = inputBuffer.order(ByteOrder.LITTLE_ENDIAN)
        // Output is always 2 channels * 2 bytes (16-bit stereo) = 4 bytes per frame
        val outBuffer = replaceOutputBuffer(frameCount * 4).order(ByteOrder.LITTLE_ENDIAN)
        val chSamples = FloatArray(channels)

        for (frame in 0 until frameCount) {
            for (ch in 0 until channels) {
                chSamples[ch] = readNormalizedSample(src, encoding)
            }

            val left: Float
            val right: Float

            when (channels) {
                1 -> {
                    left = chSamples[0]
                    right = chSamples[0]
                }
                2 -> {
                    left = chSamples[0]
                    right = chSamples[1]
                }
                3 -> {
                    // 3.0: FL, FR, FC
                    left = chSamples[0] * 0.65f + chSamples[2] * 0.45f
                    right = chSamples[1] * 0.65f + chSamples[2] * 0.45f
                }
                4 -> {
                    // Quad / 4.0: FL, FR, BL, BR
                    left = chSamples[0] * 0.65f + chSamples[2] * 0.45f
                    right = chSamples[1] * 0.65f + chSamples[3] * 0.45f
                }
                5 -> {
                    // 5.0: FL, FR, FC, BL, BR
                    left = chSamples[0] * 0.55f + chSamples[2] * 0.50f + chSamples[3] * 0.35f
                    right = chSamples[1] * 0.55f + chSamples[2] * 0.50f + chSamples[4] * 0.35f
                }
                6 -> {
                    // 5.1 Surround (HE-AAC / AAC / AC3): FL(0), FR(1), FC(2), LFE(3), BL(4), BR(5)
                    // Center channel (FC) carries primary spoken dialogue; preserve it clearly!
                    val fl = chSamples[0]
                    val fr = chSamples[1]
                    val fc = chSamples[2]
                    val lfe = chSamples[3]
                    val bl = chSamples[4]
                    val br = chSamples[5]
                    left = fl * 0.55f + fc * 0.58f + lfe * 0.15f + bl * 0.35f
                    right = fr * 0.55f + fc * 0.58f + lfe * 0.15f + br * 0.35f
                }
                else -> {
                    // 7.1 or higher: FL(0), FR(1), FC(2), LFE(3), BL(4), BR(5), SL(6), SR(7)
                    val fl = chSamples[0]
                    val fr = chSamples[1]
                    val fc = chSamples[2]
                    val lfe = chSamples[3]
                    val bl = chSamples[4]
                    val br = chSamples[5]
                    val sl = if (channels > 6) chSamples[6] else 0f
                    val sr = if (channels > 7) chSamples[7] else 0f
                    left = fl * 0.48f + fc * 0.55f + lfe * 0.12f + bl * 0.28f + sl * 0.28f
                    right = fr * 0.48f + fc * 0.55f + lfe * 0.12f + br * 0.28f + sr * 0.28f
                }
            }

            val leftShort = (left * 32767.0f).toInt().coerceIn(-32768, 32767).toShort()
            val rightShort = (right * 32767.0f).toInt().coerceIn(-32768, 32767).toShort()
            outBuffer.putShort(leftShort)
            outBuffer.putShort(rightShort)
        }

        inputBuffer.position(inputBuffer.limit())
        outBuffer.flip()
    }

    private fun readNormalizedSample(src: ByteBuffer, encoding: Int): Float {
        return when (encoding) {
            C.ENCODING_PCM_16BIT -> src.short / 32768.0f
            C.ENCODING_PCM_FLOAT -> {
                val f = src.float
                if (f.isNaN() || f.isInfinite()) 0f else f.coerceIn(-1.0f, 1.0f)
            }
            C.ENCODING_PCM_8BIT -> ((src.get().toInt() and 0xFF) - 128) / 128.0f
            C.ENCODING_PCM_24BIT -> {
                val b0 = src.get().toInt() and 0xFF
                val b1 = src.get().toInt() and 0xFF
                val b2 = src.get().toInt() // signed MSB
                val sample24 = (b2 shl 16) or (b1 shl 8) or b0
                sample24 / 8388608.0f
            }
            C.ENCODING_PCM_32BIT -> src.int / 2147483648.0f
            else -> 0f
        }
    }
}

/**
 * VLC-Grade RenderersFactory for Media3 ExoPlayer:
 * 1. Handles 10-bit HEVC (Main 10, yuv420p10le) and non-16-aligned dimensions (e.g., 1280x536)
 *    by prioritizing Android's 10-bit capable software HEVC decoder (c2.android.hevc.decoder /
 *    OMX.google.hevc.decoder) with seamless fallback to hardware decoders.
 * 2. Requests SDR color transfer output (COLOR_TRANSFER_SDR_VIDEO) on API 31+ for 10-bit SDR HEVC
 *    while preserving MediaCodec PlaceholderSurface and setOutputSurface support.
 * 3. Synchronizes MediaCodecAudioRenderer output channel counts in onOutputFormatChanged so 5.1 HE-AAC
 *    tracks never trigger ChannelMappingAudioProcessor UnhandledAudioFormatException, and downmixes
 *    multichannel/float PCM to 16-bit Stereo via UniversalStereoAudioProcessor.
 */
@UnstableApi
class VlcGradeRenderersFactory(
    context: Context,
    private val preferSoftwareVideoDecoderProvider: () -> Boolean = { true }
) : DefaultRenderersFactory(context) {

    companion object {
        private const val TAG = "VlcGradeRenderers"

        fun createUniversalAudioSink(context: Context): AudioSink {
            return DefaultAudioSink.Builder(context)
                .setAudioProcessors(arrayOf(UniversalStereoAudioProcessor()))
                .setEnableFloatOutput(false)
                .setEnableAudioTrackPlaybackParams(false)
                .build()
        }

        fun isFormat10BitHevc(format: Format): Boolean {
            if (format.sampleMimeType != MimeTypes.VIDEO_H265) return false
            val codecs = format.codecs?.lowercase() ?: ""
            val hasMain10CodecTag = codecs.startsWith("hev1.2") ||
                    codecs.startsWith("hvc1.2") ||
                    codecs.contains(".2.") ||
                    codecs.contains("main10")
            val has10BitColor = (format.colorInfo?.lumaBitdepth ?: -1) >= 10 ||
                    (format.colorInfo?.chromaBitdepth ?: -1) >= 10
            if (hasMain10CodecTag || has10BitColor) return true

            // Also inspect raw HEVC SPS/VPS NAL units in initializationData for Main 10 profile_idc == 2
            return hasHevcMain10NalUnit(format.initializationData)
        }

        fun hasNonMacroblockAlignedDimensions(format: Format): Boolean {
            if (format.sampleMimeType != MimeTypes.VIDEO_H265) return false
            val w = format.width
            val h = format.height
            return (w > 0 && w % 16 != 0) || (h > 0 && h % 16 != 0)
        }

        private fun hasHevcMain10NalUnit(initData: List<ByteArray>?): Boolean {
            if (initData.isNullOrEmpty()) return false
            for (buf in initData) {
                var i = 0
                while (i + 7 < buf.size) {
                    // Find 3-byte or 4-byte Annex-B start code: 0x000001 or 0x00000001
                    if (buf[i].toInt() == 0 && buf[i + 1].toInt() == 0) {
                        val nalStart = when {
                            buf[i + 2].toInt() == 1 -> i + 3
                            i + 3 < buf.size && buf[i + 2].toInt() == 0 && buf[i + 3].toInt() == 1 -> i + 4
                            else -> -1
                        }
                        if (nalStart != -1 && nalStart + 4 < buf.size) {
                            val nalType = (buf[nalStart].toInt() and 0x7E) ushr 1
                            // SPS (nal_unit_type == 33)
                            if (nalType == 33) {
                                val ptlByte0 = buf[nalStart + 3].toInt() and 0xFF
                                val profileIdc = ptlByte0 and 0x1F
                                val compatByte0 = buf[nalStart + 4].toInt() and 0xFF
                                val main10Compat = (compatByte0 and 0x20) != 0
                                if (profileIdc == 2 || main10Compat) return true
                            }
                            i = nalStart + 2
                            continue
                        }
                    }
                    i++
                }
            }
            return false
        }

        fun isSoftwareDecoder(info: ExoMediaCodecInfo): Boolean {
            val name = info.name.lowercase()
            return info.softwareOnly ||
                    !info.hardwareAccelerated ||
                    name.startsWith("c2.android.") ||
                    name.startsWith("omx.google.") ||
                    name.contains(".sw.") ||
                    name.contains("ffmpeg")
        }

        fun supportsHevcMain10Profile(info: ExoMediaCodecInfo): Boolean {
            return try {
                val levels = info.profileLevels
                levels.any {
                    it.profile == MediaCodecInfo.CodecProfileLevel.HEVCProfileMain10 ||
                            it.profile == MediaCodecInfo.CodecProfileLevel.HEVCProfileMain10HDR10 ||
                            it.profile == MediaCodecInfo.CodecProfileLevel.HEVCProfileMain10HDR10Plus
                }
            } catch (_: Exception) {
                false
            }
        }
    }

    init {
        setExtensionRendererMode(EXTENSION_RENDERER_MODE_PREFER)
        setEnableDecoderFallback(true)
    }

    override fun buildAudioSink(
        context: Context,
        enableFloatOutput: Boolean,
        enableAudioTrackPlaybackParams: Boolean
    ): AudioSink {
        return createUniversalAudioSink(context)
    }

    override fun buildVideoRenderers(
        context: Context,
        extensionRendererMode: Int,
        mediaCodecSelector: MediaCodecSelector,
        enableDecoderFallback: Boolean,
        eventHandler: Handler,
        eventListener: VideoRendererEventListener,
        allowedVideoJoiningTimeMs: Long,
        out: ArrayList<Renderer>
    ) {
        val smartVideoCodecSelector = MediaCodecSelector { mimeType, requiresSecureDecoder, requiresTunnelingDecoder ->
            val defaultList = mediaCodecSelector.getDecoderInfos(
                mimeType,
                requiresSecureDecoder,
                requiresTunnelingDecoder
            ).toMutableList()

            try {
                val rawSystemDecoders = MediaCodecUtil.getDecoderInfos(
                    mimeType,
                    requiresSecureDecoder,
                    requiresTunnelingDecoder
                )
                for (d in rawSystemDecoders) {
                    if (defaultList.none { it.name == d.name }) {
                        defaultList.add(d)
                    }
                }
            } catch (_: Exception) {}

            if (defaultList.size <= 1) return@MediaCodecSelector defaultList

            val preferSw = preferSoftwareVideoDecoderProvider()
            if (mimeType.equals(MimeTypes.VIDEO_H265, ignoreCase = true)) {
                defaultList.sortedBy { info ->
                    if (preferSw) {
                        when {
                            isSoftwareDecoder(info) -> 0
                            supportsHevcMain10Profile(info) -> 1
                            else -> 2
                        }
                    } else {
                        when {
                            !isSoftwareDecoder(info) && supportsHevcMain10Profile(info) -> 0
                            !isSoftwareDecoder(info) -> 1
                            else -> 2
                        }
                    }
                }
            } else {
                defaultList
            }
        }

        val customVideoRenderer = object : MediaCodecVideoRenderer(
            context,
            codecAdapterFactory,
            smartVideoCodecSelector,
            allowedVideoJoiningTimeMs,
            enableDecoderFallback,
            eventHandler,
            eventListener,
            MAX_DROPPED_VIDEO_FRAME_COUNT_TO_NOTIFY
        ) {
            override fun supportsFormat(
                mediaCodecSelector: MediaCodecSelector,
                format: Format
            ): Int {
                val baseSupport = super.supportsFormat(mediaCodecSelector, format)
                val formatSupport = RendererCapabilities.getFormatSupport(baseSupport)
                if (format.sampleMimeType == MimeTypes.VIDEO_H265 && formatSupport != C.FORMAT_HANDLED) {
                    return RendererCapabilities.create(
                        C.FORMAT_HANDLED,
                        RendererCapabilities.ADAPTIVE_SEAMLESS,
                        RendererCapabilities.TUNNELING_NOT_SUPPORTED
                    )
                }
                return baseSupport
            }

            override fun getDecoderInfos(
                selector: MediaCodecSelector,
                format: Format,
                requiresSecureDecoder: Boolean
            ): MutableList<ExoMediaCodecInfo> {
                val infos = super.getDecoderInfos(selector, format, requiresSecureDecoder).toMutableList()
                val mime = format.sampleMimeType ?: MimeTypes.VIDEO_H265

                try {
                    val allFromSelector = selector.getDecoderInfos(mime, requiresSecureDecoder, false)
                    for (candidate in allFromSelector) {
                        if (infos.none { it.name == candidate.name }) {
                            infos.add(candidate)
                        }
                    }
                    val allRaw = MediaCodecUtil.getDecoderInfos(mime, requiresSecureDecoder, false)
                    for (candidate in allRaw) {
                        if (infos.none { it.name == candidate.name }) {
                            infos.add(candidate)
                        }
                    }
                } catch (_: Exception) {}

                val is10BitOrOddSizeHevc = isFormat10BitHevc(format) || hasNonMacroblockAlignedDimensions(format)
                val preferSw = preferSoftwareVideoDecoderProvider()

                if (mime.equals(MimeTypes.VIDEO_H265, ignoreCase = true) || is10BitOrOddSizeHevc) {
                    infos.sortBy { info ->
                        if (preferSw && is10BitOrOddSizeHevc) {
                            when {
                                isSoftwareDecoder(info) -> 0
                                supportsHevcMain10Profile(info) -> 1
                                else -> 2
                            }
                        } else if (preferSw) {
                            when {
                                isSoftwareDecoder(info) -> 0
                                supportsHevcMain10Profile(info) -> 1
                                else -> 2
                            }
                        } else {
                            when {
                                !isSoftwareDecoder(info) && supportsHevcMain10Profile(info) -> 0
                                !isSoftwareDecoder(info) -> 1
                                else -> 2
                            }
                        }
                    }
                    Log.i(
                        TAG,
                        "Resolved HEVC decoders (10bitOrOdd=$is10BitOrOddSizeHevc, preferSw=$preferSw, ${format.width}x${format.height}): ${infos.map { it.name }}"
                    )
                }
                return infos
            }

            override fun getMediaFormat(
                format: Format,
                codecMimeType: String,
                codecMaxValues: CodecMaxValues,
                codecOperatingRate: Float,
                deviceNeedsNoPostProcessWorkaround: Boolean,
                tunnelingAudioSessionId: Int
            ): MediaFormat {
                val mediaFormat = super.getMediaFormat(
                    format,
                    codecMimeType,
                    codecMaxValues,
                    codecOperatingRate,
                    deviceNeedsNoPostProcessWorkaround,
                    tunnelingAudioSessionId
                )
                // Request 8-bit SDR output transfer for 10-bit SDR HEVC (e.g. yuv420p10le bt709)
                // while keeping KEY_PROFILE intact so both software and hardware Main 10 decoders initialize properly.
                if (Build.VERSION.SDK_INT >= 31 && isFormat10BitHevc(format)) {
                    try {
                        mediaFormat.setInteger(
                            MediaFormat.KEY_COLOR_TRANSFER_REQUEST,
                            MediaFormat.COLOR_TRANSFER_SDR_VIDEO
                        )
                    } catch (_: Exception) {}
                }
                return mediaFormat
            }
        }

        out.add(customVideoRenderer)

        val tempOut = ArrayList<Renderer>()
        super.buildVideoRenderers(
            context,
            extensionRendererMode,
            smartVideoCodecSelector,
            enableDecoderFallback,
            eventHandler,
            eventListener,
            allowedVideoJoiningTimeMs,
            tempOut
        )
        for (r in tempOut) {
            if (r !is MediaCodecVideoRenderer) {
                out.add(r)
            }
        }
    }

    override fun buildAudioRenderers(
        context: Context,
        extensionRendererMode: Int,
        mediaCodecSelector: MediaCodecSelector,
        enableDecoderFallback: Boolean,
        audioSink: AudioSink,
        eventHandler: Handler,
        eventListener: AudioRendererEventListener,
        out: ArrayList<Renderer>
    ) {
        val customAudioRenderer = object : MediaCodecAudioRenderer(
            context,
            codecAdapterFactory,
            mediaCodecSelector,
            enableDecoderFallback,
            eventHandler,
            eventListener,
            audioSink
        ) {
            override fun getMediaFormat(
                format: Format,
                codecMimeType: String,
                codecMaxValues: Int,
                codecOperatingRate: Float
            ): MediaFormat {
                val mediaFormat = super.getMediaFormat(format, codecMimeType, codecMaxValues, codecOperatingRate)
                // Allow the decoder to output all channels (up to 99) so Media3's 5.1/7.1 AAC
                // ChannelMappingAudioProcessor never sees a 6-vs-2 channel mismatch, while
                // UniversalStereoAudioProcessor in DefaultAudioSink downmixes 5.1/7.1 to 2-channel Stereo.
                mediaFormat.setInteger(MediaFormat.KEY_MAX_OUTPUT_CHANNEL_COUNT, 99)
                if (Build.VERSION.SDK_INT >= 28) {
                    mediaFormat.setInteger("aac-max-output-channel_count", 99)
                }
                if (Build.VERSION.SDK_INT >= 24) {
                    mediaFormat.setInteger(
                        MediaFormat.KEY_PCM_ENCODING,
                        android.media.AudioFormat.ENCODING_PCM_16BIT
                    )
                }
                return mediaFormat
            }

            override fun onOutputFormatChanged(format: Format, outputMediaFormat: MediaFormat?) {
                // Synchronize format.channelCount with the actual channel count emitted by the decoder
                // so MediaCodecAudioRenderer never attaches a 6-channel AAC channelMap if the device's
                // decoder downmixed 5.1 to 2 channels internally.
                val syncedFormat = if (outputMediaFormat != null) {
                    val actualChannels = if (outputMediaFormat.containsKey(MediaFormat.KEY_CHANNEL_COUNT)) {
                        outputMediaFormat.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
                    } else {
                        format.channelCount
                    }
                    val actualSampleRate = if (outputMediaFormat.containsKey(MediaFormat.KEY_SAMPLE_RATE)) {
                        outputMediaFormat.getInteger(MediaFormat.KEY_SAMPLE_RATE)
                    } else {
                        format.sampleRate
                    }
                    format.buildUpon()
                        .setChannelCount(actualChannels)
                        .setSampleRate(actualSampleRate)
                        .setEncoderDelay(0)
                        .setEncoderPadding(0)
                        .build()
                } else {
                    format
                }
                super.onOutputFormatChanged(syncedFormat, outputMediaFormat)
            }
        }

        out.add(customAudioRenderer)

        val tempOut = ArrayList<Renderer>()
        super.buildAudioRenderers(
            context,
            extensionRendererMode,
            mediaCodecSelector,
            enableDecoderFallback,
            audioSink,
            eventHandler,
            eventListener,
            tempOut
        )
        for (r in tempOut) {
            if (r !is MediaCodecAudioRenderer) {
                out.add(r)
            }
        }
    }
}

    FIT("Fit (ফিট)", AspectRatioFrameLayout.RESIZE_MODE_FIT),
    ZOOM_CROP("Full Zoom (ফুউল জুম)", AspectRatioFrameLayout.RESIZE_MODE_ZOOM),
    STRETCH("Stretch (টেনে পূর্ণ)", AspectRatioFrameLayout.RESIZE_MODE_FILL)
}

data class PlayerState(
    val isPlaying: Boolean = false,
    val currentPositionMs: Long = 0L,
    val durationMs: Long = 0L,
    val audioChoice: AudioTrackChoice = AudioTrackChoice.BANGLA_DUB,
    val subtitleChoice: SubtitleChoice = SubtitleChoice.BANGLA,
    val resizeMode: ResizeModeChoice = ResizeModeChoice.FIT,
    val playbackSpeed: Float = 1.0f,
    val isLocked: Boolean = false,
    val activeSubtitleText: String? = null,
    val hasDubbedAudio: Boolean = false,
    val hasBanglaSubtitles: Boolean = false,
    val hasEnglishSubtitles: Boolean = false,
    val dubbedAudioFile: File? = null,
    val isSpeechActive: Boolean = false,
    val isLiveDubbing: Boolean = false,
    val isAudioOnlyFallback: Boolean = false,
    val hasError: Boolean = false,
    val errorMessage: String? = null
)

@OptIn(UnstableApi::class)
class MediaPlayerManager(private val context: Context) {

    private val TAG = "MediaPlayerManager"

    // Primary player: plays video and original video soundtrack natively
    var player: ExoPlayer? = null
        private set

    // Secondary player: plays synchronized dubbed speech audio track
    private var dubbedAudioPlayer: ExoPlayer? = null

    private val _playerState = MutableStateFlow(PlayerState())
    val playerState: StateFlow<PlayerState> = _playerState.asStateFlow()

    private var currentVideoUri: Uri? = null
    private var currentDubbedAudioFile: File? = null
    private var currentSrtFile: File? = null
    private var currentSourceSrtFile: File? = null

    // Parsed subtitles for instant on-screen rendering and dialogue presence detection
    private val bnSubtitleEntries = mutableListOf<ParsedSubtitle>()
    private val enSubtitleEntries = mutableListOf<ParsedSubtitle>()
    private var embeddedCueText: String? = null

    data class ParsedSubtitle(val startMs: Long, val endMs: Long, val text: String)

    private var liveTts: android.speech.tts.TextToSpeech? = null
    private var isLiveTtsReady = false
    private var lastSpokenCueStartMs: Long = -1L

    private var errorRecoveryAttempts = 0
    private var preferSoftwareVideoDecoder = true
    private var decoderSwitchCount = 0
    private var zeroFramePlayingChecks = 0

    private fun initLiveTts() {
        if (liveTts == null) {
            liveTts = android.speech.tts.TextToSpeech(context) { status ->
                if (status == android.speech.tts.TextToSpeech.SUCCESS) {
                    val res = liveTts?.setLanguage(java.util.Locale("bn", "BD"))
                    if (res == android.speech.tts.TextToSpeech.LANG_MISSING_DATA || res == android.speech.tts.TextToSpeech.LANG_NOT_SUPPORTED) {
                        liveTts?.setLanguage(java.util.Locale("bn", "IN"))
                    }
                    isLiveTtsReady = true
                    Log.i(TAG, "Live Real-Time Bengali TTS Synthesizer initialized successfully")
                }
            }
        }
    }

    private fun createExtractorsFactory(): DefaultExtractorsFactory {
        return DefaultExtractorsFactory()
            .setConstantBitrateSeekingEnabled(true)
            .setConstantBitrateSeekingAlwaysEnabled(true)
    }

    fun initializePlayer(): ExoPlayer {
        if (player == null) {
            val renderersFactory = VlcGradeRenderersFactory(context) { preferSoftwareVideoDecoder }

            val trackSelector = DefaultTrackSelector(context).apply {
                parameters = buildUponParameters()
                    .setExceedRendererCapabilitiesIfNecessary(true)
                    .setExceedVideoConstraintsIfNecessary(true)
                    .setExceedAudioConstraintsIfNecessary(true)
                    .setAllowAudioMixedMimeTypeAdaptiveness(true)
                    .setAllowAudioMixedSampleRateAdaptiveness(true)
                    .setAllowAudioMixedChannelCountAdaptiveness(true)
                    .setAllowVideoMixedMimeTypeAdaptiveness(true)
                    .setAllowVideoNonSeamlessAdaptiveness(true)
                    .setTrackTypeDisabled(C.TRACK_TYPE_VIDEO, false)
                    .setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                    .build()
            }

            val audioAttributes = AudioAttributes.Builder()
                .setContentType(C.AUDIO_CONTENT_TYPE_MOVIE)
                .setUsage(C.USAGE_MEDIA)
                .build()

            val mediaSourceFactory = DefaultMediaSourceFactory(context, createExtractorsFactory())

            player = ExoPlayer.Builder(context, renderersFactory)
                .setTrackSelector(trackSelector)
                .setMediaSourceFactory(mediaSourceFactory)
                .setAudioAttributes(audioAttributes, /* handleAudioFocus= */ true)
                .build().apply {
                repeatMode = Player.REPEAT_MODE_OFF
                playWhenReady = true
                addListener(object : Player.Listener {
                    override fun onIsPlayingChanged(isPlaying: Boolean) {
                        _playerState.value = _playerState.value.copy(isPlaying = isPlaying)
                        if (isPlaying) {
                            dubbedAudioPlayer?.play()
                        } else {
                            dubbedAudioPlayer?.pause()
                            liveTts?.stop()
                        }
                    }

                    override fun onPlaybackStateChanged(playbackState: Int) {
                        if (playbackState == Player.STATE_READY) {
                            errorRecoveryAttempts = 0
                            zeroFramePlayingChecks = 0
                        }
                        val isPlaying = player?.isPlaying ?: false
                        val playWhenReady = player?.playWhenReady ?: false
                        val active = isPlaying || (playWhenReady && (playbackState == Player.STATE_READY || playbackState == Player.STATE_BUFFERING))
                        _playerState.value = _playerState.value.copy(isPlaying = active)
                    }

                    override fun onPlayWhenReadyChanged(playWhenReady: Boolean, reason: Int) {
                        val state = player?.playbackState ?: Player.STATE_IDLE
                        val active = (player?.isPlaying ?: false) || (playWhenReady && state != Player.STATE_ENDED && state != Player.STATE_IDLE)
                        _playerState.value = _playerState.value.copy(isPlaying = active)
                        if (playWhenReady) {
                            dubbedAudioPlayer?.playWhenReady = true
                            dubbedAudioPlayer?.play()
                        } else {
                            dubbedAudioPlayer?.playWhenReady = false
                            dubbedAudioPlayer?.pause()
                            liveTts?.stop()
                        }
                    }

                    override fun onPlaybackParametersChanged(playbackParameters: PlaybackParameters) {
                        _playerState.value = _playerState.value.copy(playbackSpeed = playbackParameters.speed)
                        dubbedAudioPlayer?.setPlaybackSpeed(playbackParameters.speed)
                    }

                    override fun onCues(cueGroup: CueGroup) {
                        val cueText = cueGroup.cues
                            .mapNotNull { it.text?.toString()?.trim() }
                            .filter { it.isNotEmpty() }
                            .joinToString("\n")
                            .ifBlank { null }
                        embeddedCueText = cueText
                        if (bnSubtitleEntries.isEmpty() && enSubtitleEntries.isEmpty()) {
                            updateCurrentSubtitleText(player?.currentPosition ?: 0L)
                        }
                    }

                    override fun onPlayerError(error: PlaybackException) {
                        Log.e(TAG, "ExoPlayer playback error: ${error.errorCodeName} (code=${error.errorCode}) - ${error.message}", error)

                        val isAudioOnlyFile = currentVideoUri?.path?.let { p ->
                            p.endsWith(".wav") || p.endsWith(".m4a") || p.endsWith(".mp3") || p.endsWith(".aac")
                        } ?: false

                        val isVideoCodecError = error.errorCode == PlaybackException.ERROR_CODE_DECODER_INIT_FAILED ||
                                error.errorCode == PlaybackException.ERROR_CODE_DECODING_FAILED ||
                                error.errorCode == PlaybackException.ERROR_CODE_DECODING_FORMAT_EXCEEDS_CAPABILITIES ||
                                error.errorCode == PlaybackException.ERROR_CODE_DECODING_FORMAT_UNSUPPORTED

                        // 1. Toggle between software and hardware video decoder in-place (preserving PlayerView attachment!)
                        if (!isAudioOnlyFile && isVideoCodecError && decoderSwitchCount < 2) {
                            decoderSwitchCount++
                            preferSoftwareVideoDecoder = !preferSoftwareVideoDecoder
                            Log.w(TAG, "Video codec error; switching preferSoftwareVideoDecoder=$preferSoftwareVideoDecoder in-place")
                            reprepareCurrentPlayerInPlace()
                            return
                        }

                        // 2. If media is a video (MKV/MP4) and audio track failed, try stereo track override or disable internal audio so video still plays
                        if (!isAudioOnlyFile && player != null && errorRecoveryAttempts < 3) {
                            errorRecoveryAttempts++
                            Log.i(TAG, "Playback retry attempt ($errorRecoveryAttempts/3) for video container.")
                            try {
                                val exo = player!!
                                val savedPos = exo.currentPosition.coerceAtLeast(0L)
                                val paramsBuilder = exo.trackSelectionParameters
                                    .buildUpon()
                                    .setTrackTypeDisabled(C.TRACK_TYPE_VIDEO, false)

                                if (errorRecoveryAttempts == 1) {
                                    // Prefer 2-channel stereo track if available in MKV (e.g., Stream #0:2 stereo)
                                    var foundStereoOverride = false
                                    for (group in exo.currentTracks.groups) {
                                        if (group.type == C.TRACK_TYPE_AUDIO) {
                                            val mtg = group.mediaTrackGroup
                                            for (i in 0 until mtg.length) {
                                                if (mtg.getFormat(i).channelCount in 1..2) {
                                                    paramsBuilder.setOverrideForType(TrackSelectionOverride(mtg, i))
                                                    foundStereoOverride = true
                                                    break
                                                }
                                            }
                                        }
                                        if (foundStereoOverride) break
                                    }
                                    paramsBuilder.setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                                } else {
                                    // Keep video playing even if internal container audio track is corrupt
                                    paramsBuilder.setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, true)
                                }

                                exo.trackSelectionParameters = paramsBuilder.build()
                                exo.prepare()
                                if (savedPos > 0L) exo.seekTo(savedPos)
                                exo.playWhenReady = true
                                exo.play()
                                return
                            } catch (e: Exception) {
                                Log.w(TAG, "Could not recover video playback: ${e.message}")
                            }
                        }

                        // 3. Only for genuine audio-only playback files
                        if (isAudioOnlyFile) {
                            val dubbed = _playerState.value.dubbedAudioFile ?: currentDubbedAudioFile
                            if (dubbed != null && dubbed.exists() && dubbed.length() > 44) {
                                Log.i(TAG, "Audio file fallback to dubbed audio track")
                                _playerState.value = _playerState.value.copy(
                                    isAudioOnlyFallback = true,
                                    hasError = false,
                                    errorMessage = null
                                )
                                val audioItem = MediaItem.fromUri(Uri.fromFile(dubbed))
                                val audioSource = DefaultMediaSourceFactory(context, createExtractorsFactory()).createMediaSource(audioItem)
                                dubbedAudioPlayer?.stop()
                                dubbedAudioPlayer?.release()
                                dubbedAudioPlayer = null
                                player?.setMediaSource(audioSource)
                                player?.prepare()
                                player?.play()
                                return
                            }
                        }

                        _playerState.value = _playerState.value.copy(
                            hasError = false,
                            errorMessage = null,
                            isPlaying = false
                        )
                    }
                })
            }
        }
        return player!!
    }

    /**
     * Re-prepares the EXISTING ExoPlayer instance in-place without releasing it,
     * so PlayerScreen's AndroidView (PlayerView) never loses its attached player or SurfaceView.
     */
    private fun reprepareCurrentPlayerInPlace() {
        val exo = player ?: return
        val targetUri = currentVideoUri ?: return
        val savedPos = exo.currentPosition.coerceAtLeast(0L)
        val savedSpeed = _playerState.value.playbackSpeed

        try {
            exo.stop()
            val mediaSourceFactory = DefaultMediaSourceFactory(context, createExtractorsFactory())
            val videoItem = MediaItem.fromUri(targetUri)
            val videoSource = mediaSourceFactory.createMediaSource(videoItem)
            exo.setMediaSource(videoSource)
            exo.setPlaybackSpeed(savedSpeed)
            exo.prepare()
            if (savedPos > 0L) {
                exo.seekTo(savedPos)
                dubbedAudioPlayer?.seekTo(savedPos)
            }
            exo.playWhenReady = true
            exo.play()
            updateAudioVolumes(savedPos)
        } catch (e: Exception) {
            Log.w(TAG, "Failed to reprepare player in-place: ${e.message}")
        }
    }

    private fun initializeDubbedPlayer(): ExoPlayer {
        if (dubbedAudioPlayer == null) {
            val renderersFactory = VlcGradeRenderersFactory(context)
            val mediaSourceFactory = DefaultMediaSourceFactory(context, createExtractorsFactory())
            dubbedAudioPlayer = ExoPlayer.Builder(context, renderersFactory)
                .setMediaSourceFactory(mediaSourceFactory)
                .build().apply {
                repeatMode = Player.REPEAT_MODE_OFF
                playWhenReady = player?.isPlaying ?: true
            }
        }
        return dubbedAudioPlayer!!
    }

    fun setupMedia(
        videoUri: Uri,
        dubbedAudioFile: File?,
        srtFile: File?,
        sourceSrtFile: File? = null
    ) {
        loadSubtitleFiles(srtFile, sourceSrtFile)

        val effectiveDubbedFile = when {
            dubbedAudioFile != null && dubbedAudioFile.exists() && dubbedAudioFile.length() > 44 -> dubbedAudioFile
            dubbedAudioFile != null -> {
                val parent = dubbedAudioFile.parentFile
                val wavFile = if (parent != null) File(parent, "dubbed_bn.wav") else null
                if (wavFile != null && wavFile.exists() && wavFile.length() > 44) wavFile else null
            }
            else -> null
        }

        if (currentVideoUri == videoUri &&
            currentDubbedAudioFile?.absolutePath == effectiveDubbedFile?.absolutePath &&
            player != null &&
            (player?.mediaItemCount ?: 0) > 0
        ) {
            Log.d(TAG, "setupMedia: Video and audio already initialized for $videoUri, ensuring active playback state")
            currentSrtFile = srtFile
            currentSourceSrtFile = sourceSrtFile
            val hasPreRendered = effectiveDubbedFile != null
            val canLive = !hasPreRendered && bnSubtitleEntries.isNotEmpty()
            _playerState.value = _playerState.value.copy(
                hasDubbedAudio = hasPreRendered || canLive,
                isLiveDubbing = canLive,
                hasBanglaSubtitles = bnSubtitleEntries.isNotEmpty(),
                hasEnglishSubtitles = enSubtitleEntries.isNotEmpty(),
                dubbedAudioFile = effectiveDubbedFile
            )
            player?.let { exo ->
                if (exo.playbackState == Player.STATE_IDLE) {
                    errorRecoveryAttempts = 0
                    exo.prepare()
                } else if (exo.playbackState == Player.STATE_ENDED) {
                    exo.seekTo(0)
                }
                exo.playWhenReady = true
                exo.play()
                dubbedAudioPlayer?.let { dp ->
                    if (dp.playbackState == Player.STATE_IDLE) dp.prepare()
                    dp.seekTo(exo.currentPosition.coerceAtLeast(0L))
                    dp.playWhenReady = true
                    dp.play()
                }
                updateAudioVolumes(exo.currentPosition.coerceAtLeast(0L))
            }
            return
        }

        currentVideoUri = videoUri
        currentDubbedAudioFile = effectiveDubbedFile
        currentSrtFile = srtFile
        currentSourceSrtFile = sourceSrtFile
        errorRecoveryAttempts = 0
        decoderSwitchCount = 0
        preferSoftwareVideoDecoder = true
        zeroFramePlayingChecks = 0

        val exo = initializePlayer()
        val mediaSourceFactory = DefaultMediaSourceFactory(context, createExtractorsFactory())

        val isVideoUriAudioFile = videoUri.path?.let { p ->
            p.endsWith(".wav") || p.endsWith(".m4a") || p.endsWith(".mp3") || p.endsWith(".aac")
        } ?: false
        val isVideoUriSameAsDubbed = effectiveDubbedFile != null && videoUri.toString() == Uri.fromFile(effectiveDubbedFile).toString()
        val isAudioOnly = isVideoUriAudioFile || isVideoUriSameAsDubbed

        val hasPreRenderedDubbed = effectiveDubbedFile != null
        val canLiveDub = !hasPreRenderedDubbed && bnSubtitleEntries.isNotEmpty()
        val hasDubbed = hasPreRenderedDubbed || canLiveDub

        val initialAudioChoice = if (hasDubbed) AudioTrackChoice.BANGLA_DUB else AudioTrackChoice.ORIGINAL
        val initialSubChoice = when {
            bnSubtitleEntries.isNotEmpty() -> SubtitleChoice.BANGLA
            enSubtitleEntries.isNotEmpty() -> SubtitleChoice.ENGLISH
            else -> SubtitleChoice.ENGLISH // Allow embedded MKV subtitles when playing directly
        }

        // Set state including dubbedAudioFile BEFORE preparing video source
        _playerState.value = _playerState.value.copy(
            hasDubbedAudio = hasDubbed,
            isLiveDubbing = canLiveDub,
            isAudioOnlyFallback = isAudioOnly,
            hasError = false,
            errorMessage = null,
            audioChoice = initialAudioChoice,
            subtitleChoice = initialSubChoice,
            hasBanglaSubtitles = bnSubtitleEntries.isNotEmpty(),
            hasEnglishSubtitles = enSubtitleEntries.isNotEmpty(),
            dubbedAudioFile = effectiveDubbedFile
        )

        // 1. Setup Video with Original Audio
        exo.trackSelectionParameters = exo.trackSelectionParameters
            .buildUpon()
            .setTrackTypeDisabled(C.TRACK_TYPE_VIDEO, false)
            .setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
            .clearOverridesOfType(C.TRACK_TYPE_AUDIO)
            .build()
        val videoItem = MediaItem.fromUri(videoUri)
        val videoSource = mediaSourceFactory.createMediaSource(videoItem)
        exo.setMediaSource(videoSource)
        exo.prepare()
        exo.playWhenReady = true
        exo.play()

        if (hasPreRenderedDubbed && !isVideoUriSameAsDubbed) {
            val dPlayer = initializeDubbedPlayer()
            val audioItem = MediaItem.fromUri(Uri.fromFile(effectiveDubbedFile!!))
            val audioSource = mediaSourceFactory.createMediaSource(audioItem)
            dPlayer.setMediaSource(audioSource)
            dPlayer.prepare()
            dPlayer.playWhenReady = true
            dPlayer.seekTo(exo.currentPosition.coerceAtLeast(0L))
            dPlayer.play()
            Log.i(TAG, "Initialized secondary dubbed player for real-time dynamic track switching: ${effectiveDubbedFile.absolutePath}")
        } else {
            dubbedAudioPlayer?.stop()
            dubbedAudioPlayer?.release()
            dubbedAudioPlayer = null
        }

        if (canLiveDub) {
            initLiveTts()
            Log.i(TAG, "Initialized Live Real-time TTS Dubbing (0-second wait mode)")
        }

        updateAudioVolumes(exo.currentPosition.coerceAtLeast(0L))
    }

    fun setAudioChoice(choice: AudioTrackChoice) {
        _playerState.value = _playerState.value.copy(audioChoice = choice)
        player?.let { exo ->
            exo.trackSelectionParameters = exo.trackSelectionParameters
                .buildUpon()
                .setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                .build()
            updateAudioVolumes(exo.currentPosition.coerceAtLeast(0L))
        }
    }

    /**
     * Dynamic Alternative Audio Track System:
     * When dialogue is active -> Dubbed Bangla speech plays, original video audio is ducked (or muted).
     * When dialogue is NOT active -> Original video audio (cars, storms, engines, footsteps, music) plays at 100% volume!
     */
    private fun updateAudioVolumes(currentPosMs: Long) {
        val exo = player ?: return
        val dPlayer = dubbedAudioPlayer

        val choice = _playerState.value.audioChoice
        val hasDubbed = _playerState.value.hasDubbedAudio
        val isLive = _playerState.value.isLiveDubbing

        if (!hasDubbed || _playerState.value.isAudioOnlyFallback) {
            exo.volume = 1.0f
            return
        }

        val isSpeaking = isDialogueActive(currentPosMs)
        if (_playerState.value.isSpeechActive != isSpeaking) {
            _playerState.value = _playerState.value.copy(isSpeechActive = isSpeaking)
        }

        when (choice) {
            AudioTrackChoice.BANGLA_DUB -> {
                if (isSpeaking) {
                    exo.volume = 0.32f
                    if (isLive) {
                        val activeCue = bnSubtitleEntries.firstOrNull { currentPosMs in it.startMs..it.endMs }
                        if (activeCue != null && activeCue.startMs != lastSpokenCueStartMs) {
                            lastSpokenCueStartMs = activeCue.startMs
                            if (isLiveTtsReady && _playerState.value.isPlaying) {
                                val params = android.os.Bundle()
                                params.putFloat(android.speech.tts.TextToSpeech.Engine.KEY_PARAM_VOLUME, 1.0f)
                                liveTts?.speak(activeCue.text, android.speech.tts.TextToSpeech.QUEUE_FLUSH, params, "cue_${activeCue.startMs}")
                            }
                        }
                    } else {
                        dPlayer?.volume = 1.0f
                    }
                } else {
                    exo.volume = if (bnSubtitleEntries.isEmpty()) 0.35f else 1.0f
                    if (!isLive) {
                        dPlayer?.volume = 1.0f
                    }
                }
            }
            AudioTrackChoice.ORIGINAL -> {
                exo.volume = 1.0f
                dPlayer?.volume = 0.0f
                if (isLive) liveTts?.stop()
            }
            AudioTrackChoice.BANGLA_ONLY -> {
                exo.volume = 0.0f
                if (isLive) {
                    val activeCue = bnSubtitleEntries.firstOrNull { currentPosMs in it.startMs..it.endMs }
                    if (activeCue != null && activeCue.startMs != lastSpokenCueStartMs) {
                        lastSpokenCueStartMs = activeCue.startMs
                        if (isLiveTtsReady && _playerState.value.isPlaying) {
                            val params = android.os.Bundle()
                            params.putFloat(android.speech.tts.TextToSpeech.Engine.KEY_PARAM_VOLUME, 1.0f)
                            liveTts?.speak(activeCue.text, android.speech.tts.TextToSpeech.QUEUE_FLUSH, params, "cue_${activeCue.startMs}")
                        }
                    }
                } else {
                    dPlayer?.volume = 1.0f
                }
            }
        }
    }

    private fun isDialogueActive(posMs: Long): Boolean {
        if (bnSubtitleEntries.isEmpty()) return true
        return bnSubtitleEntries.any { posMs in (it.startMs - 40L)..(it.endMs + 80L) }
    }

    fun setSubtitleChoice(choice: SubtitleChoice) {
        _playerState.value = _playerState.value.copy(subtitleChoice = choice)
        updateCurrentSubtitleText(_playerState.value.currentPositionMs)
    }

    fun setResizeMode(mode: ResizeModeChoice) {
        _playerState.value = _playerState.value.copy(resizeMode = mode)
    }

    fun cycleResizeMode(): ResizeModeChoice {
        val modes = ResizeModeChoice.values()
        val currentIdx = modes.indexOf(_playerState.value.resizeMode)
        val nextMode = modes[(currentIdx + 1) % modes.size]
        _playerState.value = _playerState.value.copy(resizeMode = nextMode)
        return nextMode
    }

    fun setPlaybackSpeed(speed: Float) {
        player?.setPlaybackSpeed(speed)
        dubbedAudioPlayer?.setPlaybackSpeed(speed)
        _playerState.value = _playerState.value.copy(playbackSpeed = speed)
    }

    fun seekBy(deltaMs: Long) {
        val exo = player ?: return
        val current = exo.currentPosition
        val target = (current + deltaMs).coerceIn(0L, exo.duration.coerceAtLeast(0L))
        seekTo(target)
    }

    fun seekTo(positionMs: Long) {
        liveTts?.stop()
        lastSpokenCueStartMs = -1L
        player?.seekTo(positionMs)
        dubbedAudioPlayer?.seekTo(positionMs)
        updateAudioVolumes(positionMs)
    }

    fun togglePlayPause() {
        val exo = player ?: initializePlayer()
        val isCurrentlyPlaying = exo.isPlaying || (exo.playWhenReady && (exo.playbackState == Player.STATE_READY || exo.playbackState == Player.STATE_BUFFERING))
        val shouldPlay = !isCurrentlyPlaying

        if (shouldPlay) {
            if (exo.playbackState == Player.STATE_IDLE) {
                exo.prepare()
            } else if (exo.playbackState == Player.STATE_ENDED) {
                exo.seekTo(0)
            }
            exo.playWhenReady = true
            exo.play()
            dubbedAudioPlayer?.let { dp ->
                if (dp.playbackState == Player.STATE_IDLE) dp.prepare()
                dp.seekTo(exo.currentPosition.coerceAtLeast(0L))
                dp.playWhenReady = true
                dp.play()
            }
            _playerState.value = _playerState.value.copy(isPlaying = true)
        } else {
            exo.playWhenReady = false
            exo.pause()
            dubbedAudioPlayer?.playWhenReady = false
            dubbedAudioPlayer?.pause()
            liveTts?.stop()
            _playerState.value = _playerState.value.copy(isPlaying = false)
        }
    }

    fun toggleLock() {
        _playerState.value = _playerState.value.copy(isLocked = !_playerState.value.isLocked)
    }

    fun updateProgress() {
        val exo = player ?: return
        val pos = exo.currentPosition.coerceAtLeast(0L)
        val dur = exo.duration.coerceAtLeast(0L)
        _playerState.value = _playerState.value.copy(
            currentPositionMs = pos,
            durationMs = dur
        )

        // Wall-clock stall watchdog: works even when ExoPlayer is stuck in STATE_BUFFERING at pos == 0L
        // waiting for the first video frame!
        val isAttemptingPlayback = exo.playWhenReady &&
                (exo.playbackState == Player.STATE_BUFFERING || exo.playbackState == Player.STATE_READY)
        if (isAttemptingPlayback && decoderSwitchCount < 2 && !_playerState.value.isAudioOnlyFallback) {
            val hasSelectedVideoTrack = exo.currentTracks.groups.any { it.type == C.TRACK_TYPE_VIDEO && it.isSelected }
            val renderedFrames = exo.videoDecoderCounters?.renderedOutputBufferCount ?: 0
            if (hasSelectedVideoTrack && renderedFrames == 0) {
                zeroFramePlayingChecks++
                if (zeroFramePlayingChecks >= 6) {
                    zeroFramePlayingChecks = 0
                    decoderSwitchCount++
                    preferSoftwareVideoDecoder = !preferSoftwareVideoDecoder
                    Log.w(TAG, "Watchdog: 0 video frames rendered; switching preferSoftwareVideoDecoder=$preferSoftwareVideoDecoder in-place")
                    reprepareCurrentPlayerInPlace()
                    return
                }
            } else if (renderedFrames > 0) {
                zeroFramePlayingChecks = 0
            }
        }

        // Resync secondary player if clock drift exceeds threshold
        val dPlayer = dubbedAudioPlayer
        if (dPlayer != null && exo.isPlaying) {
            if (!dPlayer.isPlaying && dPlayer.playbackState == Player.STATE_READY) {
                dPlayer.play()
            }
            val drift = abs(pos - dPlayer.currentPosition)
            if (drift > 120L) {
                dPlayer.seekTo(pos)
            }
        }

        updateCurrentSubtitleText(pos)
        updateAudioVolumes(pos)
    }

    private fun updateCurrentSubtitleText(currentPositionMs: Long) {
        val choice = _playerState.value.subtitleChoice
        if (choice == SubtitleChoice.OFF) {
            if (_playerState.value.activeSubtitleText != null) {
                _playerState.value = _playerState.value.copy(activeSubtitleText = null)
            }
            return
        }

        val entries = when (choice) {
            SubtitleChoice.BANGLA -> bnSubtitleEntries
            SubtitleChoice.ENGLISH -> enSubtitleEntries
            SubtitleChoice.OFF -> emptyList()
        }

        if (entries.isEmpty()) {
            val fallbackText = embeddedCueText
            if (_playerState.value.activeSubtitleText != fallbackText) {
                _playerState.value = _playerState.value.copy(activeSubtitleText = fallbackText)
            }
            return
        }

        val active = entries.firstOrNull { currentPositionMs in it.startMs..it.endMs }
        val text = active?.text
        if (_playerState.value.activeSubtitleText != text) {
            _playerState.value = _playerState.value.copy(activeSubtitleText = text)
        }
    }

    private fun loadSubtitleFiles(bnSrtFile: File?, enSrtFile: File?) {
        bnSubtitleEntries.clear()
        enSubtitleEntries.clear()

        bnSrtFile?.let { parseSrtIntoList(it, bnSubtitleEntries) }
        enSrtFile?.let { parseSrtIntoList(it, enSubtitleEntries) }
    }

    private fun parseSrtIntoList(srtFile: File, targetList: MutableList<ParsedSubtitle>) {
        if (!srtFile.exists() || srtFile.length() == 0L) return
        try {
            val content = srtFile.readText(Charsets.UTF_8)
            val cues = com.example.subtitle.SubtitleExtractor.parseSubtitleContent(content)
            for (cue in cues) {
                if (cue.text.isNotBlank()) {
                    targetList.add(ParsedSubtitle(cue.startMs, cue.endMs, cue.text.replace("\n", " ").trim()))
                }
            }
            Log.d(TAG, "Loaded ${targetList.size} subtitle entries from ${srtFile.name}")
        } catch (e: Exception) {
            Log.w(TAG, "Error parsing SRT file: ${srtFile.name}", e)
        }
    }

    fun release() {
        player?.release()
        player = null
        dubbedAudioPlayer?.release()
        dubbedAudioPlayer = null
        try {
            liveTts?.stop()
            liveTts?.shutdown()
            liveTts = null
        } catch (_: Exception) {}
    }

    private fun parseIntentData(intent: Intent?) {
        if (intent == null) return
        val rawUriStr = intent.getStringExtra(EXTRA_VIDEO_URI)

        videoUri = when {

            if (uri.scheme == "content") {
                try {
                    contentResolver.takePersistableUriPermission(

                        Intent.FLAG_GRANT_READ_URI_PERMISSION
                    )
                } catch (_: Exception) {}
            }
        }

        val resolvedDisplayName = videoUri?.let { queryDisplayName(it) }
        videoTitle = intent.getStringExtra(EXTRA_TITLE)?.takeIf { it.isNotBlank() }
            ?: resolvedDisplayName

        srtPath = intent.getStringExtra(EXTRA_SRT_PATH)
        launchedWithExplicitDub = !dubbedAudioPath.isNullOrBlank()

        val titleForLookup = resolvedDisplayName ?: videoTitle ?: ""
        if (titleForLookup.isNotBlank() && (dubbedAudioPath.isNullOrBlank() || srtPath.isNullOrBlank())) {
            val baseTitle = titleForLookup.substringBeforeLast(".")
            val publicDownloads = Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS)
            val fileTranslatoDir = File(publicDownloads, "FileTranslato")
            val banglaDubbingDir = File(publicDownloads, "BanglaDubbing")

            if (dubbedAudioPath.isNullOrBlank()) {
                val candidateAudio = File(fileTranslatoDir, "${baseTitle}_bangla_dubbed_audio.wav")
                if (candidateAudio.exists() && candidateAudio.length() > 44) {
                    dubbedAudioPath = candidateAudio.absolutePath
                }
            }
            if (srtPath.isNullOrBlank()) {
                val candidateSrt1 = File(fileTranslatoDir, "${baseTitle}_bangla.srt")
                val candidateSrt2 = File(banglaDubbingDir, "${baseTitle}_bangla.srt")
                if (candidateSrt1.exists() && candidateSrt1.length() > 0) {
                    srtPath = candidateSrt1.absolutePath
                } else if (candidateSrt2.exists() && candidateSrt2.length() > 0) {
                    srtPath = candidateSrt2.absolutePath

    private fun initializePlayer() {
        if (player != null) return
        val targetUri = videoUri ?: run {
            Toast.makeText(this, "ভিডিও ফাইল খুঁজে পাওয়া যায়নি", Toast.LENGTH_SHORT).show()
            finish()
            return
        }

        try {
            val renderersFactory = VlcGradeRenderersFactory(this) { preferSoftwareVideoDecoder }

            trackSelector = DefaultTrackSelector(this).apply {
                parameters = buildUponParameters()
                    .setExceedRendererCapabilitiesIfNecessary(true)
                    .setExceedVideoConstraintsIfNecessary(true)
                    .setExceedAudioConstraintsIfNecessary(true)
                    .setAllowAudioMixedMimeTypeAdaptiveness(true)
                    .setAllowAudioMixedSampleRateAdaptiveness(true)
                    .setAllowAudioMixedChannelCountAdaptiveness(true)
                    .setAllowVideoMixedMimeTypeAdaptiveness(true)
                    .setAllowVideoNonSeamlessAdaptiveness(true)
                    .setTrackTypeDisabled(C.TRACK_TYPE_VIDEO, false)
                    .setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                    .build()
            }

            val audioAttributes = AudioAttributes.Builder()
                .setContentType(C.AUDIO_CONTENT_TYPE_MOVIE)
                .setUsage(C.USAGE_MEDIA)
                .build()

            val dataSourceFactory = DefaultDataSource.Factory(this)
            val mediaSourceFactory = DefaultMediaSourceFactory(dataSourceFactory, createExtractorsFactory())

            player = ExoPlayer.Builder(this, renderersFactory)
                .setTrackSelector(trackSelector!!)
                .setMediaSourceFactory(mediaSourceFactory)
                .setAudioAttributes(audioAttributes, /* handleAudioFocus= */ true)
                .build().apply {
                    repeatMode = Player.REPEAT_MODE_OFF
                    playWhenReady = startPlayWhenReady
                    addListener(PlayerEventListener())
                }

            playerView.player = player
            playerView.resizeMode = resizeModes[currentResizeModeIndex]

            val mainMediaSource = mediaSourceFactory.createMediaSource(buildMainMediaItem(targetUri))
            player?.setMediaSource(mainMediaSource)
            player?.prepare()

            val dubbedFile = dubbedAudioPath?.let { File(it) }
            if (dubbedFile != null && dubbedFile.exists() && dubbedFile.length() > 44) {
                val dubbedRenderersFactory = VlcGradeRenderersFactory(this)
                dubbedPlayer = ExoPlayer.Builder(this, dubbedRenderersFactory)
                    .setMediaSourceFactory(mediaSourceFactory)
                    .build().apply {
                        val audioItem = MediaItem.fromUri(Uri.fromFile(dubbedFile))
                        val audioSource = mediaSourceFactory.createMediaSource(audioItem)
                        setMediaSource(audioSource)
                        prepare()
                        playWhenReady = startPlayWhenReady
                        seekTo(startPositionMs)
                    }
                currentAudioMode = if (launchedWithExplicitDub) PlayerAudioMode.SMART_DUB else PlayerAudioMode.ORIGINAL_ONLY
                Log.i(TAG, "Dubbed audio player initialized (mode=$currentAudioMode): ${dubbedFile.absolutePath}")
            } else {
                dubbedPlayer = null
                currentAudioMode = PlayerAudioMode.ORIGINAL_ONLY
            }

            applyAudioModeVolumes(startPositionMs)

            if (startPositionMs > 0) {
                player?.seekTo(startPositionMs)
                dubbedPlayer?.seekTo(startPositionMs)
            }

            if (!isAdShownOrSkipped) {
                isAdShownOrSkipped = true
                isWaitingForAd = true
                AdMobManager.showRewardedAd(this) {
                    isWaitingForAd = false
                    startPlayWhenReady = true
                    if (player == null) {
                        initializePlayer()
                    }
                    player?.playWhenReady = true
                    player?.play()
                    dubbedPlayer?.playWhenReady = true
                    dubbedPlayer?.play()
                    btnPlayPause.setImageResource(R.drawable.ic_pause)
                }
            }

            handler.removeCallbacks(updateProgressRunnable)
            handler.post(updateProgressRunnable)
            resetControlsHideTimer()

            Log.i(TAG, "ExoPlayer initialized with SurfaceView, 10-bit HEVC decoding, and 5.1 HE-AAC stereo downmixing")
        } catch (e: Exception) {
            Log.e(TAG, "ExoPlayer initialization failed: ${e.message}", e)
            Toast.makeText(this, "ভিডিও প্লেয়ার শুরু করতে সমস্যা হয়েছে: ${e.message}", Toast.LENGTH_LONG).show()
        }
    }

    private fun repreparePlayerInPlace() {
        val p = player ?: return
        val targetUri = videoUri ?: return
        val savedPos = p.currentPosition.coerceAtLeast(0L)
        try {
            p.stop()
            val dataSourceFactory = DefaultDataSource.Factory(this)
            val mediaSourceFactory = DefaultMediaSourceFactory(dataSourceFactory, createExtractorsFactory())
            val mainMediaSource = mediaSourceFactory.createMediaSource(buildMainMediaItem(targetUri))
            p.setMediaSource(mainMediaSource)
            p.prepare()
            if (savedPos > 0L) {
                p.seekTo(savedPos)
                dubbedPlayer?.seekTo(savedPos)
            }
            p.playWhenReady = true
            p.play()
        } catch (e: Exception) {
            Log.w(TAG, "Failed to reprepare player in-place: ${e.message}")
        }
    }

    private fun releasePlayer() {
        player?.let { p ->
            startPositionMs = p.currentPosition.coerceAtLeast(0L)
            startPlayWhenReady = if (isWaitingForAd) true else p.playWhenReady
            p.stop()
            p.release()
        }
        player = null
        trackSelector = null
        playerView.player = null

        dubbedPlayer?.let { dp ->
            dp.stop()
            dp.release()
        }
        dubbedPlayer = null

        handler.removeCallbacks(updateProgressRunnable)
        handler.removeCallbacks(hideControlsRunnable)
    }

        super.onResume()
        setupImmersiveMode()
        if (Build.VERSION.SDK_INT < 24 || player == null) {
            initializePlayer()
        }
    }

    override fun onPause() {
        super.onPause()
        if (Build.VERSION.SDK_INT < 24) {
            releasePlayer()
        } else {
            player?.pause()
            dubbedPlayer?.pause()
        }
    }

    override fun onStop() {
        super.onStop()
        if (Build.VERSION.SDK_INT >= 24) {
            releasePlayer()
        }
    }

    override fun onDestroy() {
        super.onDestroy()
        releasePlayer()
    }

    private inner class PlayerEventListener : Player.Listener {
        override fun onPlaybackStateChanged(playbackState: Int) {
            when (playbackState) {
                Player.STATE_BUFFERING -> {
                    progressBuffering.visibility = View.VISIBLE
                }
                Player.STATE_READY -> {
                    progressBuffering.visibility = View.GONE
                    errorRecoveryAttempts = 0
                    zeroFrameChecks = 0
                    updateTimeline()
                }
                Player.STATE_ENDED -> {
                    progressBuffering.visibility = View.GONE
                    btnPlayPause.setImageResource(R.drawable.ic_play_arrow)
                    showControls()
                }
                Player.STATE_IDLE -> {
                    progressBuffering.visibility = View.GONE
                }
            }
        }

        override fun onIsPlayingChanged(isPlaying: Boolean) {
            btnPlayPause.setImageResource(if (isPlaying) R.drawable.ic_pause else R.drawable.ic_play_arrow)
            if (isPlaying) {
                dubbedPlayer?.play()
                resetControlsHideTimer()
            } else {
                dubbedPlayer?.pause()
            }
        }

        override fun onPlayerError(error: PlaybackException) {
            progressBuffering.visibility = View.GONE
            Log.e(TAG, "Player playback error: [${error.errorCodeName}] (code=${error.errorCode}) ${error.message}", error)

            val isVideoCodecError = error.errorCode == PlaybackException.ERROR_CODE_DECODER_INIT_FAILED ||
                    error.errorCode == PlaybackException.ERROR_CODE_DECODING_FAILED ||
                    error.errorCode == PlaybackException.ERROR_CODE_DECODING_FORMAT_EXCEEDS_CAPABILITIES ||
                    error.errorCode == PlaybackException.ERROR_CODE_DECODING_FORMAT_UNSUPPORTED

            if (isVideoCodecError && decoderSwitchCount < 2) {
                decoderSwitchCount++
                preferSoftwareVideoDecoder = !preferSoftwareVideoDecoder
                Log.w(TAG, "Video decoder error; toggling preferSoftwareVideoDecoder=$preferSoftwareVideoDecoder in-place")
                repreparePlayerInPlace()
                return
            }

            if (errorRecoveryAttempts < 3) {
                errorRecoveryAttempts++
                Log.w(TAG, "Attempting playback recovery ($errorRecoveryAttempts/3)")
                try {
                    val p = player ?: return
                    val savedPos = p.currentPosition.coerceAtLeast(0L)
                    trackSelector?.let { ts ->
                        val builder = ts.buildUponParameters()
                            .setTrackTypeDisabled(C.TRACK_TYPE_VIDEO, false)
                        if (errorRecoveryAttempts == 1) {
                            var foundStereo = false
                            for (group in p.currentTracks.groups) {
                                if (group.type == C.TRACK_TYPE_AUDIO) {
                                    val mtg = group.mediaTrackGroup
                                    for (i in 0 until mtg.length) {
                                        if (mtg.getFormat(i).channelCount in 1..2) {
                                            builder.setOverrideForType(TrackSelectionOverride(mtg, i))
                                            foundStereo = true
                                            break
                                        }
                                    }
                                }
                                if (foundStereo) break
                            }
                            builder.setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                        } else {
                            builder.setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, true)
                        }
                        ts.setParameters(builder)
                    }
                    p.prepare()
                    if (savedPos > 0L) p.seekTo(savedPos)
                    p.playWhenReady = true
                    p.play()
                } catch (e: Exception) {
                    Log.e(TAG, "Failed to recover playback: ${e.message}")
                }
            }
        }
    }

        val subtitle: String,
        val mode: PlayerAudioMode? = null,

    private fun showAudioTrackSelectionDialog() {
        val p = player ?: return
        val ts = trackSelector ?: return
        val currentTracks = p.currentTracks

        val audioOptions = mutableListOf<AudioOptionItem>()
        var selectedItemIndex = 0

        if (dubbedPlayer != null) {
            for (mode in PlayerAudioMode.values()) {
                if (currentAudioMode == mode) {
                    selectedItemIndex = audioOptions.size
                }
                audioOptions.add(AudioOptionItem(mode.label, mode.description, mode = mode))
            }
        }

        var internalTrackNum = 1
        for (group in currentTracks.groups) {
            if (group.type == C.TRACK_TYPE_AUDIO) {
                val mediaTrackGroup = group.mediaTrackGroup
                for (i in 0 until mediaTrackGroup.length) {
                    val format = mediaTrackGroup.getFormat(i)
                    val lang = format.language ?: "und"
                    val channels = format.channelCount
                    val chLabel = when (channels) {
                        6 -> "5.1"
                        8 -> "7.1"
                        2 -> "Stereo"
                        1 -> "Mono"
                        else -> "${channels}-ch"
                    }
                    val mime = format.sampleMimeType?.substringAfterLast("/")?.uppercase(Locale.ROOT) ?: "AUDIO"
                    val langName = when (lang.lowercase(Locale.ROOT)) {
                        "ben", "bn" -> "বাংলা (Bangla)"
                        "hin", "hi" -> "Hindi (হিন্দি)"
                        "mal", "ml" -> "Malayalam (মালায়ালাম)"
                        "tam", "ta" -> "Tamil (তামিল)"
                        "tel", "te" -> "Telugu (তেলেগু)"
                        "eng", "en" -> "English (ইংরেজি)"
                        else -> lang.uppercase(Locale.ROOT)
                    }
                    val label = if (!format.label.isNullOrBlank()) {
                        "${format.label} [$langName • $chLabel $mime]"
                    } else {
                        "মূল অডিও #$internalTrackNum: $langName ($chLabel $mime)"
                    }
                    internalTrackNum++

                    if (dubbedPlayer == null && group.isTrackSelected(i)) {
                        selectedItemIndex = audioOptions.size
                    }
                    audioOptions.add(
                        AudioOptionItem(
                            title = label,
                            subtitle = "ভিডিওর অভ্যন্তরীণ অডিও ট্র্যাক",
                            trackOverride = TrackSelectionOverride(mediaTrackGroup, i)
                        )
                    )
                }
            }
        }

        if (audioOptions.isEmpty()) {
            Toast.makeText(this, "এই ভিডিওতে বিকল্প অডিও ট্র্যাক পাওয়া যায়নি", Toast.LENGTH_SHORT).show()
            return
        }

        val items = audioOptions.map { it.title }.toTypedArray()
        AlertDialog.Builder(this)
            .setTitle("অডিও ট্র্যাক নির্বাচন করুন (Smart Audio & Multi-Track)")
            .setSingleChoiceItems(items, selectedItemIndex) { dialog, which ->
                val chosen = audioOptions[which]
                if (chosen.mode != null) {
                    setAudioMode(chosen.mode)
                } else if (chosen.trackOverride != null) {
                    ts.setParameters(
                        ts.buildUponParameters()
                            .setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                            .setOverrideForType(chosen.trackOverride)
                    )
                    if (dubbedPlayer != null && currentAudioMode == PlayerAudioMode.BANGLA_ONLY) {
                        setAudioMode(PlayerAudioMode.SMART_DUB)
                    } else {
                        applyAudioModeVolumes(p.currentPosition.coerceAtLeast(0L))
                    }
                }
                Toast.makeText(this, "নির্বাচিত অডিও: ${chosen.title}", Toast.LENGTH_SHORT).show()
                dialog.dismiss()
            }
            .setNegativeButton("বাতিল", null)
            .show()
    }

    private fun setAudioMode(mode: PlayerAudioMode) {
        currentAudioMode = mode
        applyAudioModeVolumes(player?.currentPosition?.coerceAtLeast(0L) ?: 0L)
    }

    private fun applyAudioModeVolumes(currentPosMs: Long = player?.currentPosition?.coerceAtLeast(0L) ?: 0L) {
        val p = player ?: return
        val dp = dubbedPlayer
        if (dp == null) {
            p.volume = 1.0f
            return
        }

        when (currentAudioMode) {
            PlayerAudioMode.SMART_DUB -> {
                val isSpeaking = if (bnSubtitleCues.isEmpty()) {
                    true
                } else {
                    bnSubtitleCues.any { currentPosMs in (it.startMs - 40L)..(it.endMs + 80L) }
                }
                p.volume = if (isSpeaking) 0.32f else 1.0f
                dp.volume = 1.0f
            }
            PlayerAudioMode.ORIGINAL_ONLY -> {
                p.volume = 1.0f
                dp.volume = 0.0f
            }
            PlayerAudioMode.BANGLA_ONLY -> {
                p.volume = 0.0f
                dp.volume = 1.0f
            }
        }
    }

    private fun showSubtitleTrackSelectionDialog() {
        val p = player ?: return

        val subtitleOptions = mutableListOf<Pair<String, TrackSelectionOverride?>>()

    private fun togglePlayPause() {
        val p = player ?: run {
            initializePlayer()
            return
        }
        if (p.isPlaying) {
            p.pause()
            dubbedPlayer?.pause()
        } else {
            if (p.playbackState == Player.STATE_IDLE) {
                p.prepare()
            } else if (p.playbackState == Player.STATE_ENDED) {
                p.seekTo(0)
            }
            p.playWhenReady = true
            p.play()
            dubbedPlayer?.let { dp ->
                if (dp.playbackState == Player.STATE_IDLE) dp.prepare()
                dp.seekTo(p.currentPosition.coerceAtLeast(0L))
                dp.playWhenReady = true
                dp.play()
            }
        }
    }

    private fun updateTimeline() {
        val p = player ?: return
        val currentMs = p.currentPosition.coerceAtLeast(0L)
        val durationMs = p.duration.coerceAtLeast(0L)

        txtTimeCurrent.text = formatTime(currentMs)
        if (durationMs > 0) {
            txtTimeDuration.text = formatTime(durationMs)
            val progress = ((currentMs * 1000L) / durationMs).toInt()
            seekBar.progress = progress
        }

        // Wall-clock stall watchdog: works even when ExoPlayer is stuck in STATE_BUFFERING at 00:00
        val isAttemptingPlayback = p.playWhenReady &&
                (p.playbackState == Player.STATE_BUFFERING || p.playbackState == Player.STATE_READY)
        if (isAttemptingPlayback && decoderSwitchCount < 2) {
            val hasVideo = p.currentTracks.groups.any { it.type == C.TRACK_TYPE_VIDEO && it.isSelected }
            val renderedFrames = p.videoDecoderCounters?.renderedOutputBufferCount ?: 0
            if (hasVideo && renderedFrames == 0) {
                zeroFrameChecks++
                if (zeroFrameChecks >= 4) {
                    zeroFrameChecks = 0
                    decoderSwitchCount++
                    preferSoftwareVideoDecoder = !preferSoftwareVideoDecoder
                    Log.w(TAG, "Watchdog: 0 video frames rendered; switching preferSoftwareVideoDecoder=$preferSoftwareVideoDecoder in-place")
                    repreparePlayerInPlace()
                    return
                }
            } else if (renderedFrames > 0) {
                zeroFrameChecks = 0
            }
        }

        dubbedPlayer?.let { dp ->
            if (p.isPlaying && !dp.isPlaying && dp.playbackState == Player.STATE_READY) {
                dp.play()
            }
            val drift = abs(currentMs - dp.currentPosition)
            if (drift > 120L) {
                dp.seekTo(currentMs)
            }
        }

        applyAudioModeVolumes(currentMs)
    }

    private fun formatTime(ms: Long): String {
        val totalSeconds = (ms / 1000).coerceAtLeast(0)
        val seconds = totalSeconds % 60
        val minutes = (totalSeconds / 60) % 60
        val hours = totalSeconds / 3600
        return if (hours > 0) {
            String.format(Locale.ROOT, "%d:%02d:%02d", hours, minutes, seconds)
        } else {
            String.format(Locale.ROOT, "%02d:%02d", minutes, seconds)
        }
    }

    @SuppressLint("ClickableViewAccessibility")
    private fun setupGestureDetector() {
        gestureDetector = GestureDetector(this, object : GestureDetector.SimpleOnGestureListener() {
            override fun onSingleTapConfirmed(e: MotionEvent): Boolean {
                if (isControlsLocked) {
                    btnUnlock.visibility = if (btnUnlock.visibility == View.VISIBLE) View.GONE else View.VISIBLE
                } else {
                    if (controlsContainer.visibility == View.VISIBLE) {
                        hideControls()
                    } else {
                        showControls()
                    }
                }
                return true
            }

            override fun onDoubleTap(e: MotionEvent): Boolean {
                if (isControlsLocked) return true
                val width = playerRoot.width.toFloat()
                if (width <= 0f) return true

                if (e.x < width / 3f) {
                    btnReplay10.performClick()
                } else if (e.x > (width * 2f) / 3f) {
                    btnForward10.performClick()
                } else {
                    togglePlayPause()
                }
                return true
            }
        })

        var initialTouchX = 0f
        var initialTouchY = 0f
        var isSwipeGesture = false
        var gestureMode = 0 // 1 = Brightness, 2 = Volume, 3 = Seek

        playerRoot.setOnTouchListener { _, event ->
            gestureDetector.onTouchEvent(event)

            if (isControlsLocked) return@setOnTouchListener true

            when (event.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    initialTouchX = event.x
                    initialTouchY = event.y
                    isSwipeGesture = false
                    gestureMode = 0
                }
                MotionEvent.ACTION_MOVE -> {
                    val deltaX = event.x - initialTouchX
                    val deltaY = event.y - initialTouchY
                    val screenWidth = playerRoot.width.toFloat().coerceAtLeast(1f)
                    val screenHeight = playerRoot.height.toFloat().coerceAtLeast(1f)

                    if (!isSwipeGesture && (abs(deltaX) > 40f || abs(deltaY) > 40f)) {
                        isSwipeGesture = true
                        gestureMode = if (abs(deltaX) > abs(deltaY)) {
                            3
                        } else if (initialTouchX < screenWidth / 2f) {
                            1
                        } else {
                            2
                        }
                    }

                    if (isSwipeGesture) {
                        when (gestureMode) {
                            1 -> adjustBrightness(-deltaY / screenHeight)
                            2 -> adjustVolume(-deltaY / screenHeight)
                            3 -> previewSeek(deltaX, screenWidth)
                        }
                    }
                }
                MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
                    if (isSwipeGesture && gestureMode == 3) {
                        commitSeek()
                    }
                    handler.postDelayed({ hudContainer.visibility = View.GONE }, 1200L)
                }
            }
            true
        }
    }

    private fun adjustBrightness(factorDelta: Float) {
        val lp = window.attributes
        var current = if (lp.screenBrightness < 0f) 0.5f else lp.screenBrightness
        current = (current + factorDelta * 0.4f).coerceIn(0.01f, 1.0f)
        lp.screenBrightness = current
        window.attributes = lp

        val percent = (current * 100).toInt()
        hudIcon.setImageResource(R.drawable.ic_brightness)
        hudText.text = "$percent%"
        hudProgress.progress = percent
        hudContainer.visibility = View.VISIBLE
    }

    private fun adjustVolume(factorDelta: Float) {
        val maxVol = audioManager.getStreamMaxVolume(AudioManager.STREAM_MUSIC).coerceAtLeast(1)
        val currentVol = audioManager.getStreamVolume(AudioManager.STREAM_MUSIC)
        val deltaSteps = (factorDelta * maxVol * 0.5f).toInt()
        val newVol = (currentVol + deltaSteps).coerceIn(0, maxVol)

        if (newVol != currentVol) {
            audioManager.setStreamVolume(AudioManager.STREAM_MUSIC, newVol, 0)
        }

        val percent = ((newVol.toFloat() / maxVol.toFloat()) * 100).toInt()
        hudIcon.setImageResource(R.drawable.ic_volume_up)
        hudText.text = "$percent%"
        hudProgress.progress = percent
        hudContainer.visibility = View.VISIBLE
    }

    private var pendingSeekPositionMs = 0L

    private fun previewSeek(deltaX: Float, screenWidth: Float) {
        val p = player ?: return
        val duration = p.duration
        if (duration <= 0) return

        val scrubMs = ((deltaX / screenWidth) * 90_000L).toLong()
        pendingSeekPositionMs = (p.currentPosition + scrubMs).coerceIn(0L, duration)

        hudIcon.setImageResource(R.drawable.ic_aspect_ratio)
        hudText.text = "${formatTime(pendingSeekPositionMs)} / ${formatTime(duration)}"
        hudProgress.progress = ((pendingSeekPositionMs * 100L) / duration).toInt()
        hudContainer.visibility = View.VISIBLE
    }

    private fun commitSeek() {
        player?.let { p ->
            if (pendingSeekPositionMs in 0L..p.duration) {
                p.seekTo(pendingSeekPositionMs)
                dubbedPlayer?.seekTo(pendingSeekPositionMs)
            }
        }
    }

    private fun showSeekHud(targetMs: Long, deltaMs: Long) {
        val p = player ?: return
        val sign = if (deltaMs >= 0) "+10s" else "-10s"
        hudIcon.setImageResource(if (deltaMs >= 0) R.drawable.ic_forward_10 else R.drawable.ic_replay_10)
        hudText.text = "$sign (${formatTime(targetMs)})"
        hudProgress.progress = if (p.duration > 0) ((targetMs * 100L) / p.duration).toInt() else 0
        hudContainer.visibility = View.VISIBLE
        handler.removeCallbacks(hideControlsRunnable)
        handler.postDelayed({ hudContainer.visibility = View.GONE }, 1200L)
    }

    @Test
    fun testSubRipSrtParsing() {
        val srtContent = """
            1
            00:00:01,200 --> 00:00:04,500
            Hello world! Welcome to the show.

            2
            00:00:05,100 --> 00:00:08,200
            This is an English dialogue line.
        """.trimIndent()

        val cues = SubtitleExtractor.parseSubtitleContent(srtContent)
        assertEquals(2, cues.size)
        assertEquals(1200L, cues[0].startMs)
        assertEquals(4500L, cues[0].endMs)
        assertEquals("Hello world! Welcome to the show.", cues[0].text)

        assertEquals(5100L, cues[1].startMs)
        assertEquals(8200L, cues[1].endMs)
        assertEquals("This is an English dialogue line.", cues[1].text)
    }

            <b>Enjoy the movie.</b>

    @Test
    fun testExportToSrt() {
        val tempFile = File.createTempFile("test_export", ".srt")

    }

    @Test
    fun testUniversalStereoAudioProcessorHandles5Point1FloatAndStereo() {
        val processor = UniversalStereoAudioProcessor()

        // 1. Standard 16-bit Stereo should pass through (isActive == false) without throwing
        val stereo16Format = AudioProcessor.AudioFormat(48000, 2, C.ENCODING_PCM_16BIT)
        val outStereo16 = processor.configure(stereo16Format)
        assertEquals(AudioProcessor.AudioFormat.NOT_SET, outStereo16)
        assertFalse(processor.isActive)

        // 2. 5.1 Surround (6 channels) Float PCM (fltp, like HE-AAC 5.1 in MKV) must configure to 2ch 16-bit PCM
        val surround51Float = AudioProcessor.AudioFormat(48000, 6, C.ENCODING_PCM_FLOAT)
        val outSurround = processor.configure(surround51Float)
        processor.flush()
        assertTrue(processor.isActive)
        assertEquals(48000, outSurround.sampleRate)
        assertEquals(2, outSurround.channelCount)
        assertEquals(C.ENCODING_PCM_16BIT, outSurround.encoding)

        // Feed 2 frames of 6-channel float PCM (FL, FR, FC, LFE, BL, BR)
        val inBuf = ByteBuffer.allocateDirect(2 * 6 * 4).order(ByteOrder.nativeOrder())
        repeat(2) {
            inBuf.putFloat(0.4f) // FL
            inBuf.putFloat(0.4f) // FR
            inBuf.putFloat(0.5f) // FC (Center dialogue)
            inBuf.putFloat(0.1f) // LFE
            inBuf.putFloat(0.2f) // BL
            inBuf.putFloat(0.2f) // BR
        }
        inBuf.flip()
        processor.queueInput(inBuf)
        val output = processor.output
        // 2 frames * 2 channels * 2 bytes = 8 bytes
        assertEquals(8, output.remaining())
    }

    @Test
    fun testHevc10BitAndNonAlignedDetection() {
        val hevcMain10Format = Format.Builder()
            .setSampleMimeType(MimeTypes.VIDEO_H265)
            .setCodecs("hvc1.2.4.L93.B0")
            .setWidth(1280)
            .setHeight(536)
            .build()
        assertTrue(VlcGradeRenderersFactory.isFormat10BitHevc(hevcMain10Format))
        assertTrue(VlcGradeRenderersFactory.hasNonMacroblockAlignedDimensions(hevcMain10Format))

        val hevcColor10Format = Format.Builder()
            .setSampleMimeType(MimeTypes.VIDEO_H265)
            .setColorInfo(ColorInfo.Builder().setLumaBitdepth(10).setChromaBitdepth(10).build())
            .setWidth(1920)
            .setHeight(1080)
            .build()
        assertTrue(VlcGradeRenderersFactory.isFormat10BitHevc(hevcColor10Format))
        assertFalse(VlcGradeRenderersFactory.hasNonMacroblockAlignedDimensions(hevcColor10Format))

        val hevc8BitFormat = Format.Builder()
            .setSampleMimeType(MimeTypes.VIDEO_H265)
            .setCodecs("hvc1.1.6.L93.B0")
            .setWidth(1280)
            .setHeight(720)
            .build()
        assertFalse(VlcGradeRenderersFactory.isFormat10BitHevc(hevc8BitFormat))
        assertFalse(VlcGradeRenderersFactory.hasNonMacroblockAlignedDimensions(hevc8BitFormat))
    }
