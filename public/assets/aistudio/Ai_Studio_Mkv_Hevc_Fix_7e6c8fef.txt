/**
 * Universal Stereo Downmix & PCM Normalizer AudioProcessor.
 *
 * Safely converts ANY channel configuration (Mono 1.0, Stereo 2.0, 2.1, 3.0, Quad 4.0, 5.0,
 * 5.1 Surround, 6.1, 7.1 Surround) and ANY PCM sample encoding (16-bit, Float/fltp, 24-bit,
 * 32-bit, 8-bit) into universal 16-bit Stereo PCM (C.ENCODING_PCM_16BIT, 2 channels).
 *
 * Avoids Media3 ChannelMixingMatrix.create(8, 2) UnsupportedOperationException and
 * prevents UnhandledAudioFormatException on 2-channel stereo or float PCM streams.
 */
@UnstableApi
class UniversalStereoAudioProcessor : BaseAudioProcessor() {

    override fun onConfigure(inputAudioFormat: AudioProcessor.AudioFormat): AudioProcessor.AudioFormat {
        val encoding = inputAudioFormat.encoding
        val channels = inputAudioFormat.channelCount
        val sampleRate = inputAudioFormat.sampleRate

        if (channels <= 0 || sampleRate <= 0) {
            return AudioProcessor.AudioFormat.NOT_SET
        }

        val isSupportedEncoding = encoding == C.ENCODING_PCM_16BIT ||
                encoding == C.ENCODING_PCM_FLOAT ||
                encoding == C.ENCODING_PCM_24BIT ||
                encoding == C.ENCODING_PCM_32BIT ||
                encoding == C.ENCODING_PCM_8BIT

        if (!isSupportedEncoding) {
            throw AudioProcessor.UnhandledAudioFormatException(inputAudioFormat)
        }

        // Standard 16-bit stereo passes through directly with zero overhead
        if (channels == 2 && encoding == C.ENCODING_PCM_16BIT) {
            return AudioProcessor.AudioFormat.NOT_SET
        }

        return AudioProcessor.AudioFormat(sampleRate, 2, C.ENCODING_PCM_16BIT)
    }

    override fun queueInput(inputBuffer: ByteBuffer) {
        val remaining = inputBuffer.remaining()
        if (remaining == 0) return

        val channels = inputAudioFormat.channelCount
        val encoding = inputAudioFormat.encoding
        val bytesPerSample = when (encoding) {
            C.ENCODING_PCM_8BIT -> 1
            C.ENCODING_PCM_16BIT -> 2
            C.ENCODING_PCM_24BIT -> 3
            C.ENCODING_PCM_32BIT, C.ENCODING_PCM_FLOAT -> 4
            else -> 2
        }
        val frameSize = channels * bytesPerSample
        if (frameSize <= 0) {
            inputBuffer.position(inputBuffer.limit())
            return
        }

        val frameCount = remaining / frameSize
        if (frameCount <= 0) {
            inputBuffer.position(inputBuffer.limit())
            return
        }

        val src = inputBuffer.order(ByteOrder.LITTLE_ENDIAN)
        // Output is always 2 channels * 2 bytes (16-bit stereo) = 4 bytes per frame
        val outBuffer = replaceOutputBuffer(frameCount * 4).order(ByteOrder.LITTLE_ENDIAN)
        val chSamples = FloatArray(channels)

        for (frame in 0 until frameCount) {
            for (ch in 0 until channels) {
                chSamples[ch] = readNormalizedSample(src, encoding)
            }

            val left: Float
            val right: Float

            when (channels) {
                1 -> {
                    left = chSamples[0]
                    right = chSamples[0]
                }
                2 -> {
                    left = chSamples[0]
                    right = chSamples[1]
                }
                3 -> {
                    // 3.0: FL, FR, FC
                    left = chSamples[0] * 0.65f + chSamples[2] * 0.45f
                    right = chSamples[1] * 0.65f + chSamples[2] * 0.45f
                }
                4 -> {
                    // Quad / 4.0: FL, FR, BL, BR
                    left = chSamples[0] * 0.65f + chSamples[2] * 0.45f
                    right = chSamples[1] * 0.65f + chSamples[3] * 0.45f
                }
                5 -> {
                    // 5.0: FL, FR, FC, BL, BR
                    left = chSamples[0] * 0.55f + chSamples[2] * 0.50f + chSamples[3] * 0.35f
                    right = chSamples[1] * 0.55f + chSamples[2] * 0.50f + chSamples[4] * 0.35f
                }
                6 -> {
                    // 5.1 Surround (HE-AAC / AAC / AC3): FL(0), FR(1), FC(2), LFE(3), BL(4), BR(5)
                    // Center channel (FC) carries primary spoken dialogue; preserve it clearly!
                    val fl = chSamples[0]
                    val fr = chSamples[1]
                    val fc = chSamples[2]
                    val lfe = chSamples[3]
                    val bl = chSamples[4]
                    val br = chSamples[5]
                    left = fl * 0.55f + fc * 0.58f + lfe * 0.15f + bl * 0.35f
                    right = fr * 0.55f + fc * 0.58f + lfe * 0.15f + br * 0.35f
                }
                else -> {
                    // 7.1 or higher: FL(0), FR(1), FC(2), LFE(3), BL(4), BR(5), SL(6), SR(7)
                    val fl = chSamples[0]
                    val fr = chSamples[1]
                    val fc = chSamples[2]
                    val lfe = chSamples[3]
                    val bl = chSamples[4]
                    val br = chSamples[5]
                    val sl = if (channels > 6) chSamples[6] else 0f
                    val sr = if (channels > 7) chSamples[7] else 0f
                    left = fl * 0.48f + fc * 0.55f + lfe * 0.12f + bl * 0.28f + sl * 0.28f
                    right = fr * 0.48f + fc * 0.55f + lfe * 0.12f + br * 0.28f + sr * 0.28f
                }
            }

            val leftShort = (left * 32767.0f).toInt().coerceIn(-32768, 32767).toShort()
            val rightShort = (right * 32767.0f).toInt().coerceIn(-32768, 32767).toShort()
            outBuffer.putShort(leftShort)
            outBuffer.putShort(rightShort)
        }

        // Advance input buffer to limit if any trailing partial bytes remain
        inputBuffer.position(inputBuffer.limit())
        outBuffer.flip()
    }

    private fun readNormalizedSample(src: ByteBuffer, encoding: Int): Float {
        return when (encoding) {
            C.ENCODING_PCM_16BIT -> src.short / 32768.0f
            C.ENCODING_PCM_FLOAT -> {
                val f = src.float
                if (f.isNaN() || f.isInfinite()) 0f else f.coerceIn(-1.0f, 1.0f)
            }
            C.ENCODING_PCM_8BIT -> ((src.get().toInt() and 0xFF) - 128) / 128.0f
            C.ENCODING_PCM_24BIT -> {
                val b0 = src.get().toInt() and 0xFF
                val b1 = src.get().toInt() and 0xFF
                val b2 = src.get().toInt() // signed MSB
                val sample24 = (b2 shl 16) or (b1 shl 8) or b0
                sample24 / 8388608.0f
            }
            C.ENCODING_PCM_32BIT -> src.int / 2147483648.0f
            else -> 0f
        }
    }
}

/**
 * VLC-Grade RenderersFactory for Media3 ExoPlayer:
 * 1. Handles 10-bit HEVC (Main 10, yuv420p10le) on devices whose hardware HEVC decoder only supports
 *    8-bit Main Profile by automatically prioritizing Android's 10-bit capable software decoder
 *    (c2.android.hevc.decoder / OMX.google.hevc.decoder) when hardware lacks Main 10 support or fails.
 * 2. Configures MediaCodecAudioRenderer with AAC Stereo Downmix hints (KEY_AAC_MAX_OUTPUT_CHANNEL_COUNT = 2
 *    and KEY_MAX_OUTPUT_CHANNEL_COUNT = 2) so 5.1 HE-AAC tracks decode cleanly without buffer overflow.
 * 3. Wraps DefaultAudioSink with UniversalStereoAudioProcessor so 5.1/7.1/float audio never causes
 *    ERROR_CODE_AUDIO_TRACK_INIT_FAILED.
 */
@UnstableApi
class VlcGradeRenderersFactory(
    context: Context,
    private val forceSoftwareVideoDecoderProvider: () -> Boolean = { false }
) : DefaultRenderersFactory(context) {

    companion object {
        private const val TAG = "VlcGradeRenderers"

        fun createUniversalAudioSink(context: Context): AudioSink {
            return DefaultAudioSink.Builder(context)
                .setAudioProcessors(arrayOf(UniversalStereoAudioProcessor()))
                .setEnableFloatOutput(false)
                .setEnableAudioTrackPlaybackParams(false)
                .build()
        }

        fun isFormat10BitHevc(format: Format): Boolean {
            if (format.sampleMimeType != MimeTypes.VIDEO_H265) return false
            val codecs = format.codecs?.lowercase() ?: ""
            val hasMain10CodecTag = codecs.startsWith("hev1.2") ||
                    codecs.startsWith("hvc1.2") ||
                    codecs.contains(".2.")
            val has10BitColor = (format.colorInfo?.lumaBitdepth ?: -1) >= 10 ||
                    (format.colorInfo?.chromaBitdepth ?: -1) >= 10
            return hasMain10CodecTag || has10BitColor
        }

        fun isSoftwareDecoder(info: ExoMediaCodecInfo): Boolean {
            val name = info.name.lowercase()
            return info.softwareOnly ||
                    !info.hardwareAccelerated ||
                    name.startsWith("c2.android.") ||
                    name.startsWith("omx.google.") ||
                    name.contains(".sw.") ||
                    name.contains("ffmpeg")
        }

        fun supportsHevcMain10Profile(info: ExoMediaCodecInfo): Boolean {
            return try {
                val levels = info.profileLevels
                levels.any {
                    it.profile == MediaCodecInfo.CodecProfileLevel.HEVCProfileMain10 ||
                            it.profile == MediaCodecInfo.CodecProfileLevel.HEVCProfileMain10HDR10 ||
                            it.profile == MediaCodecInfo.CodecProfileLevel.HEVCProfileMain10HDR10Plus
                }
            } catch (_: Exception) {
                false
            }
        }
    }

    init {
        setExtensionRendererMode(EXTENSION_RENDERER_MODE_PREFER)
        setEnableDecoderFallback(true)
    }

    override fun buildAudioSink(
        context: Context,
        enableFloatOutput: Boolean,
        enableAudioTrackPlaybackParams: Boolean
    ): AudioSink {
        return createUniversalAudioSink(context)
    }

    override fun buildVideoRenderers(
        context: Context,
        extensionRendererMode: Int,
        mediaCodecSelector: MediaCodecSelector,
        enableDecoderFallback: Boolean,
        eventHandler: Handler,
        eventListener: VideoRendererEventListener,
        allowedVideoJoiningTimeMs: Long,
        out: ArrayList<Renderer>
    ) {
        val smartVideoCodecSelector = MediaCodecSelector { mimeType, requiresSecureDecoder, requiresTunnelingDecoder ->
            val defaultList = mediaCodecSelector.getDecoderInfos(
                mimeType,
                requiresSecureDecoder,
                requiresTunnelingDecoder
            )
            if (defaultList.size <= 1) return@MediaCodecSelector defaultList

            val forceSw = forceSoftwareVideoDecoderProvider()
            if (forceSw) {
                val swFirst = defaultList.sortedBy { if (isSoftwareDecoder(it)) 0 else 1 }
                Log.i(TAG, "Software video decoding forced for $mimeType: ${swFirst.map { it.name }}")
                return@MediaCodecSelector swFirst
            }

            if (mimeType.equals(MimeTypes.VIDEO_H265, ignoreCase = true)) {
                defaultList.sortedBy { info ->
                    when {
                        !isSoftwareDecoder(info) && supportsHevcMain10Profile(info) -> 0
                        isSoftwareDecoder(info) -> 1
                        else -> 2
                    }
                }
            } else {
                defaultList
            }
        }

        val customVideoRenderer = object : MediaCodecVideoRenderer(
            context,
            codecAdapterFactory,
            smartVideoCodecSelector,
            allowedVideoJoiningTimeMs,
            enableDecoderFallback,
            eventHandler,
            eventListener,
            MAX_DROPPED_VIDEO_FRAME_COUNT_TO_NOTIFY
        ) {
            override fun getDecoderInfos(
                selector: MediaCodecSelector,
                format: Format,
                requiresSecureDecoder: Boolean
            ): MutableList<ExoMediaCodecInfo> {
                val infos = super.getDecoderInfos(selector, format, requiresSecureDecoder).toMutableList()
                val forceSw = forceSoftwareVideoDecoderProvider()
                val is10BitHevc = isFormat10BitHevc(format)

                if (forceSw || is10BitHevc) {
                    try {
                        val allForMime = selector.getDecoderInfos(
                            format.sampleMimeType ?: MimeTypes.VIDEO_H265,
                            requiresSecureDecoder,
                            false
                        )
                        for (candidate in allForMime) {
                            if (infos.none { it.name == candidate.name }) {
                                infos.add(candidate)
                            }
                        }
                    } catch (_: Exception) {}

                    infos.sortBy { info ->
                        if (forceSw) {
                            if (isSoftwareDecoder(info)) 0 else 1
                        } else {
                            when {
                                !isSoftwareDecoder(info) && supportsHevcMain10Profile(info) -> 0
                                isSoftwareDecoder(info) -> 1
                                else -> 2
                            }
                        }
                    }
                    Log.i(TAG, "Resolved HEVC/Video decoders (10bit=$is10BitHevc, forceSw=$forceSw): ${infos.map { it.name }}")
                }
                return infos
            }

            override fun getMediaFormat(
                format: Format,
                codecMimeType: String,
                codecMaxValues: CodecMaxValues,
                codecOperatingRate: Float,
                deviceNeedsNoPostProcessWorkaround: Boolean,
                tunnelingAudioSessionId: Int
            ): MediaFormat {
                val mediaFormat = super.getMediaFormat(
                    format,
                    codecMimeType,
                    codecMaxValues,
                    codecOperatingRate,
                    deviceNeedsNoPostProcessWorkaround,
                    tunnelingAudioSessionId
                )
                if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q &&
                    (isFormat10BitHevc(format) || forceSoftwareVideoDecoderProvider())
                ) {
                    try {
                        mediaFormat.removeKey(MediaFormat.KEY_PROFILE)
                        mediaFormat.removeKey(MediaFormat.KEY_LEVEL)
                    } catch (_: Exception) {}
                }
                return mediaFormat
            }

            override fun codecNeedsSetOutputSurfaceWorkaround(name: String): Boolean {
                return true
            }
        }

        out.add(customVideoRenderer)

        val tempOut = ArrayList<Renderer>()
        super.buildVideoRenderers(
            context,
            extensionRendererMode,
            smartVideoCodecSelector,
            enableDecoderFallback,
            eventHandler,
            eventListener,
            allowedVideoJoiningTimeMs,
            tempOut
        )
        for (r in tempOut) {
            if (r !is MediaCodecVideoRenderer) {
                out.add(r)
            }
        }
    }

    override fun buildAudioRenderers(
        context: Context,
        extensionRendererMode: Int,
        mediaCodecSelector: MediaCodecSelector,
        enableDecoderFallback: Boolean,
        audioSink: AudioSink,
        eventHandler: Handler,
        eventListener: AudioRendererEventListener,
        out: ArrayList<Renderer>
    ) {
        val customAudioRenderer = object : MediaCodecAudioRenderer(
            context,
            codecAdapterFactory,
            mediaCodecSelector,
            enableDecoderFallback,
            eventHandler,
            eventListener,
            audioSink
        ) {
            override fun getMediaFormat(
                format: Format,
                codecMimeType: String,
                codecMaxValues: Int,
                codecOperatingRate: Float
            ): MediaFormat {
                val mediaFormat = super.getMediaFormat(format, codecMimeType, codecMaxValues, codecOperatingRate)
                mediaFormat.setInteger(MediaFormat.KEY_MAX_OUTPUT_CHANNEL_COUNT, 2)
                if (Build.VERSION.SDK_INT >= 28) {
                    mediaFormat.setInteger("aac-max-output-channel_count", 2)
                }
                if (Build.VERSION.SDK_INT >= 24) {
                    mediaFormat.setInteger(
                        MediaFormat.KEY_PCM_ENCODING,
                        android.media.AudioFormat.ENCODING_PCM_16BIT
                    )
                }
                return mediaFormat
            }
        }

        out.add(customAudioRenderer)

        val tempOut = ArrayList<Renderer>()
        super.buildAudioRenderers(
            context,
            extensionRendererMode,
            mediaCodecSelector,
            enableDecoderFallback,
            audioSink,
            eventHandler,
            eventListener,
            tempOut
        )
        for (r in tempOut) {
            if (r !is MediaCodecAudioRenderer) {
                out.add(r)
            }
        }
    }
}

                        liveTts?.setLanguage(java.util.Locale("bn", "IN"))
                    }
                    isLiveTtsReady = true

                }
            }
        }
    }

                    .setTrackTypeDisabled(C.TRACK_TYPE_VIDEO, false)
                    .setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                    .build()

                .setContentType(C.AUDIO_CONTENT_TYPE_MOVIE)
                .setUsage(C.USAGE_MEDIA)
                .build()

                .build().apply {
                repeatMode = Player.REPEAT_MODE_OFF
                playWhenReady = true
                addListener(object : Player.Listener {
                    override fun onIsPlayingChanged(isPlaying: Boolean) {
                        _playerState.value = _playerState.value.copy(isPlaying = isPlaying)
                        if (isPlaying) {
                            dubbedAudioPlayer?.play()
                        } else {
                            dubbedAudioPlayer?.pause()
                            liveTts?.stop()
                        }
                    }

                    override fun onPlaybackStateChanged(playbackState: Int) {
                        if (playbackState == Player.STATE_READY) {
                            errorRecoveryAttempts = 0
                        }
                        val isPlaying = player?.isPlaying ?: false
                        val playWhenReady = player?.playWhenReady ?: false
                        val active = isPlaying || (playWhenReady && (playbackState == Player.STATE_READY || playbackState == Player.STATE_BUFFERING))
                        _playerState.value = _playerState.value.copy(isPlaying = active)
                    }

                    override fun onPlayWhenReadyChanged(playWhenReady: Boolean, reason: Int) {
                        val state = player?.playbackState ?: Player.STATE_IDLE
                        val active = (player?.isPlaying ?: false) || (playWhenReady && state != Player.STATE_ENDED && state != Player.STATE_IDLE)
                        _playerState.value = _playerState.value.copy(isPlaying = active)
                        if (playWhenReady) {
                            dubbedAudioPlayer?.playWhenReady = true
                            dubbedAudioPlayer?.play()
                        } else {
                            dubbedAudioPlayer?.playWhenReady = false
                            dubbedAudioPlayer?.pause()
                            liveTts?.stop()
                        }
                    }

                    override fun onPlaybackParametersChanged(playbackParameters: PlaybackParameters) {
                        _playerState.value = _playerState.value.copy(playbackSpeed = playbackParameters.speed)
                        dubbedAudioPlayer?.setPlaybackSpeed(playbackParameters.speed)
                    }

                    override fun onCues(cueGroup: CueGroup) {
                        val cueText = cueGroup.cues
                            .mapNotNull { it.text?.toString()?.trim() }
                            .filter { it.isNotEmpty() }
                            .joinToString("\n")
                            .ifBlank { null }
                        embeddedCueText = cueText
                        if (bnSubtitleEntries.isEmpty() && enSubtitleEntries.isEmpty()) {
                            updateCurrentSubtitleText(player?.currentPosition ?: 0L)
                        }
                    }

                    override fun onPlayerError(error: PlaybackException) {
                        Log.e(TAG, "ExoPlayer playback error: ${error.errorCodeName} (code=${error.errorCode}) - ${error.message}", error)

                        val isAudioOnlyFile = currentVideoUri?.path?.let { p ->
                            p.endsWith(".wav") || p.endsWith(".m4a") || p.endsWith(".mp3") || p.endsWith(".aac")
                        } ?: false

                        val isVideoCodecError = error.errorCode == PlaybackException.ERROR_CODE_DECODER_INIT_FAILED ||
                                error.errorCode == PlaybackException.ERROR_CODE_DECODING_FAILED ||
                                error.errorCode == PlaybackException.ERROR_CODE_DECODING_FORMAT_EXCEEDS_CAPABILITIES ||
                                error.errorCode == PlaybackException.ERROR_CODE_DECODING_FORMAT_UNSUPPORTED

                        // 1. If hardware video decoder failed (e.g., 10-bit HEVC Main 10 on 8-bit hardware),
                        // immediately switch to Android software video decoder (c2.android.hevc.decoder)
                        if (!isAudioOnlyFile && isVideoCodecError && !forceSoftwareVideoDecoder) {
                            Log.w(TAG, "Video decoder error detected; switching to software video decoder (10-bit HEVC fallback)")
                            forceSoftwareVideoDecoder = true
                            rebuildPrimaryPlayerWithSoftwareFallback()
                            return
                        }

                        // 2. If media is a video (MKV/MP4), retry with stereo channel cap if audio failed
                        if (!isAudioOnlyFile && player != null && errorRecoveryAttempts < 3) {
                            errorRecoveryAttempts++
                            Log.i(TAG, "Playback retry attempt ($errorRecoveryAttempts/3) with stereo downmix parameters.")
                            try {
                                val savedPos = player?.currentPosition?.coerceAtLeast(0L) ?: 0L
                                player?.trackSelectionParameters = player?.trackSelectionParameters
                                    ?.buildUpon()
                                    ?.setTrackTypeDisabled(C.TRACK_TYPE_VIDEO, false)
                                    ?.setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                                    ?.setMaxAudioChannelCount(2)
                                    ?.build() ?: androidx.media3.common.TrackSelectionParameters.DEFAULT_WITHOUT_CONTEXT
                                player?.prepare()
                                if (savedPos > 0L) player?.seekTo(savedPos)
                                player?.playWhenReady = true
                                player?.play()
                                return
                            } catch (e: Exception) {
                                Log.w(TAG, "Could not recover video playback: ${e.message}")
                            }
                        }

                        // 3. Only for genuine audio-only playback files
                        if (isAudioOnlyFile) {
                            val dubbed = _playerState.value.dubbedAudioFile ?: currentDubbedAudioFile
                            if (dubbed != null && dubbed.exists() && dubbed.length() > 44) {
                                Log.i(TAG, "Audio file fallback to dubbed audio track")
                                _playerState.value = _playerState.value.copy(
                                    isAudioOnlyFallback = true,
                                    hasError = false,
                                    errorMessage = null
                                )
                                val audioItem = MediaItem.fromUri(Uri.fromFile(dubbed))
                                val audioSource = DefaultMediaSourceFactory(context, createExtractorsFactory()).createMediaSource(audioItem)
                                dubbedAudioPlayer?.stop()
                                dubbedAudioPlayer?.release()
                                dubbedAudioPlayer = null
                                player?.setMediaSource(audioSource)
                                player?.prepare()
                                player?.play()
                                return
                            }
                        }

                        _playerState.value = _playerState.value.copy(
                            hasError = false,
                            errorMessage = null,
                            isPlaying = false
                        )
                    }
                })
            }
        }
        return player!!
    }

    private fun rebuildPrimaryPlayerWithSoftwareFallback() {
        val targetUri = currentVideoUri ?: return
        val savedPos = player?.currentPosition?.coerceAtLeast(0L) ?: _playerState.value.currentPositionMs
        val savedSpeed = _playerState.value.playbackSpeed

        try {
            player?.stop()
            player?.release()
        } catch (_: Exception) {}
        player = null

        val exo = initializePlayer()
        val mediaSourceFactory = DefaultMediaSourceFactory(context, createExtractorsFactory())
        val videoItem = MediaItem.fromUri(targetUri)
        val videoSource = mediaSourceFactory.createMediaSource(videoItem)
        exo.setMediaSource(videoSource)
        exo.setPlaybackSpeed(savedSpeed)
        exo.prepare()
        if (savedPos > 0L) {
            exo.seekTo(savedPos)
            dubbedAudioPlayer?.seekTo(savedPos)
        }
        exo.playWhenReady = true
        exo.play()
        updateAudioVolumes(savedPos)
    }

    private fun initializeDubbedPlayer(): ExoPlayer {
        if (dubbedAudioPlayer == null) {
            val renderersFactory = VlcGradeRenderersFactory(context)
            val mediaSourceFactory = DefaultMediaSourceFactory(context, createExtractorsFactory())
            dubbedAudioPlayer = ExoPlayer.Builder(context, renderersFactory)
                .setMediaSourceFactory(mediaSourceFactory)
                .build().apply {
                repeatMode = Player.REPEAT_MODE_OFF
                playWhenReady = player?.isPlaying ?: true
            }
        }
        return dubbedAudioPlayer!!
    }

    fun setupMedia(
        videoUri: Uri,
        dubbedAudioFile: File?,
        srtFile: File?,
        sourceSrtFile: File? = null
    ) {
        loadSubtitleFiles(srtFile, sourceSrtFile)

        val effectiveDubbedFile = when {
            dubbedAudioFile != null && dubbedAudioFile.exists() && dubbedAudioFile.length() > 44 -> dubbedAudioFile
            dubbedAudioFile != null -> {
                val parent = dubbedAudioFile.parentFile
                val wavFile = if (parent != null) File(parent, "dubbed_bn.wav") else null
                if (wavFile != null && wavFile.exists() && wavFile.length() > 44) wavFile else null
            }
            else -> null
        }

        if (currentVideoUri == videoUri &&
            currentDubbedAudioFile?.absolutePath == effectiveDubbedFile?.absolutePath &&
            player != null &&
            (player?.mediaItemCount ?: 0) > 0
        ) {
            Log.d(TAG, "setupMedia: Video and audio already initialized for $videoUri, ensuring active playback state")
            currentSrtFile = srtFile
            currentSourceSrtFile = sourceSrtFile
            val hasPreRendered = effectiveDubbedFile != null
            val canLive = !hasPreRendered && bnSubtitleEntries.isNotEmpty()
            _playerState.value = _playerState.value.copy(
                hasDubbedAudio = hasPreRendered || canLive,
                isLiveDubbing = canLive,
                hasBanglaSubtitles = bnSubtitleEntries.isNotEmpty(),
                hasEnglishSubtitles = enSubtitleEntries.isNotEmpty(),
                dubbedAudioFile = effectiveDubbedFile
            )
            player?.let { exo ->
                if (exo.playbackState == Player.STATE_IDLE) {
                    exo.prepare()
                } else if (exo.playbackState == Player.STATE_ENDED) {
                    exo.seekTo(0)
                }
                exo.playWhenReady = true
                exo.play()
                dubbedAudioPlayer?.let { dp ->
                    if (dp.playbackState == Player.STATE_IDLE) dp.prepare()
                    dp.seekTo(exo.currentPosition)
                    dp.playWhenReady = true
                    dp.play()
                }
                updateAudioVolumes(exo.currentPosition)
            }
            return
        }

        currentVideoUri = videoUri
        currentDubbedAudioFile = effectiveDubbedFile
        currentSrtFile = srtFile
        currentSourceSrtFile = sourceSrtFile
        errorRecoveryAttempts = 0
        zeroFramePlayingChecks = 0

        val exo = initializePlayer()
        val mediaSourceFactory = DefaultMediaSourceFactory(context, createExtractorsFactory())

        val isVideoUriAudioFile = videoUri.path?.let { p ->
            p.endsWith(".wav") || p.endsWith(".m4a") || p.endsWith(".mp3") || p.endsWith(".aac")
        } ?: false
        val isVideoUriSameAsDubbed = effectiveDubbedFile != null && videoUri.toString() == Uri.fromFile(effectiveDubbedFile).toString()
        val isAudioOnly = isVideoUriAudioFile || isVideoUriSameAsDubbed

        val hasPreRenderedDubbed = effectiveDubbedFile != null
        val canLiveDub = !hasPreRenderedDubbed && bnSubtitleEntries.isNotEmpty()
        val hasDubbed = hasPreRenderedDubbed || canLiveDub

        val initialAudioChoice = if (hasDubbed) AudioTrackChoice.BANGLA_DUB else AudioTrackChoice.ORIGINAL
        val initialSubChoice = when {
            bnSubtitleEntries.isNotEmpty() -> SubtitleChoice.BANGLA
            enSubtitleEntries.isNotEmpty() -> SubtitleChoice.ENGLISH
            else -> SubtitleChoice.ENGLISH // Allow embedded MKV subtitles when playing directly
        }

        // Set state including dubbedAudioFile BEFORE preparing video source
        _playerState.value = _playerState.value.copy(
            hasDubbedAudio = hasDubbed,
            isLiveDubbing = canLiveDub,
            isAudioOnlyFallback = isAudioOnly,
            hasError = false,
            errorMessage = null,
            audioChoice = initialAudioChoice,
            subtitleChoice = initialSubChoice,
            hasBanglaSubtitles = bnSubtitleEntries.isNotEmpty(),
            hasEnglishSubtitles = enSubtitleEntries.isNotEmpty(),
            dubbedAudioFile = effectiveDubbedFile
        )

        // 1. Setup Video with Original Audio
        exo.trackSelectionParameters = exo.trackSelectionParameters
            .buildUpon()
            .setTrackTypeDisabled(C.TRACK_TYPE_VIDEO, false)
            .setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
            .build()
        val videoItem = MediaItem.fromUri(videoUri)
        val videoSource = mediaSourceFactory.createMediaSource(videoItem)
        exo.setMediaSource(videoSource)
        exo.prepare()
        exo.playWhenReady = true
        exo.play()

        if (hasPreRenderedDubbed && !isVideoUriSameAsDubbed) {
            val dPlayer = initializeDubbedPlayer()
            val audioItem = MediaItem.fromUri(Uri.fromFile(effectiveDubbedFile!!))
            val audioSource = mediaSourceFactory.createMediaSource(audioItem)
            dPlayer.setMediaSource(audioSource)
            dPlayer.prepare()
            dPlayer.playWhenReady = true
            dPlayer.seekTo(exo.currentPosition.coerceAtLeast(0L))
            dPlayer.play()
            Log.i(TAG, "Initialized secondary dubbed player for real-time dynamic track switching: ${effectiveDubbedFile.absolutePath}")
        } else {
            dubbedAudioPlayer?.stop()
            dubbedAudioPlayer?.release()
            dubbedAudioPlayer = null
        }

        if (canLiveDub) {
            initLiveTts()
            Log.i(TAG, "Initialized Live Real-time TTS Dubbing (0-second wait mode)")
        }

        updateAudioVolumes(exo.currentPosition.coerceAtLeast(0L))
    }

    fun setAudioChoice(choice: AudioTrackChoice) {
        _playerState.value = _playerState.value.copy(audioChoice = choice)
        player?.let { exo ->
            exo.trackSelectionParameters = exo.trackSelectionParameters
                .buildUpon()
                .setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                .build()
            updateAudioVolumes(exo.currentPosition.coerceAtLeast(0L))
        }
    }

    /**
     * Dynamic Alternative Audio Track System:
     * When dialogue is active -> Dubbed Bangla speech plays, original video audio is ducked (or muted).
     * When dialogue is NOT active -> Original video audio (cars, storms, engines, footsteps, music) plays at 100% volume!
     */
    private fun updateAudioVolumes(currentPosMs: Long) {
        val exo = player ?: return
        val dPlayer = dubbedAudioPlayer

        val choice = _playerState.value.audioChoice
        val hasDubbed = _playerState.value.hasDubbedAudio
        val isLive = _playerState.value.isLiveDubbing

        if (!hasDubbed || _playerState.value.isAudioOnlyFallback) {
            exo.volume = 1.0f
            return
        }

        val isSpeaking = isDialogueActive(currentPosMs)
        if (_playerState.value.isSpeechActive != isSpeaking) {
            _playerState.value = _playerState.value.copy(isSpeechActive = isSpeaking)
        }

        when (choice) {
            AudioTrackChoice.BANGLA_DUB -> {
                if (isSpeaking) {
                    // Smart Ducking: Original ambient audio at 0.32f keeps cars, storms, music,
                    // footsteps, and explosions clearly audible behind the crisp 1.0f Bangla speech!
                    exo.volume = 0.32f
                    if (isLive) {
                        val activeCue = bnSubtitleEntries.firstOrNull { currentPosMs in it.startMs..it.endMs }
                        if (activeCue != null && activeCue.startMs != lastSpokenCueStartMs) {
                            lastSpokenCueStartMs = activeCue.startMs
                            if (isLiveTtsReady && _playerState.value.isPlaying) {
                                val params = android.os.Bundle()
                                params.putFloat(android.speech.tts.TextToSpeech.Engine.KEY_PARAM_VOLUME, 1.0f)
                                liveTts?.speak(activeCue.text, android.speech.tts.TextToSpeech.QUEUE_FLUSH, params, "cue_${activeCue.startMs}")
                            }
                        }
                    } else {
                        dPlayer?.volume = 1.0f
                    }
                } else {
                    // Ambient / music scene between dialogues: original video sound at full 100% volume!
                    exo.volume = if (bnSubtitleEntries.isEmpty()) 0.35f else 1.0f
                    if (!isLive) {
                        dPlayer?.volume = 1.0f
                    }
                }
            }
            AudioTrackChoice.ORIGINAL -> {
                exo.volume = 1.0f
                dPlayer?.volume = 0.0f
                if (isLive) liveTts?.stop()
            }
            AudioTrackChoice.BANGLA_ONLY -> {
                exo.volume = 0.0f
                if (isLive) {
                    val activeCue = bnSubtitleEntries.firstOrNull { currentPosMs in it.startMs..it.endMs }
                    if (activeCue != null && activeCue.startMs != lastSpokenCueStartMs) {
                        lastSpokenCueStartMs = activeCue.startMs
                        if (isLiveTtsReady && _playerState.value.isPlaying) {
                            val params = android.os.Bundle()
                            params.putFloat(android.speech.tts.TextToSpeech.Engine.KEY_PARAM_VOLUME, 1.0f)
                            liveTts?.speak(activeCue.text, android.speech.tts.TextToSpeech.QUEUE_FLUSH, params, "cue_${activeCue.startMs}")
                        }
                    }
                } else {
                    dPlayer?.volume = 1.0f
                }
            }
        }
    }

    private fun isDialogueActive(posMs: Long): Boolean {
        if (bnSubtitleEntries.isEmpty()) return true
        // Include small 40ms lead-in/lead-out for natural speech pacing
        return bnSubtitleEntries.any { posMs in (it.startMs - 40L)..(it.endMs + 80L) }
    }

    fun setSubtitleChoice(choice: SubtitleChoice) {
        _playerState.value = _playerState.value.copy(subtitleChoice = choice)
        updateCurrentSubtitleText(_playerState.value.currentPositionMs)
    }

        val currentIdx = modes.indexOf(_playerState.value.resizeMode)

        _playerState.value = _playerState.value.copy(resizeMode = nextMode)

    }

    }

    fun seekBy(deltaMs: Long) {
        val exo = player ?: return
        val current = exo.currentPosition
        val target = (current + deltaMs).coerceIn(0L, exo.duration.coerceAtLeast(0L))
        seekTo(target)
    }

    fun seekTo(positionMs: Long) {
        liveTts?.stop()
        lastSpokenCueStartMs = -1L
        player?.seekTo(positionMs)
        dubbedAudioPlayer?.seekTo(positionMs)
        updateAudioVolumes(positionMs)
    }

    fun togglePlayPause() {
        val exo = player ?: initializePlayer()
        val isCurrentlyPlaying = exo.isPlaying || (exo.playWhenReady && (exo.playbackState == Player.STATE_READY || exo.playbackState == Player.STATE_BUFFERING))
        val shouldPlay = !isCurrentlyPlaying

        if (shouldPlay) {
            if (exo.playbackState == Player.STATE_IDLE) {
                exo.prepare()
            } else if (exo.playbackState == Player.STATE_ENDED) {
                exo.seekTo(0)
            }
            exo.playWhenReady = true
            exo.play()
            dubbedAudioPlayer?.let { dp ->
                if (dp.playbackState == Player.STATE_IDLE) dp.prepare()
                dp.seekTo(exo.currentPosition.coerceAtLeast(0L))
                dp.playWhenReady = true
                dp.play()
            }
            _playerState.value = _playerState.value.copy(isPlaying = true)
        } else {
            exo.playWhenReady = false
            exo.pause()
            dubbedAudioPlayer?.playWhenReady = false
            dubbedAudioPlayer?.pause()
            liveTts?.stop()
            _playerState.value = _playerState.value.copy(isPlaying = false)
        }
    }

    fun toggleLock() {
        _playerState.value = _playerState.value.copy(isLocked = !_playerState.value.isLocked)
    }

    fun updateProgress() {
        val exo = player ?: return
        val pos = exo.currentPosition.coerceAtLeast(0L)
        val dur = exo.duration.coerceAtLeast(0L)
        _playerState.value = _playerState.value.copy(
            currentPositionMs = pos,
            durationMs = dur
        )

        // Detect silent hardware black-screen on 10-bit HEVC (Main 10) where audio/clock advances
        // but 8-bit hardware decoder renders 0 video frames
        if (exo.isPlaying && !forceSoftwareVideoDecoder && !_playerState.value.isAudioOnlyFallback) {
            val hasSelectedVideoTrack = exo.currentTracks.groups.any { it.type == C.TRACK_TYPE_VIDEO && it.isSelected }
            val renderedFrames = exo.videoDecoderCounters?.renderedOutputBufferCount ?: -1
            if (hasSelectedVideoTrack && renderedFrames == 0 && pos > 1500L) {
                zeroFramePlayingChecks++
                if (zeroFramePlayingChecks >= 10) {
                    Log.w(TAG, "Detected 0 rendered video frames after ${pos}ms; switching to software video decoder for 10-bit HEVC")
                    zeroFramePlayingChecks = 0
                    forceSoftwareVideoDecoder = true
                    rebuildPrimaryPlayerWithSoftwareFallback()
                    return
                }
            } else if (renderedFrames > 0) {
                zeroFramePlayingChecks = 0
            }
        }

        // Resync secondary player if clock drift exceeds threshold
        val dPlayer = dubbedAudioPlayer
        if (dPlayer != null && exo.isPlaying) {
            if (!dPlayer.isPlaying && dPlayer.playbackState == Player.STATE_READY) {
                dPlayer.play()
            }
            val drift = abs(pos - dPlayer.currentPosition)
            if (drift > 120L) {
                dPlayer.seekTo(pos)
            }
        }

        updateCurrentSubtitleText(pos)
        updateAudioVolumes(pos)
    }

    private fun updateCurrentSubtitleText(currentPositionMs: Long) {
        val choice = _playerState.value.subtitleChoice
        if (choice == SubtitleChoice.OFF) {
            if (_playerState.value.activeSubtitleText != null) {
                _playerState.value = _playerState.value.copy(activeSubtitleText = null)
            }
            return
        }

        val entries = when (choice) {
            SubtitleChoice.BANGLA -> bnSubtitleEntries
            SubtitleChoice.ENGLISH -> enSubtitleEntries
            SubtitleChoice.OFF -> emptyList()
        }

        if (entries.isEmpty()) {
            // Fall back to embedded MKV subtitle cues if external SRT list is empty
            val fallbackText = embeddedCueText
            if (_playerState.value.activeSubtitleText != fallbackText) {
                _playerState.value = _playerState.value.copy(activeSubtitleText = fallbackText)
            }
            return
        }

        val active = entries.firstOrNull { currentPositionMs in it.startMs..it.endMs }
        val text = active?.text
        if (_playerState.value.activeSubtitleText != text) {
            _playerState.value = _playerState.value.copy(activeSubtitleText = text)
        }
    }

    private fun loadSubtitleFiles(bnSrtFile: File?, enSrtFile: File?) {
        bnSubtitleEntries.clear()
        enSubtitleEntries.clear()

        bnSrtFile?.let { parseSrtIntoList(it, bnSubtitleEntries) }
        enSrtFile?.let { parseSrtIntoList(it, enSubtitleEntries) }
    }

    private fun parseSrtIntoList(srtFile: File, targetList: MutableList<ParsedSubtitle>) {
        if (!srtFile.exists() || srtFile.length() == 0L) return
        try {
            val content = srtFile.readText(Charsets.UTF_8)
            val cues = com.example.subtitle.SubtitleExtractor.parseSubtitleContent(content)
            for (cue in cues) {
                if (cue.text.isNotBlank()) {
                    targetList.add(ParsedSubtitle(cue.startMs, cue.endMs, cue.text.replace("\n", " ").trim()))
                }
            }
            Log.d(TAG, "Loaded ${targetList.size} subtitle entries from ${srtFile.name}")
        } catch (e: Exception) {
            Log.w(TAG, "Error parsing SRT file: ${srtFile.name}", e)
        }
    }

    fun release() {
        player?.release()
        player = null
        dubbedAudioPlayer?.release()
        dubbedAudioPlayer = null
        try {
            liveTts?.stop()
            liveTts?.shutdown()
            liveTts = null
        } catch (_: Exception) {}
    }
}

/**

 *
 * Features & Architecture:

 *    and automatic software decoder fallback for 10-bit HEVC (Main 10, yuv420p10le), Hi10P, and legacy codecs.
 * 2. Multichannel Audio Downmixing: Automatic 5.1/7.1 HE-AAC/AAC/AC3 & float PCM to 16-bit Stereo via
 *    UniversalStereoAudioProcessor and decoder channel hints.
 * 3. Dual-Player Smart Audio Ducking: Simultaneous playback of video ambient soundtrack and Bangla speech,
 *    ducking original background audio to 32% during speech so ambient cars, music, footsteps, and explosions
 *    remain audible behind clear Bangla dubbing.

 * 5. Local Storage / SAF: Handles content:// and file:// local URIs seamlessly with persistable permissions.

    private fun parseIntentData(intent: Intent?) {
        if (intent == null) return
        val rawUriStr = intent.getStringExtra(EXTRA_VIDEO_URI)
        val dataUri = intent.data

        videoUri = when {
            dataUri != null -> dataUri
            rawUriStr != null -> try { Uri.parse(rawUriStr) } catch (_: Exception) { null }
            else -> null
        }

        videoUri?.let { uri ->
            if (uri.scheme == "content") {
                try {
                    contentResolver.takePersistableUriPermission(
                        uri,
                        Intent.FLAG_GRANT_READ_URI_PERMISSION
                    )
                } catch (_: Exception) {}
            }
        }

        val resolvedDisplayName = videoUri?.let { queryDisplayName(it) }
        videoTitle = intent.getStringExtra(EXTRA_TITLE)?.takeIf { it.isNotBlank() }
            ?: resolvedDisplayName
            ?: videoUri?.lastPathSegment
            ?: "Offline Video"

        dubbedAudioPath = intent.getStringExtra(EXTRA_DUBBED_AUDIO_PATH)
        srtPath = intent.getStringExtra(EXTRA_SRT_PATH)
        launchedWithExplicitDub = !dubbedAudioPath.isNullOrBlank()

        val titleForLookup = resolvedDisplayName ?: videoTitle ?: ""
        if (titleForLookup.isNotBlank() && (dubbedAudioPath.isNullOrBlank() || srtPath.isNullOrBlank())) {
            val baseTitle = titleForLookup.substringBeforeLast(".")
            val publicDownloads = Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS)
            val fileTranslatoDir = File(publicDownloads, "FileTranslato")
            val banglaDubbingDir = File(publicDownloads, "BanglaDubbing")

            if (dubbedAudioPath.isNullOrBlank()) {
                val candidateAudio = File(fileTranslatoDir, "${baseTitle}_bangla_dubbed_audio.wav")
                if (candidateAudio.exists() && candidateAudio.length() > 44) {
                    dubbedAudioPath = candidateAudio.absolutePath
                }
            }
            if (srtPath.isNullOrBlank()) {
                val candidateSrt1 = File(fileTranslatoDir, "${baseTitle}_bangla.srt")
                val candidateSrt2 = File(banglaDubbingDir, "${baseTitle}_bangla.srt")
                if (candidateSrt1.exists() && candidateSrt1.length() > 0) {
                    srtPath = candidateSrt1.absolutePath
                } else if (candidateSrt2.exists() && candidateSrt2.length() > 0) {
                    srtPath = candidateSrt2.absolutePath
                }
            }
        }

        bnSubtitleCues.clear()
        srtPath?.let { path ->
            val f = File(path)
            if (f.exists() && f.length() > 0) {
                try {
                    bnSubtitleCues.addAll(SubtitleExtractor.parseSubtitleContent(f.readText(Charsets.UTF_8)))
                } catch (e: Exception) {
                    Log.w(TAG, "Could not parse SRT for smart ducking: ${e.message}")
                }
            }
        }
    }

    private fun queryDisplayName(uri: Uri): String? {
        if (uri.scheme == "file") {
            return uri.path?.let { File(it).name }
        }
        return try {
            contentResolver.query(uri, arrayOf(OpenableColumns.DISPLAY_NAME), null, null, null)?.use { cursor ->
                if (cursor.moveToFirst()) {
                    val idx = cursor.getColumnIndex(OpenableColumns.DISPLAY_NAME)
                    if (idx != -1) cursor.getString(idx) else null
                } else null
            }
        } catch (_: Exception) {
            null
        }
    }

    private fun initializePlayer() {
        if (player != null) return
        val targetUri = videoUri ?: run {
            Toast.makeText(this, "ভিডিও ফাইল খুঁজে পাওয়া যায়নি", Toast.LENGTH_SHORT).show()
            finish()
            return
        }

        try {
            val renderersFactory = VlcGradeRenderersFactory(this) { forceSoftwareVideoDecoder }

            trackSelector = DefaultTrackSelector(this).apply {
                parameters = buildUponParameters()
                    .setExceedRendererCapabilitiesIfNecessary(true)
                    .setExceedVideoConstraintsIfNecessary(true)
                    .setExceedAudioConstraintsIfNecessary(true)
                    .setAllowAudioMixedMimeTypeAdaptiveness(true)
                    .setAllowAudioMixedSampleRateAdaptiveness(true)
                    .setAllowAudioMixedChannelCountAdaptiveness(true)
                    .setAllowVideoMixedMimeTypeAdaptiveness(true)
                    .setAllowVideoNonSeamlessAdaptiveness(true)
                    .setTrackTypeDisabled(C.TRACK_TYPE_VIDEO, false)
                    .setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                    .build()
            }

            val audioAttributes = AudioAttributes.Builder()
                .setContentType(C.AUDIO_CONTENT_TYPE_MOVIE)
                .setUsage(C.USAGE_MEDIA)
                .build()

            val dataSourceFactory = DefaultDataSource.Factory(this)
            val mediaSourceFactory = DefaultMediaSourceFactory(dataSourceFactory, createExtractorsFactory())

            player = ExoPlayer.Builder(this, renderersFactory)
                .setTrackSelector(trackSelector!!)
                .setMediaSourceFactory(mediaSourceFactory)
                .setAudioAttributes(audioAttributes, /* handleAudioFocus= */ true)
                .build().apply {
                    repeatMode = Player.REPEAT_MODE_OFF
                    playWhenReady = startPlayWhenReady
                    addListener(PlayerEventListener())
                }

            playerView.player = player
            playerView.resizeMode = resizeModes[currentResizeModeIndex]

            val mediaItemBuilder = MediaItem.Builder().setUri(targetUri)

            srtPath?.let { sPath ->
                val srtFile = File(sPath)
                if (srtFile.exists() && srtFile.length() > 0) {
                    val subConfig = MediaItem.SubtitleConfiguration.Builder(Uri.fromFile(srtFile))
                        .setMimeType(MimeTypes.APPLICATION_SUBRIP)
                        .setLanguage("ben")
                        .setLabel("অনূদিত বাংলা সাবটাইটেল (Bangla Dub)")
                        .setSelectionFlags(if (launchedWithExplicitDub) C.SELECTION_FLAG_DEFAULT else 0)
                        .build()
                    mediaItemBuilder.setSubtitleConfigurations(listOf(subConfig))
                }
            }

            val mainMediaSource = mediaSourceFactory.createMediaSource(mediaItemBuilder.build())
            player?.setMediaSource(mainMediaSource)
            player?.prepare()

            val dubbedFile = dubbedAudioPath?.let { File(it) }
            if (dubbedFile != null && dubbedFile.exists() && dubbedFile.length() > 44) {
                val dubbedRenderersFactory = VlcGradeRenderersFactory(this)
                dubbedPlayer = ExoPlayer.Builder(this, dubbedRenderersFactory)
                    .setMediaSourceFactory(mediaSourceFactory)
                    .build().apply {
                        val audioItem = MediaItem.fromUri(Uri.fromFile(dubbedFile))
                        val audioSource = mediaSourceFactory.createMediaSource(audioItem)
                        setMediaSource(audioSource)
                        prepare()
                        playWhenReady = startPlayWhenReady
                        seekTo(startPositionMs)
                    }
                currentAudioMode = if (launchedWithExplicitDub) PlayerAudioMode.SMART_DUB else PlayerAudioMode.ORIGINAL_ONLY
                Log.i(TAG, "Dubbed audio player initialized (mode=$currentAudioMode): ${dubbedFile.absolutePath}")
            } else {
                dubbedPlayer = null
                currentAudioMode = PlayerAudioMode.ORIGINAL_ONLY
            }

            applyAudioModeVolumes(startPositionMs)

            if (startPositionMs > 0) {
                player?.seekTo(startPositionMs)
                dubbedPlayer?.seekTo(startPositionMs)
            }

            if (!isAdShownOrSkipped) {
                isAdShownOrSkipped = true
                isWaitingForAd = true
                player?.pause()
                dubbedPlayer?.pause()
                btnPlayPause.setImageResource(R.drawable.ic_play_arrow)
                AdMobManager.showRewardedAd(this) {
                    isWaitingForAd = false
                    startPlayWhenReady = true
                    if (player == null) {
                        initializePlayer()
                    }
                    player?.playWhenReady = true
                    player?.play()
                    dubbedPlayer?.playWhenReady = true
                    dubbedPlayer?.play()
                    btnPlayPause.setImageResource(R.drawable.ic_pause)
                }
            }

            handler.removeCallbacks(updateProgressRunnable)
            handler.post(updateProgressRunnable)
            resetControlsHideTimer()

            Log.i(TAG, "ExoPlayer initialized with VLC-grade 10-bit HEVC decoding, stereo downmixing, and Smart Dubbing")
        } catch (e: Exception) {
            Log.e(TAG, "ExoPlayer initialization failed: ${e.message}", e)
            Toast.makeText(this, "ভিডিও প্লেয়ার শুরু করতে সমস্যা হয়েছে: ${e.message}", Toast.LENGTH_LONG).show()
        }
    }

    private fun rebuildPlayerWithSoftwareVideoFallback() {
        val currentPos = player?.currentPosition?.coerceAtLeast(0L) ?: startPositionMs
        startPositionMs = currentPos
        startPlayWhenReady = true
        releasePlayer()
        initializePlayer()
    }

    private fun releasePlayer() {
        player?.let { p ->
            startPositionMs = p.currentPosition.coerceAtLeast(0L)
            startPlayWhenReady = if (isWaitingForAd) true else p.playWhenReady
            p.stop()
            p.release()
        }
        player = null
        trackSelector = null
        playerView.player = null

        dubbedPlayer?.let { dp ->
            dp.stop()
            dp.release()
        }
        dubbedPlayer = null

        handler.removeCallbacks(updateProgressRunnable)
        handler.removeCallbacks(hideControlsRunnable)
    }

    private inner class PlayerEventListener : Player.Listener {
        override fun onPlaybackStateChanged(playbackState: Int) {
            when (playbackState) {
                Player.STATE_BUFFERING -> {
                    progressBuffering.visibility = View.VISIBLE
                }
                Player.STATE_READY -> {
                    progressBuffering.visibility = View.GONE
                    errorRecoveryAttempts = 0
                    updateTimeline()
                }
                Player.STATE_ENDED -> {
                    progressBuffering.visibility = View.GONE
                    btnPlayPause.setImageResource(R.drawable.ic_play_arrow)
                    showControls()
                }
                Player.STATE_IDLE -> {
                    progressBuffering.visibility = View.GONE
                }
            }
        }

        override fun onIsPlayingChanged(isPlaying: Boolean) {
            btnPlayPause.setImageResource(if (isPlaying) R.drawable.ic_pause else R.drawable.ic_play_arrow)
            if (isPlaying) {
                dubbedPlayer?.play()
                resetControlsHideTimer()
            } else {
                dubbedPlayer?.pause()
            }
        }

        override fun onPlayerError(error: PlaybackException) {
            progressBuffering.visibility = View.GONE
            Log.e(TAG, "Player playback error: [${error.errorCodeName}] (code=${error.errorCode}) ${error.message}", error)

            val isVideoCodecError = error.errorCode == PlaybackException.ERROR_CODE_DECODER_INIT_FAILED ||
                    error.errorCode == PlaybackException.ERROR_CODE_DECODING_FAILED ||
                    error.errorCode == PlaybackException.ERROR_CODE_DECODING_FORMAT_EXCEEDS_CAPABILITIES ||
                    error.errorCode == PlaybackException.ERROR_CODE_DECODING_FORMAT_UNSUPPORTED

            if (isVideoCodecError && !forceSoftwareVideoDecoder) {
                Log.w(TAG, "Video decoder error; switching to software video decoder for 10-bit HEVC compatibility")
                forceSoftwareVideoDecoder = true
                rebuildPlayerWithSoftwareVideoFallback()
                return
            }

            if (errorRecoveryAttempts < 3) {
                errorRecoveryAttempts++
                Log.w(TAG, "Attempting playback recovery ($errorRecoveryAttempts/3) with 2-channel stereo audio constraint")
                try {
                    val savedPos = player?.currentPosition?.coerceAtLeast(0L) ?: startPositionMs
                    trackSelector?.let { ts ->
                        ts.setParameters(
                            ts.buildUponParameters()
                                .setTrackTypeDisabled(C.TRACK_TYPE_VIDEO, false)
                                .setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                                .setMaxAudioChannelCount(2)
                        )
                    }
                    player?.prepare()
                    if (savedPos > 0L) player?.seekTo(savedPos)
                    player?.playWhenReady = true
                    player?.play()
                } catch (e: Exception) {
                    Log.e(TAG, "Failed to recover playback: ${e.message}")
                }
            }
        }
    }

    private fun showAudioTrackSelectionDialog() {
        val p = player ?: return
        val ts = trackSelector ?: return
        val currentTracks = p.currentTracks

        val audioOptions = mutableListOf<AudioOptionItem>()
        var selectedItemIndex = 0

        if (dubbedPlayer != null) {
            for (mode in PlayerAudioMode.values()) {
                if (currentAudioMode == mode) {
                    selectedItemIndex = audioOptions.size
                }
                audioOptions.add(AudioOptionItem(mode.label, mode.description, mode = mode))
            }
        }

        var internalTrackNum = 1
        for (group in currentTracks.groups) {
            if (group.type == C.TRACK_TYPE_AUDIO) {
                val mediaTrackGroup = group.mediaTrackGroup
                for (i in 0 until mediaTrackGroup.length) {
                    val format = mediaTrackGroup.getFormat(i)
                    val lang = format.language ?: "und"
                    val channels = format.channelCount
                    val chLabel = when (channels) {
                        6 -> "5.1"
                        8 -> "7.1"
                        2 -> "Stereo"
                        1 -> "Mono"
                        else -> "${channels}-ch"
                    }
                    val mime = format.sampleMimeType?.substringAfterLast("/")?.uppercase(Locale.ROOT) ?: "AUDIO"
                    val langName = when (lang.lowercase(Locale.ROOT)) {
                        "ben", "bn" -> "বাংলা (Bangla)"
                        "hin", "hi" -> "Hindi (হিন্দি)"
                        "mal", "ml" -> "Malayalam (মালায়ালাম)"
                        "tam", "ta" -> "Tamil (তামিল)"
                        "tel", "te" -> "Telugu (তেলেগু)"
                        "eng", "en" -> "English (ইংরেজি)"
                        else -> lang.uppercase(Locale.ROOT)
                    }
                    val label = if (!format.label.isNullOrBlank()) {
                        "${format.label} [$langName • $chLabel $mime]"
                    } else {
                        "মূল অডিও #$internalTrackNum: $langName ($chLabel $mime)"
                    }
                    internalTrackNum++

                    if (dubbedPlayer == null && group.isTrackSelected(i)) {
                        selectedItemIndex = audioOptions.size
                    }
                    audioOptions.add(
                        AudioOptionItem(
                            title = label,
                            subtitle = "ভিডিওর অভ্যন্তরীণ অডিও ট্র্যাক",
                            trackOverride = TrackSelectionOverride(mediaTrackGroup, i)
                        )
                    )
                }
            }
        }

        if (audioOptions.isEmpty()) {
            Toast.makeText(this, "এই ভিডিওতে বিকল্প অডিও ট্র্যাক পাওয়া যায়নি", Toast.LENGTH_SHORT).show()
            return
        }

        val items = audioOptions.map { it.title }.toTypedArray()
        AlertDialog.Builder(this)
            .setTitle("অডিও ট্র্যাক নির্বাচন করুন (Smart Audio & Multi-Track)")
            .setSingleChoiceItems(items, selectedItemIndex) { dialog, which ->
                val chosen = audioOptions[which]
                if (chosen.mode != null) {
                    setAudioMode(chosen.mode)
                } else if (chosen.trackOverride != null) {
                    ts.setParameters(
                        ts.buildUponParameters()
                            .setTrackTypeDisabled(C.TRACK_TYPE_AUDIO, false)
                            .setOverrideForType(chosen.trackOverride)
                    )
                    if (dubbedPlayer != null && currentAudioMode == PlayerAudioMode.BANGLA_ONLY) {
                        setAudioMode(PlayerAudioMode.SMART_DUB)
                    } else {
                        applyAudioModeVolumes(p.currentPosition.coerceAtLeast(0L))
                    }
                }
                Toast.makeText(this, "নির্বাচিত অডিও: ${chosen.title}", Toast.LENGTH_SHORT).show()
                dialog.dismiss()
            }
            .setNegativeButton("বাতিল", null)
            .show()
    }

    private fun setAudioMode(mode: PlayerAudioMode) {
        currentAudioMode = mode
        applyAudioModeVolumes(player?.currentPosition?.coerceAtLeast(0L) ?: 0L)
    }

    private fun applyAudioModeVolumes(currentPosMs: Long = player?.currentPosition?.coerceAtLeast(0L) ?: 0L) {
        val p = player ?: return
        val dp = dubbedPlayer
        if (dp == null) {
            p.volume = 1.0f
            return
        }

        when (currentAudioMode) {
            PlayerAudioMode.SMART_DUB -> {
                val isSpeaking = if (bnSubtitleCues.isEmpty()) {
                    true
                } else {
                    bnSubtitleCues.any { currentPosMs in (it.startMs - 40L)..(it.endMs + 80L) }
                }
                p.volume = if (isSpeaking) 0.32f else 1.0f
                dp.volume = 1.0f
            }
            PlayerAudioMode.ORIGINAL_ONLY -> {
                p.volume = 1.0f
                dp.volume = 0.0f
            }
            PlayerAudioMode.BANGLA_ONLY -> {
                p.volume = 0.0f
                dp.volume = 1.0f
            }
        }
    }

    private fun showSubtitleTrackSelectionDialog() {
        val p = player ?: return

    private fun togglePlayPause() {
        val p = player ?: run {
            initializePlayer()
            return
        }
        if (p.isPlaying) {
            p.pause()
            dubbedPlayer?.pause()
        } else {
            if (p.playbackState == Player.STATE_IDLE) {
                p.prepare()
            } else if (p.playbackState == Player.STATE_ENDED) {
                p.seekTo(0)
            }
            p.playWhenReady = true
            p.play()
            dubbedPlayer?.let { dp ->
                if (dp.playbackState == Player.STATE_IDLE) dp.prepare()
                dp.seekTo(p.currentPosition.coerceAtLeast(0L))
                dp.playWhenReady = true
                dp.play()
            }
        }
    }

    private fun updateTimeline() {
        val p = player ?: return
        val currentMs = p.currentPosition.coerceAtLeast(0L)
        val durationMs = p.duration.coerceAtLeast(0L)

        txtTimeCurrent.text = formatTime(currentMs)
        if (durationMs > 0) {
            txtTimeDuration.text = formatTime(durationMs)
            val progress = ((currentMs * 1000L) / durationMs).toInt()
            seekBar.progress = progress
        }

        if (p.isPlaying && !forceSoftwareVideoDecoder && currentMs > 1500L) {
            val hasVideo = p.currentTracks.groups.any { it.type == C.TRACK_TYPE_VIDEO && it.isSelected }
            val renderedFrames = p.videoDecoderCounters?.renderedOutputBufferCount ?: -1
            if (hasVideo && renderedFrames == 0) {
                zeroFrameChecks++
                if (zeroFrameChecks >= 3) {
                    Log.w(TAG, "0 video frames rendered on hardware decoder; switching to software decoder for 10-bit HEVC")
                    zeroFrameChecks = 0
                    forceSoftwareVideoDecoder = true
                    rebuildPlayerWithSoftwareVideoFallback()
                    return
                }
            } else if (renderedFrames > 0) {
                zeroFrameChecks = 0
            }
        }

        dubbedPlayer?.let { dp ->
            if (p.isPlaying && !dp.isPlaying && dp.playbackState == Player.STATE_READY) {
                dp.play()
            }
            val drift = abs(currentMs - dp.currentPosition)
            if (drift > 120L) {
                dp.seekTo(currentMs)
            }
        }

        applyAudioModeVolumes(currentMs)
    }

    private fun formatTime(ms: Long): String {
        val totalSeconds = (ms / 1000).coerceAtLeast(0)
        val seconds = totalSeconds % 60
        val minutes = (totalSeconds / 60) % 60
        val hours = totalSeconds / 3600
        return if (hours > 0) {
            String.format(Locale.ROOT, "%d:%02d:%02d", hours, minutes, seconds)
        } else {
            String.format(Locale.ROOT, "%02d:%02d", minutes, seconds)
        }
    }

    @SuppressLint("ClickableViewAccessibility")
    private fun setupGestureDetector() {
        gestureDetector = GestureDetector(this, object : GestureDetector.SimpleOnGestureListener() {
            override fun onSingleTapConfirmed(e: MotionEvent): Boolean {
                if (isControlsLocked) {
                    btnUnlock.visibility = if (btnUnlock.visibility == View.VISIBLE) View.GONE else View.VISIBLE
                } else {
                    if (controlsContainer.visibility == View.VISIBLE) {
                        hideControls()
                    } else {
                        showControls()
                    }
                }
                return true
            }

            override fun onDoubleTap(e: MotionEvent): Boolean {
                if (isControlsLocked) return true
                val width = playerRoot.width.toFloat()
                if (width <= 0f) return true

                if (e.x < width / 3f) {
                    btnReplay10.performClick()
                } else if (e.x > (width * 2f) / 3f) {
                    btnForward10.performClick()
                } else {
                    togglePlayPause()
                }
                return true
            }
        })

        var initialTouchX = 0f
        var initialTouchY = 0f
        var isSwipeGesture = false
        var gestureMode = 0 // 1 = Brightness, 2 = Volume, 3 = Seek

        playerRoot.setOnTouchListener { _, event ->
            gestureDetector.onTouchEvent(event)

            if (isControlsLocked) return@setOnTouchListener true

            when (event.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    initialTouchX = event.x
                    initialTouchY = event.y
                    isSwipeGesture = false
                    gestureMode = 0
                }
                MotionEvent.ACTION_MOVE -> {
                    val deltaX = event.x - initialTouchX
                    val deltaY = event.y - initialTouchY
                    val screenWidth = playerRoot.width.toFloat().coerceAtLeast(1f)
                    val screenHeight = playerRoot.height.toFloat().coerceAtLeast(1f)

                    if (!isSwipeGesture && (abs(deltaX) > 40f || abs(deltaY) > 40f)) {
                        isSwipeGesture = true
                        gestureMode = if (abs(deltaX) > abs(deltaY)) {
                            3
                        } else if (initialTouchX < screenWidth / 2f) {
                            1
                        } else {
                            2
                        }
                    }

                    if (isSwipeGesture) {
                        when (gestureMode) {
                            1 -> adjustBrightness(-deltaY / screenHeight)
                            2 -> adjustVolume(-deltaY / screenHeight)
                            3 -> previewSeek(deltaX, screenWidth)
                        }
                    }
                }
                MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
                    if (isSwipeGesture && gestureMode == 3) {
                        commitSeek()
                    }
                    handler.postDelayed({ hudContainer.visibility = View.GONE }, 1200L)
                }
            }
            true
        }
    }

    private fun adjustBrightness(factorDelta: Float) {
        val lp = window.attributes
        var current = if (lp.screenBrightness < 0f) 0.5f else lp.screenBrightness
        current = (current + factorDelta * 0.4f).coerceIn(0.01f, 1.0f)
        lp.screenBrightness = current
        window.attributes = lp

        val percent = (current * 100).toInt()
        hudIcon.setImageResource(R.drawable.ic_brightness)
        hudText.text = "$percent%"
        hudProgress.progress = percent
        hudContainer.visibility = View.VISIBLE
    }

    private fun adjustVolume(factorDelta: Float) {
        val maxVol = audioManager.getStreamMaxVolume(AudioManager.STREAM_MUSIC).coerceAtLeast(1)
        val currentVol = audioManager.getStreamVolume(AudioManager.STREAM_MUSIC)
        val deltaSteps = (factorDelta * maxVol * 0.5f).toInt()
        val newVol = (currentVol + deltaSteps).coerceIn(0, maxVol)

        if (newVol != currentVol) {
            audioManager.setStreamVolume(AudioManager.STREAM_MUSIC, newVol, 0)
        }

        val percent = ((newVol.toFloat() / maxVol.toFloat()) * 100).toInt()
        hudIcon.setImageResource(R.drawable.ic_volume_up)
        hudText.text = "$percent%"
        hudProgress.progress = percent
        hudContainer.visibility = View.VISIBLE
    }

    private var pendingSeekPositionMs = 0L

    private fun previewSeek(deltaX: Float, screenWidth: Float) {
        val p = player ?: return
        val duration = p.duration
        if (duration <= 0) return

        val scrubMs = ((deltaX / screenWidth) * 90_000L).toLong()
        pendingSeekPositionMs = (p.currentPosition + scrubMs).coerceIn(0L, duration)

        hudIcon.setImageResource(R.drawable.ic_aspect_ratio)
        hudText.text = "${formatTime(pendingSeekPositionMs)} / ${formatTime(duration)}"
        hudProgress.progress = ((pendingSeekPositionMs * 100L) / duration).toInt()
        hudContainer.visibility = View.VISIBLE
    }

    private fun commitSeek() {
        player?.let { p ->
            if (pendingSeekPositionMs in 0L..p.duration) {
                p.seekTo(pendingSeekPositionMs)
                dubbedPlayer?.seekTo(pendingSeekPositionMs)
            }
        }
    }

    private fun showSeekHud(targetMs: Long, deltaMs: Long) {
        val p = player ?: return
        val sign = if (deltaMs >= 0) "+10s" else "-10s"
        hudIcon.setImageResource(if (deltaMs >= 0) R.drawable.ic_forward_10 else R.drawable.ic_replay_10)
        hudText.text = "$sign (${formatTime(targetMs)})"
        hudProgress.progress = if (p.duration > 0) ((targetMs * 100L) / p.duration).toInt() else 0
        hudContainer.visibility = View.VISIBLE
        handler.removeCallbacks(hideControlsRunnable)
        handler.postDelayed({ hudContainer.visibility = View.GONE }, 1200L)
    }

    /**
     * Inspects a video file (MKV, MP4, WebM) using streaming Matroska EBML parser and MediaExtractor
     * to discover all embedded subtitle tracks accurately without loading large files into RAM.
     */
    suspend fun inspectSubtitleTracks(context: Context, videoUri: Uri): List<SubtitleTrackInfo> =
        withContext(Dispatchers.IO) {
            // 1. First probe via streaming Matroska (.mkv / .webm) EBML header parser
            try {
                val mkvTracks = inspectMatroskaTracks(context, videoUri)
                if (mkvTracks.isNotEmpty()) {
                    Log.i(TAG, "Discovered ${mkvTracks.size} MKV subtitle track(s) via EBML header parser")
                    return@withContext mkvTracks
                }
            } catch (e: Exception) {
                Log.w(TAG, "Matroska EBML track inspection note: ${e.message}")
            }

            // 2. Probe via Android MediaExtractor (for MP4, WebM, 3GP, or fallback)
            val tracks = mutableListOf<SubtitleTrackInfo>()
            val extractor = MediaExtractor()
            try {
                extractor.setDataSource(context, videoUri, null)
                val trackCount = extractor.trackCount

                for (i in 0 until trackCount) {
                    val format = extractor.getTrackFormat(i)
                    val mime = format.getString(MediaFormat.KEY_MIME) ?: ""

                    if (isSubtitleMime(mime)) {
                        val lang = if (format.containsKey(MediaFormat.KEY_LANGUAGE)) {
                            format.getString(MediaFormat.KEY_LANGUAGE) ?: "und"
                        } else "und"

                        val title = if (format.containsKey("title")) {
                            format.getString("title") ?: "Subtitle Track #${i + 1}"
                        } else {
                            "Subtitle Track #${i + 1}"
                        }

                        val formatName = getFriendlyFormatName(mime)
                        val displayLang = getFriendlyLanguage(lang)

                        tracks.add(
                            SubtitleTrackInfo(
                                trackIndex = i,
                                mimeType = mime,
                                language = lang,
                                displayLanguage = displayLang,
                                title = title,
                                formatName = formatName
                            )
                        )
                    }
                }
            } catch (e: Exception) {
                Log.w(TAG, "MediaExtractor track inspection failed or partial: ${e.message}")
            } finally {
                try { extractor.release() } catch (_: Exception) {}
            }

            tracks
        }

    /**
     * Extracts all subtitle cues from a specified track in a video file.
     * Uses zero-OOM streaming EBML extraction for MKV files and MediaExtractor for MP4/other containers.
     */
    suspend fun extractCues(
        context: Context,
        videoUri: Uri,
        trackIndex: Int
    ): List<SubtitleCue> = withContext(Dispatchers.IO) {
        // 1. Try fast streaming Matroska (.mkv) EBML extractor first if the file is an MKV container
        try {
            val mkvCues = extractMatroskaCues(context, videoUri, trackIndex)
            if (mkvCues.isNotEmpty()) {
                Log.i(TAG, "Extracted ${mkvCues.size} subtitle cues from MKV track $trackIndex via streaming EBML parser")
                return@withContext adjustCueTimestamps(mkvCues)
            }
        } catch (e: Exception) {
            Log.w(TAG, "Streaming Matroska extraction note: ${e.message}")
        }

        // 2. Try Android MediaExtractor ONLY if the target track is genuinely a subtitle track
        val cues = mutableListOf<SubtitleCue>()
        val extractor = MediaExtractor()
        try {
            extractor.setDataSource(context, videoUri, null)
            val trackCount = extractor.trackCount
            var effectiveSubTrack = -1

            if (trackIndex in 0 until trackCount) {
                val format = extractor.getTrackFormat(trackIndex)
                val mime = format.getString(MediaFormat.KEY_MIME) ?: ""
                if (isSubtitleMime(mime)) {
                    effectiveSubTrack = trackIndex
                }
            }

            if (effectiveSubTrack == -1) {
                for (i in 0 until trackCount) {
                    val format = extractor.getTrackFormat(i)
                    val mime = format.getString(MediaFormat.KEY_MIME) ?: ""
                    if (isSubtitleMime(mime)) {
                        effectiveSubTrack = i
                        break
                    }
                }
            }

            if (effectiveSubTrack != -1) {
                extractor.selectTrack(effectiveSubTrack)
                val buffer = ByteBuffer.allocateDirect(256 * 1024)
                var rawIndex = 1

                while (true) {
                    buffer.clear()
                    val sampleSize = extractor.readSampleData(buffer, 0)
                    if (sampleSize < 0) break

                    val timeUs = extractor.sampleTime
                    val startMs = maxOf(0L, timeUs / 1000L)

                    val bytes = ByteArray(sampleSize)
                    buffer.get(bytes)
                    val rawText = String(bytes, Charsets.UTF_8).trim()

                    if (rawText.isNotBlank()) {
                        val parsed = parsePacketText(rawText, startMs, rawIndex)
                        if (parsed.isNotEmpty()) {
                            cues.addAll(parsed)
                            rawIndex += parsed.size
                        }
                    }

                    extractor.advance()
                }
            }
        } catch (e: Exception) {
            Log.e(TAG, "MediaExtractor subtitle extraction failed: ${e.message}", e)
        } finally {
            try { extractor.release() } catch (_: Exception) {}
        }

        if (cues.isEmpty()) {
            try {
                val ffmpegCues = extractCuesViaFfmpeg(context, videoUri, trackIndex)
                if (ffmpegCues.isNotEmpty()) {
                    return@withContext adjustCueTimestamps(ffmpegCues)
                }
            } catch (e: Exception) {
                Log.w(TAG, "FFmpeg subtitle extraction fallback failed: ${e.message}")
            }
        }

        adjustCueTimestamps(cues)
    }

    /**
     * Parses an external subtitle file (.srt, .vtt, .ass, .ssa)

    suspend fun parseSubtitleFile(file: File): List<SubtitleCue> = withContext(Dispatchers.IO) {
        if (!file.exists() || file.length() == 0L) return@withContext emptyList()
        val content = file.readText(Charsets.UTF_8)
        parseSubtitleContent(content)
    }

    /**
     * Parses an external subtitle from Uri (.srt, .vtt, .ass, .ssa)
     */
    suspend fun parseSubtitleUri(context: Context, uri: Uri): List<SubtitleCue> = withContext(Dispatchers.IO) {
        val stream = context.contentResolver.openInputStream(uri) ?: return@withContext emptyList()
        val content = stream.bufferedReader(Charsets.UTF_8).use { it.readText() }
        parseSubtitleContent(content)
    }

    /**
     * Parses subtitle text from String, auto-detecting SRT, WebVTT, or ASS/SSA format.
     */
    fun parseSubtitleContent(content: String): List<SubtitleCue> {
        val cleanContent = content.trim().removePrefix("\uFEFF").replace("\r\n", "\n").replace("\r", "\n")
        return when {
            cleanContent.contains("[Events]") || cleanContent.contains("Format: Layer") || cleanContent.contains("Dialogue:") -> {
                parseAssSubtitle(cleanContent)
            }
            cleanContent.startsWith("WEBVTT") || cleanContent.contains("-->") && cleanContent.contains(".") -> {
                parseVttOrSrtSubtitle(cleanContent)
            }
            cleanContent.contains("-->") -> {
                parseVttOrSrtSubtitle(cleanContent)
            }
            else -> {
                parseGenericTextSubtitle(cleanContent)
            }
        }
    }

    /**
     * Exports a list of subtitle cues to a standard SubRip (.srt) file.
     */
    fun exportToSrt(cues: List<SubtitleCue>, outputFile: File): File {
        val sb = StringBuilder()
        var validIdx = 1
        for (cue in cues) {
            val text = cue.text.trim()
            if (text.isBlank()) continue

            val startTimeStr = formatSrtTimestamp(cue.startMs)
            val endTimeStr = formatSrtTimestamp(cue.endMs)

            sb.append(validIdx++).append("\n")
            sb.append(startTimeStr).append(" --> ").append(endTimeStr).append("\n")
            sb.append(text).append("\n\n")
        }
        outputFile.parentFile?.mkdirs()
        outputFile.writeText(sb.toString(), Charsets.UTF_8)
        return outputFile
    }

    /**
     * Saves SRT content into the device's public Download folder (Environment.DIRECTORY_DOWNLOADS)
     * so the user can easily find, share, or use it in external video players.
     */
    fun saveSrtToPublicDownloads(context: Context, fileName: String, srtContent: String): File? {
        val cleanName = if (fileName.endsWith(".srt", ignoreCase = true)) fileName else "$fileName.srt"
        try {
            if (android.os.Build.VERSION.SDK_INT >= android.os.Build.VERSION_CODES.Q) {
                val values = android.content.ContentValues().apply {
                    put(android.provider.MediaStore.MediaColumns.DISPLAY_NAME, cleanName)
                    put(android.provider.MediaStore.MediaColumns.MIME_TYPE, "application/x-subrip")
                    put(android.provider.MediaStore.MediaColumns.RELATIVE_PATH, android.os.Environment.DIRECTORY_DOWNLOADS + "/BanglaDubbing")
                }
                val uri = context.contentResolver.insert(android.provider.MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
                if (uri != null) {
                    context.contentResolver.openOutputStream(uri)?.use { stream ->
                        stream.write(srtContent.toByteArray(Charsets.UTF_8))
                    }
                    Log.i(TAG, "Saved subtitle to public Downloads via MediaStore: $cleanName")
                }
            }

            val publicDownloads = android.os.Environment.getExternalStoragePublicDirectory(android.os.Environment.DIRECTORY_DOWNLOADS)
            val subDir = File(publicDownloads, "FileTranslato").apply { if (!exists()) mkdirs() }
            val targetFile = File(subDir, cleanName)
            targetFile.writeText(srtContent, Charsets.UTF_8)
            Log.i(TAG, "Saved subtitle directly to file: ${targetFile.absolutePath}")
            return targetFile
        } catch (e: Exception) {
            Log.w(TAG, "Could not save to public Downloads, falling back to app external files dir: ${e.message}")
            try {
                val extDir = context.getExternalFilesDir(android.os.Environment.DIRECTORY_DOWNLOADS) ?: context.filesDir
                val fallbackFile = File(extDir, cleanName)
                fallbackFile.writeText(srtContent, Charsets.UTF_8)
                return fallbackFile
            } catch (ex: Exception) {
                Log.e(TAG, "Failed fallback subtitle export: ${ex.message}")
                return null
            }
        }
    }

    fun formatSrtTimestamp(ms: Long): String {
        val safeMs = ms.coerceAtLeast(0L)
        val hours = safeMs / 3600000
        val minutes = (safeMs % 3600000) / 60000
        val seconds = (safeMs % 60000) / 1000
        val millis = safeMs % 1000
        return String.format(Locale.US, "%02d:%02d:%02d,%03d", hours, minutes, seconds, millis)
    }

    private fun isSubtitleMime(mime: String): Boolean {
        val m = mime.lowercase(Locale.ROOT)
        return m.startsWith("text/") ||
                m.contains("sub") ||
                m.contains("vtt") ||
                m.contains("caption") ||
                m.contains("ssa") ||
                m.contains("ass") ||
                m.contains("timedtext") ||
                m == "application/x-subrip" ||
                m == "application/cea-608" ||
                m == "application/cea-708" ||
                m == "application/ttml+xml"
    }

    private fun getFriendlyFormatName(mime: String): String {
        val m = mime.lowercase(Locale.ROOT)
        return when {
            m.contains("subrip") || m.contains("srt") -> "SubRip (SRT)"
            m.contains("vtt") -> "WebVTT"
            m.contains("ssa") || m.contains("ass") -> "Advanced SubStation (ASS)"
            m.contains("cea-608") || m.contains("cea-708") -> "Closed Captions (CEA)"
            m.contains("ttml") -> "TTML"
            else -> "Text Subtitle"
        }
    }

    private fun getFriendlyLanguage(code: String): String {
        val c = code.lowercase(Locale.ROOT).trim()
        return when (c) {
            "eng", "en" -> "English"
            "ben", "bn" -> "Bengali (বাংলা)"
            "hin", "hi" -> "Hindi"
            "mal", "ml" -> "Malayalam"
            "tam", "ta" -> "Tamil"
            "tel", "te" -> "Telugu"
            "spa", "es" -> "Spanish"
            "fra", "fre", "fr" -> "French"
            "deu", "ger", "de" -> "German"
            "jpn", "ja" -> "Japanese"
            "kor", "ko" -> "Korean"
            "zho", "chi", "zh" -> "Chinese"
            "ara", "ar" -> "Arabic"
            "rus", "ru" -> "Russian"
            "und", "" -> "English / Default"
            else -> Locale(c).displayLanguage.ifBlank { c.uppercase(Locale.ROOT) }
        }
    }

    private val MKV_ASS_BLOCK_PREFIX_REGEX = Regex("^\\d+,\\s*\\d*,[^,]*,[^,]*,[^,]*,[^,]*,[^,]*,[^,]*,")

    private fun cleanSubtitleText(raw: String, isAssCodec: Boolean = false): String {
        var t = raw.trim().removePrefix("\uFEFF")
        if (isAssCodec || MKV_ASS_BLOCK_PREFIX_REGEX.containsMatchIn(t)) {
            t = t.replaceFirst(MKV_ASS_BLOCK_PREFIX_REGEX, "")
        }
        t = t.replace(Regex("\\{[^}]*\\}"), "")
        t = t.replace(Regex("<[^>]+>"), "")
        t = t.replace("\\N", "\n").replace("\\n", "\n").replace("\\h", " ")
        t = t.replace("♪", "").replace("♫", "")
        return t.lines().map { it.trim() }.filter { it.isNotEmpty() }.joinToString("\n").trim()
    }

    private fun parsePacketText(packetText: String, defaultStartMs: Long, indexOffset: Int): List<SubtitleCue> {
        if (packetText.startsWith("Dialogue:") || packetText.contains("\nDialogue:")) {
            return parseAssSubtitle(packetText)
        }

        if (packetText.contains("-->")) {
            val cues = parseVttOrSrtSubtitle(packetText)
            if (cues.isNotEmpty()) return cues
        }

        val clean = cleanSubtitleText(packetText)
        if (clean.isBlank()) return emptyList()

        val estimatedDuration = estimateCueDuration(clean)
        return listOf(
            SubtitleCue(
                index = indexOffset,
                startMs = defaultStartMs,
                endMs = defaultStartMs + estimatedDuration,
                text = clean
            )
        )
    }

    private fun estimateCueDuration(text: String): Long {
        val wordCount = text.split(Regex("\\s+")).filter { it.isNotBlank() }.size
        return (wordCount * 380L).coerceIn(1500L, 6500L)
    }

    private fun adjustCueTimestamps(cues: List<SubtitleCue>): List<SubtitleCue> {
        if (cues.isEmpty()) return emptyList()

        val sorted = cues.sortedBy { it.startMs }
        val adjusted = mutableListOf<SubtitleCue>()

        for (i in sorted.indices) {
            val cur = sorted[i]
            val nextStart = if (i + 1 < sorted.size) sorted[i + 1].startMs else Long.MAX_VALUE

            val estDuration = estimateCueDuration(cur.text)
            var endMs = if (cur.endMs > cur.startMs) cur.endMs else cur.startMs + estDuration

            if (nextStart > cur.startMs && endMs > nextStart) {
                endMs = maxOf(cur.startMs + 400L, nextStart - 40L)
            }

            adjusted.add(
                SubtitleCue(
                    index = i + 1,
                    startMs = cur.startMs,
                    endMs = endMs,
                    text = cur.text
                )
            )
        }

        return adjusted
    }

    private fun parseVttOrSrtSubtitle(content: String): List<SubtitleCue> {
        val cues = mutableListOf<SubtitleCue>()
        val blocks = content.split(Regex("\n\n+"))
        var index = 1

        val timestampRegex = Regex("(\\d{1,2}:\\d{2}:\\d{2}[,\\.]\\d{1,3}|\\d{2}:\\d{2}[,\\.]\\d{1,3})\\s*-->\\s*(\\d{1,2}:\\d{2}:\\d{2}[,\\.]\\d{1,3}|\\d{2}:\\d{2}[,\\.]\\d{1,3})")

        for (block in blocks) {
            val lines = block.lines().map { it.trim() }.filter { it.isNotBlank() }
            if (lines.isEmpty()) continue

            var timeLineIndex = -1
            var match: MatchResult? = null

            for (i in lines.indices) {
                val m = timestampRegex.find(lines[i])
                if (m != null) {
                    timeLineIndex = i
                    match = m
                    break
                }
            }

            if (match != null && timeLineIndex >= 0) {
                val startStr = match.groupValues[1]
                val endStr = match.groupValues[2]
                val startMs = parseTimestampToMs(startStr)
                val endMs = parseTimestampToMs(endStr)

                val textLines = lines.subList(timeLineIndex + 1, lines.size)
                val rawText = textLines.joinToString("\n")
                val clean = cleanSubtitleText(rawText)

                if (clean.isNotBlank()) {
                    cues.add(
                        SubtitleCue(
                            index = index++,
                            startMs = startMs,
                            endMs = if (endMs > startMs) endMs else startMs + estimateCueDuration(clean),
                            text = clean
                        )
                    )
                }
            }
        }

        return cues
    }

    private fun parseAssSubtitle(content: String): List<SubtitleCue> {
        val cues = mutableListOf<SubtitleCue>()
        val lines = content.lines()
        var index = 1

        for (line in lines) {
            val trimmed = line.trim()
            if (!trimmed.startsWith("Dialogue:")) continue

            val parts = trimmed.substringAfter("Dialogue:").split(",", limit = 10)
            if (parts.size >= 10) {
                val startStr = parts[1].trim()
                val endStr = parts[2].trim()
                val rawText = parts[9].trim()

                val startMs = parseAssTimestamp(startStr)
                val endMs = parseAssTimestamp(endStr)
                val clean = cleanSubtitleText(rawText, isAssCodec = true)

                if (clean.isNotBlank()) {
                    cues.add(
                        SubtitleCue(
                            index = index++,
                            startMs = startMs,
                            endMs = if (endMs > startMs) endMs else startMs + estimateCueDuration(clean),
                            text = clean
                        )
                    )
                }
            }
        }

        return cues
    }

    private fun parseGenericTextSubtitle(content: String): List<SubtitleCue> {
        val lines = content.lines().map { cleanSubtitleText(it) }.filter { it.isNotBlank() }
        var currentMs = 0L
        return lines.mapIndexed { idx, line ->
            val duration = estimateCueDuration(line)
            val cue = SubtitleCue(idx + 1, currentMs, currentMs + duration, line)
            currentMs += duration + 300L
            cue
        }
    }

    private fun parseTimestampToMs(ts: String): Long {
        val clean = ts.replace(',', '.')
        val parts = clean.split(":")
        return try {
            if (parts.size == 3) {
                val h = parts[0].toLong()
                val m = parts[1].toLong()
                val sParts = parts[2].split(".")
                val s = sParts[0].toLong()
                val ms = if (sParts.size > 1) sParts[1].padEnd(3, '0').take(3).toLong() else 0L
                (h * 3600000L) + (m * 60000L) + (s * 1000L) + ms
            } else if (parts.size == 2) {
                val m = parts[0].toLong()
                val sParts = parts[1].split(".")
                val s = sParts[0].toLong()
                val ms = if (sParts.size > 1) sParts[1].padEnd(3, '0').take(3).toLong() else 0L
                (m * 60000L) + (s * 1000L) + ms
            } else 0L
        } catch (_: Exception) {
            0L
        }
    }

    private fun parseAssTimestamp(ts: String): Long {
        val parts = ts.split(":")
        return try {
            if (parts.size == 3) {
                val h = parts[0].toLong()
                val m = parts[1].toLong()
                val sParts = parts[2].split(".")
                val s = sParts[0].toLong()
                val cs = if (sParts.size > 1) sParts[1].padEnd(2, '0').take(2).toLong() else 0L
                (h * 3600000L) + (m * 60000L) + (s * 1000L) + (cs * 10L)
            } else 0L
        } catch (_: Exception) {
            0L
        }
    }

    // Streaming Zero-OOM Matroska (.mkv) EBML Container Parser

    private fun extractCuesViaFfmpeg(context: Context, videoUri: Uri, trackIndex: Int): List<SubtitleCue> {
        val directFile = if (videoUri.scheme == "file" && videoUri.path != null) {
            File(videoUri.path!!)
        } else null
        if (directFile == null || !directFile.exists()) return emptyList()

        return try {
            val srtOutput = StringBuilder()
            val cmd = arrayOf(
                "ffmpeg", "-y",
                "-i", directFile.absolutePath,
                "-map", "0:s:0",
                "-f", "srt",
                "-"
            )
            val process = ProcessBuilder(*cmd)
                .redirectErrorStream(false)
                .start()
            val reader = process.inputStream.bufferedReader(Charsets.UTF_8)
            var line = reader.readLine()
            while (line != null) {
                srtOutput.append(line).append("\n")
                line = reader.readLine()
            }
            val exitCode = process.waitFor()
            if (exitCode == 0 && srtOutput.isNotBlank()) {
                parseSubtitleContent(srtOutput.toString())
            } else {
                emptyList()
            }
        } catch (_: Throwable) {
            emptyList()
        }
    }

    private fun openEbmlReader(context: Context, videoUri: Uri): FastEbmlStream? {
        try {
            if (videoUri.scheme == "file" && !videoUri.path.isNullOrBlank()) {
                val file = File(videoUri.path!!)
                if (file.exists() && file.length() > 16) {
                    val fis = FileInputStream(file)
                    return FastEbmlStream(fis, fis.channel, null)
                }
            }
            val pfd: ParcelFileDescriptor? = try {
                context.contentResolver.openFileDescriptor(videoUri, "r")
            } catch (_: Exception) {
                null
            }
            if (pfd != null) {
                val fis = FileInputStream(pfd.fileDescriptor)
                return FastEbmlStream(fis, fis.channel, pfd)
            }
            val ins = context.contentResolver.openInputStream(videoUri) ?: return null
            return FastEbmlStream(BufferedInputStream(ins, 64 * 1024), null, null)
        } catch (e: Exception) {
            Log.w(TAG, "Could not open EBML stream for $videoUri: ${e.message}")
            return null
        }
    }

    private fun inspectMatroskaTracks(context: Context, videoUri: Uri): List<SubtitleTrackInfo> {
        val stream = openEbmlReader(context, videoUri) ?: return emptyList()
        return stream.use { ebml ->
            MatroskaStreamingParser(ebml).findSubtitleTracks()
        }
    }

    private fun extractMatroskaCues(context: Context, videoUri: Uri, trackIndex: Int): List<SubtitleCue> {
        val stream = openEbmlReader(context, videoUri) ?: return emptyList()
        return stream.use { ebml ->
            MatroskaStreamingParser(ebml).extractCuesForTrack(trackIndex)
        }
    }

    private class FastEbmlStream(
        private val inputStream: InputStream,
        private val fileChannel: FileChannel?,
        private val pfd: ParcelFileDescriptor?
    ) : AutoCloseable {

        private val buffer = ByteArray(64 * 1024)
        private var bufPos = 0
        private var bufLimit = 0
        private var baseFileOffset = 0L
        var isEof = false
            private set

        val position: Long
            get() = baseFileOffset + bufPos

        private fun refill(): Boolean {
            if (isEof) return false
            baseFileOffset += bufLimit
            bufPos = 0
            bufLimit = 0

            val read = if (fileChannel != null) {
                val bb = ByteBuffer.wrap(buffer)
                fileChannel.read(bb)
            } else {
                inputStream.read(buffer, 0, buffer.size)
            }

            if (read <= 0) {
                isEof = true
                return false
            }
            bufLimit = read
            return true
        }

        fun readByte(): Int {
            if (bufPos >= bufLimit && !refill()) return -1
            return buffer[bufPos++].toInt() and 0xFF
        }

        fun readSignedShort(): Short {
            val b0 = readByte()
            val b1 = readByte()
            if (b0 < 0 || b1 < 0) return 0
            return ((b0 shl 8) or b1).toShort()
        }

        fun readBytes(length: Int): ByteArray {
            if (length <= 0) return ByteArray(0)
            val safeLen = length.coerceAtMost(2 * 1024 * 1024)
            val out = ByteArray(safeLen)
            var offset = 0
            while (offset < safeLen) {
                if (bufPos >= bufLimit && !refill()) break
                val avail = (bufLimit - bufPos).coerceAtMost(safeLen - offset)
                System.arraycopy(buffer, bufPos, out, offset, avail)
                bufPos += avail
                offset += avail
            }
            return if (offset == safeLen) out else out.copyOf(offset)
        }

        fun skipTo(targetPosition: Long) {
            val currentPos = position
            if (targetPosition <= currentPos) return
            val delta = targetPosition - currentPos

            if (bufPos + delta <= bufLimit) {
                bufPos += delta.toInt()
                return
            }

            if (fileChannel != null) {
                try {
                    fileChannel.position(targetPosition)
                    baseFileOffset = targetPosition
                    bufPos = 0
                    bufLimit = 0
                    return
                } catch (_: Exception) {}
            }

            var remaining = delta - (bufLimit - bufPos)
            baseFileOffset += bufLimit
            bufPos = 0
            bufLimit = 0

            while (remaining > 0 && !isEof) {
                val skipped = inputStream.skip(remaining)
                if (skipped > 0) {
                    baseFileOffset += skipped
                    remaining -= skipped
                } else {
                    val b = inputStream.read()
                    if (b < 0) {
                        isEof = true
                        break
                    }
                    baseFileOffset += 1
                    remaining -= 1
                }
            }
        }

        fun readElementId(): Long {
            val b0 = readByte()
            if (b0 <= 0) return -1L
            var numBytes = 1
            var mask = 0x80
            while ((b0 and mask) == 0 && numBytes <= 4) {
                mask = mask ushr 1
                numBytes++
            }
            if (numBytes > 4) return -1L
            var value = b0.toLong()
            for (i in 1 until numBytes) {
                val bx = readByte()
                if (bx < 0) return -1L
                value = (value shl 8) or bx.toLong()
            }
            return value
        }

        fun readVintSize(): Pair<Long, Boolean> {
            val b0 = readByte()
            if (b0 <= 0) return Pair(-1L, false)
            var numBytes = 1
            var mask = 0x80
            while ((b0 and mask) == 0 && numBytes <= 8) {
                mask = mask ushr 1
                numBytes++
            }
            if (numBytes > 8) return Pair(-1L, false)
            var value = (b0 and (mask - 1)).toLong()
            var allOnes = (value == (mask - 1).toLong())
            for (i in 1 until numBytes) {
                val bx = readByte()
                if (bx < 0) return Pair(-1L, false)
                if (bx != 0xFF) allOnes = false
                value = (value shl 8) or bx.toLong()
            }
            return Pair(value, allOnes)
        }

        fun readUnsignedInt(size: Long): Long {
            val len = size.toInt().coerceIn(0, 8)
            var v = 0L
            for (i in 0 until len) {
                val b = readByte()
                if (b < 0) break
                v = (v shl 8) or b.toLong()
            }
            return v
        }

        fun readAsciiOrUtf8String(size: Long): String {
            val bytes = readBytes(size.toInt().coerceIn(0, 65536))
            var validLen = bytes.size
            while (validLen > 0 && bytes[validLen - 1] == 0.toByte()) {
                validLen--
            }
            return String(bytes, 0, validLen, Charsets.UTF_8).trim()
        }

        override fun close() {
            try { inputStream.close() } catch (_: Exception) {}
            try { fileChannel?.close() } catch (_: Exception) {}
            try { pfd?.close() } catch (_: Exception) {}
        }
    }

    private data class MkvTrackMeta(
        var trackNumber: Int = 0,
        var trackType: Int = 0,
        var codecId: String = "",
        var language: String = "eng",
        var languageBcp47: String = "",
        var name: String = "",
        var isDefault: Boolean = true,
        var isForced: Boolean = false,
        var isCompressed: Boolean = false,
        var compAlgo: Int = -1,
        var compHeaderStripBytes: ByteArray? = null
    ) {
        val effectiveLanguage: String
            get() = languageBcp47.ifBlank { language.ifBlank { "eng" } }

        val isSubtitleTrack: Boolean
            get() = trackType == 0x11 ||
                    codecId.startsWith("S_TEXT", ignoreCase = true) ||
                    codecId.contains("SUBRIP", ignoreCase = true) ||
                    codecId.contains("WEBVTT", ignoreCase = true) ||
                    codecId.contains("ASS", ignoreCase = true) ||
                    codecId.contains("SSA", ignoreCase = true)

        val isAssOrSsa: Boolean
            get() = codecId.contains("ASS", ignoreCase = true) || codecId.contains("SSA", ignoreCase = true)
    }

    private class MatroskaStreamingParser(private val stream: FastEbmlStream) {

        private var timecodeScaleNs: Long = 1_000_000L
        private val subtitleTracks = mutableListOf<MkvTrackMeta>()
        private var pendingTopElement: Triple<Long, Long, Boolean>? = null

        fun findSubtitleTracks(): List<SubtitleTrackInfo> {
            parseHeaderAndTracks(stopAtFirstCluster = true)
            return subtitleTracks.map { meta ->
                val mime = when {
                    meta.isAssOrSsa -> "text/x-ssa"
                    meta.codecId.contains("WEBVTT", ignoreCase = true) -> "text/vtt"
                    else -> "application/x-subrip"
                }
                val formatName = getFriendlyFormatName(mime)
                val displayLang = getFriendlyLanguage(meta.effectiveLanguage)
                val label = if (meta.name.isNotBlank()) {
                    "${meta.name} ($displayLang)"
                } else {
                    "$displayLang Subtitle (#${meta.trackNumber})"
                }
                SubtitleTrackInfo(
                    trackIndex = meta.trackNumber,
                    mimeType = mime,
                    language = meta.effectiveLanguage,
                    displayLanguage = displayLang,
                    title = label,
                    formatName = formatName,
                    isDefault = meta.isDefault,
                    isForced = meta.isForced
                )
            }
        }

        fun extractCuesForTrack(requestedTrackIndex: Int): List<SubtitleCue> {
            val rawCues = mutableListOf<SubtitleCue>()
            if (!verifyEbmlHeader()) return emptyList()

            var segmentEnd = Long.MAX_VALUE
            var foundSegment = false
            while (!stream.isEof) {
                val id = stream.readElementId()
                if (id < 0) break
                val (size, unknown) = stream.readVintSize()
                if (size < 0) break
                if (id == 0x18538067L) {
                    foundSegment = true
                    segmentEnd = if (unknown) Long.MAX_VALUE else stream.position + size
                    break
                } else {
                    stream.skipTo(stream.position + size)
                }
            }
            if (!foundSegment) return emptyList()

            var resolvedTargetTrack: MkvTrackMeta? = null

            while (!stream.isEof && stream.position < segmentEnd) {
                val currentElem = pendingTopElement
                val id: Long
                val size: Long
                val unknown: Boolean
                if (currentElem != null) {
                    id = currentElem.first
                    size = currentElem.second
                    unknown = currentElem.third
                    pendingTopElement = null
                } else {
                    id = stream.readElementId()
                    if (id < 0) break
                    val szPair = stream.readVintSize()
                    size = szPair.first
                    unknown = szPair.second
                    if (size < 0) break
                }

                val elemEnd = if (unknown) segmentEnd else (stream.position + size)

                when (id) {
                    0x1549A966L -> parseInfoElement(elemEnd)
                    0x1654AE6BL -> {
                        parseTracksElement(elemEnd)
                        resolvedTargetTrack = resolveTargetSubtitleTrack(requestedTrackIndex)
                    }
                    0x1F43B675L -> {
                        if (resolvedTargetTrack == null) {
                            resolvedTargetTrack = resolveTargetSubtitleTrack(requestedTrackIndex)
                        }
                        val target = resolvedTargetTrack ?: return emptyList()
                        parseClusterElement(elemEnd, unknown, target, rawCues)
                    }
                    else -> {
                        if (!unknown) {
                            stream.skipTo(elemEnd)
                        }
                    }
                }
            }

            return rawCues
        }

        private fun resolveTargetSubtitleTrack(requestedTrackIndex: Int): MkvTrackMeta? {
            if (subtitleTracks.isEmpty()) return null
            subtitleTracks.firstOrNull { it.trackNumber == requestedTrackIndex }?.let { return it }
            if (requestedTrackIndex in subtitleTracks.indices) {
                return subtitleTracks[requestedTrackIndex]
            }
            return subtitleTracks.firstOrNull {
                it.effectiveLanguage.startsWith("en", ignoreCase = true)
            } ?: subtitleTracks.firstOrNull { it.isDefault } ?: subtitleTracks.first()
        }

        private fun verifyEbmlHeader(): Boolean {
            val firstId = stream.readElementId()
            if (firstId != 0x1A45DFA3L) return false
            val (headerSize, _) = stream.readVintSize()
            if (headerSize < 0) return false
            stream.skipTo(stream.position + headerSize)
            return true
        }

        private fun parseHeaderAndTracks(stopAtFirstCluster: Boolean) {
            if (!verifyEbmlHeader()) return

            while (!stream.isEof) {
                val id = stream.readElementId()
                if (id < 0) break
                val (size, unknown) = stream.readVintSize()
                if (size < 0) break
                if (id == 0x18538067L) {
                    val segmentEnd = if (unknown) Long.MAX_VALUE else stream.position + size
                    while (!stream.isEof && stream.position < segmentEnd) {
                        val subId = stream.readElementId()
                        if (subId < 0) break
                        val (subSize, subUnknown) = stream.readVintSize()
                        if (subSize < 0) break
                        val subEnd = if (subUnknown) segmentEnd else stream.position + subSize

                        when (subId) {
                            0x1549A966L -> parseInfoElement(subEnd)
                            0x1654AE6BL -> {
                                parseTracksElement(subEnd)
                                if (stopAtFirstCluster && subtitleTracks.isNotEmpty()) return
                            }
                            0x1F43B675L -> {
                                if (stopAtFirstCluster) return
                                stream.skipTo(subEnd)
                            }
                            else -> {
                                if (!subUnknown) stream.skipTo(subEnd)
                            }
                        }
                    }
                    return
                } else {
                    stream.skipTo(stream.position + size)
                }
            }
        }

        private fun parseInfoElement(infoEnd: Long) {
            while (!stream.isEof && stream.position < infoEnd) {
                val id = stream.readElementId()
                if (id < 0) break
                val (size, _) = stream.readVintSize()
                if (size < 0) break
                val childEnd = stream.position + size
                if (id == 0x2AD7B1L) {
                    val scale = stream.readUnsignedInt(size)
                    if (scale > 0) timecodeScaleNs = scale
                }
                stream.skipTo(childEnd)
            }
        }

        private fun parseTracksElement(tracksEnd: Long) {
            subtitleTracks.clear()
            while (!stream.isEof && stream.position < tracksEnd) {
                val id = stream.readElementId()
                if (id < 0) break
                val (size, _) = stream.readVintSize()
                if (size < 0) break
                val entryEnd = stream.position + size
                if (id == 0xAEL) {
                    val meta = parseTrackEntry(entryEnd)
                    if (meta.trackNumber > 0 && meta.isSubtitleTrack) {
                        subtitleTracks.add(meta)
                    }
                }
                stream.skipTo(entryEnd)
            }
        }

        private fun parseTrackEntry(entryEnd: Long): MkvTrackMeta {
            val meta = MkvTrackMeta()
            while (!stream.isEof && stream.position < entryEnd) {
                val id = stream.readElementId()
                if (id < 0) break
                val (size, _) = stream.readVintSize()
                if (size < 0) break
                val propEnd = stream.position + size

                when (id) {
                    0xD7L -> meta.trackNumber = stream.readUnsignedInt(size).toInt()
                    0x83L -> meta.trackType = stream.readUnsignedInt(size).toInt()
                    0x86L -> meta.codecId = stream.readAsciiOrUtf8String(size)
                    0x22B59CL -> meta.language = stream.readAsciiOrUtf8String(size)
                    0x22B59DL -> meta.languageBcp47 = stream.readAsciiOrUtf8String(size)
                    0x536EL -> meta.name = stream.readAsciiOrUtf8String(size)
                    0x88L -> meta.isDefault = stream.readUnsignedInt(size) != 0L
                    0x55AAL -> meta.isForced = stream.readUnsignedInt(size) != 0L
                    0x6D80L -> parseContentEncodings(propEnd, meta)
                }
                stream.skipTo(propEnd)
            }
            return meta
        }

        private fun parseContentEncodings(encodingsEnd: Long, meta: MkvTrackMeta) {
            while (!stream.isEof && stream.position < encodingsEnd) {
                val id = stream.readElementId()
                if (id < 0) break
                val (size, _) = stream.readVintSize()
                if (size < 0) break
                val childEnd = stream.position + size
                if (id == 0x6240L) {
                    while (!stream.isEof && stream.position < childEnd) {
                        val ceId = stream.readElementId()
                        if (ceId < 0) break
                        val (ceSize, _) = stream.readVintSize()
                        if (ceSize < 0) break
                        val ceEnd = stream.position + ceSize
                        if (ceId == 0x5034L) {
                            meta.isCompressed = true
                            meta.compAlgo = 0
                            while (!stream.isEof && stream.position < ceEnd) {
                                val ccId = stream.readElementId()
                                if (ccId < 0) break
                                val (ccSize, _) = stream.readVintSize()
                                if (ccSize < 0) break
                                val ccEnd = stream.position + ccSize
                                if (ccId == 0x4254L) {
                                    meta.compAlgo = stream.readUnsignedInt(ccSize).toInt()
                                } else if (ccId == 0x4255L) {
                                    meta.compHeaderStripBytes = stream.readBytes(ccSize.toInt())
                                }
                                stream.skipTo(ccEnd)
                            }
                        }
                        stream.skipTo(ceEnd)
                    }
                }
                stream.skipTo(childEnd)
            }
        }

        private fun parseClusterElement(
            clusterEnd: Long,
            clusterUnknownSize: Boolean,
            targetTrack: MkvTrackMeta,
            outCues: MutableList<SubtitleCue>
        ) {
            var clusterTimecode = 0L

            while (!stream.isEof && stream.position < clusterEnd) {
                val id = stream.readElementId()
                if (id < 0) break
                val (size, unknown) = stream.readVintSize()
                if (size < 0) break

                // If Cluster had unknown size and we reached the next top-level Segment element,
                // save it in pendingTopElement so the outer Segment loop processes it without losing a single byte!
                if (clusterUnknownSize && (
                        id == 0x1F43B675L ||
                        id == 0x1C53BB6BL ||
                        id == 0x1254C367L ||
                        id == 0x1043A770L ||
                        id == 0x1941A469L ||
                        id == 0x1654AE6BL ||
                        id == 0x1549A966L
                    )
                ) {
                    pendingTopElement = Triple(id, size, unknown)
                    return
                }

                val elemEnd = if (unknown) clusterEnd else (stream.position + size)

                when (id) {
                    0xE7L -> { // Cluster Timecode
                        clusterTimecode = stream.readUnsignedInt(size)
                        stream.skipTo(elemEnd)
                    }
                    0xA3L -> { // SimpleBlock
                        val (blockTrackNum, _) = stream.readVintSize()
                        if (blockTrackNum.toInt() != targetTrack.trackNumber) {
                            stream.skipTo(elemEnd)
                        } else {
                            val relTime = stream.readSignedShort().toLong()
                            val flags = stream.readByte()
                            val payloadLen = (elemEnd - stream.position).toInt()
                            if (payloadLen in 1..65536 && (flags and 0x06) == 0) {
                                val rawBytes = stream.readBytes(payloadLen)
                                val decodedBytes = decodePayloadBytes(rawBytes, targetTrack)
                                val rawStr = String(decodedBytes, Charsets.UTF_8)
                                val clean = cleanSubtitleText(rawStr, targetTrack.isAssOrSsa)
                                if (clean.isNotBlank()) {
                                    val startMs = maxOf(0L, ((clusterTimecode + relTime) * timecodeScaleNs) / 1_000_000L)
                                    val durMs = estimateCueDuration(clean)
                                    outCues.add(
                                        SubtitleCue(
                                            index = outCues.size + 1,
                                            startMs = startMs,
                                            endMs = startMs + durMs,
                                            text = clean
                                        )
                                    )
                                }
                            }
                            stream.skipTo(elemEnd)
                        }
                    }
                    0xA0L -> { // BlockGroup
                        var blockRelTime = 0L
                        var blockPayload: ByteArray? = null
                        var blockDurationTicks = -1L
                        var isMatchingTrack = false

                        while (!stream.isEof && stream.position < elemEnd) {
                            val bgId = stream.readElementId()
                            if (bgId < 0) break
                            val (bgSize, _) = stream.readVintSize()
                            if (bgSize < 0) break
                            val bgChildEnd = stream.position + bgSize

                            when (bgId) {
                                0xA1L -> { // Block
                                    val (bTrackNum, _) = stream.readVintSize()
                                    if (bTrackNum.toInt() != targetTrack.trackNumber) {
                                        stream.skipTo(elemEnd)
                                        break
                                    } else {
                                        isMatchingTrack = true
                                        blockRelTime = stream.readSignedShort().toLong()
                                        stream.readByte() // flags
                                        val pLen = (bgChildEnd - stream.position).toInt()
                                        if (pLen in 1..65536) {
                                            blockPayload = stream.readBytes(pLen)
                                        }
                                        stream.skipTo(bgChildEnd)
                                    }
                                }
                                0x9BL -> { // BlockDuration
                                    blockDurationTicks = stream.readUnsignedInt(bgSize)
                                    stream.skipTo(bgChildEnd)
                                }
                                else -> {
                                    stream.skipTo(bgChildEnd)
                                }
                            }
                        }

                        if (isMatchingTrack && blockPayload != null) {
                            val decodedBytes = decodePayloadBytes(blockPayload!!, targetTrack)
                            val rawStr = String(decodedBytes, Charsets.UTF_8)
                            val clean = cleanSubtitleText(rawStr, targetTrack.isAssOrSsa)
                            if (clean.isNotBlank()) {
                                val startMs = maxOf(0L, ((clusterTimecode + blockRelTime) * timecodeScaleNs) / 1_000_000L)
                                val durationMs = if (blockDurationTicks > 0L) {
                                    ((blockDurationTicks * timecodeScaleNs) / 1_000_000L).coerceIn(300L, 15000L)
                                } else {
                                    estimateCueDuration(clean)
                                }
                                outCues.add(
                                    SubtitleCue(
                                        index = outCues.size + 1,
                                        startMs = startMs,
                                        endMs = startMs + durationMs,
                                        text = clean
                                    )
                                )
                            }
                        }
                        stream.skipTo(elemEnd)
                    }
                    else -> {
                        if (!unknown) {
                            stream.skipTo(elemEnd)
                        }
                    }
                }
            }
        }

        private fun decodePayloadBytes(raw: ByteArray, track: MkvTrackMeta): ByteArray {
            if (!track.isCompressed) return raw
            return try {
                when (track.compAlgo) {
                    0 -> {
                        val inflater = Inflater()
                        inflater.setInput(raw)
                        val bos = ByteArrayOutputStream(raw.size * 2)
                        val buf = ByteArray(1024)
                        while (!inflater.finished() && !inflater.needsInput()) {
                            val count = inflater.inflate(buf)
                            if (count <= 0) break
                            bos.write(buf, 0, count)
                        }
                        inflater.end()
                        bos.toByteArray()
                    }
                    3 -> {
                        val prefix = track.compHeaderStripBytes
                        if (prefix != null && prefix.isNotEmpty()) {
                            prefix + raw
                        } else raw
                    }
                    else -> raw
                }
            } catch (_: Exception) {
                raw
            }
        }
    }

class MediaCodecAudioExtractor(private val context: Context) : AudioExtractor {

    private val TAG = "AudioExtractor"

    override suspend fun extractAudio(
        videoUri: Uri,
        outputWavFile: File,
        onProgress: (Float) -> Unit
    ): Result<File> = withContext(Dispatchers.IO) {
        val extractor = MediaExtractor()
        var codec: MediaCodec? = null

        try {
            extractor.setDataSource(context, videoUri, null)
            val numTracks = extractor.trackCount
            var audioTrackIndex = -1
            var inputFormat: MediaFormat? = null

            // Prefer stereo track if multiple audio tracks exist, or fallback to first audio track (e.g. 5.1 HE-AAC)
            for (i in 0 until numTracks) {
                val format = extractor.getTrackFormat(i)
                val mime = format.getString(MediaFormat.KEY_MIME) ?: ""
                if (mime.startsWith("audio/")) {
                    if (audioTrackIndex == -1) {
                        audioTrackIndex = i
                        inputFormat = format
                    }
                    val ch = if (format.containsKey(MediaFormat.KEY_CHANNEL_COUNT)) {
                        format.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
                    } else 2
                    if (ch in 1..2) {
                        audioTrackIndex = i
                        inputFormat = format
                        break
                    }
                }
            }

            if (audioTrackIndex == -1 || inputFormat == null) {
                return@withContext Result.failure(IllegalStateException("No audio track found in the selected video."))
            }

            extractor.selectTrack(audioTrackIndex)
            val mime = inputFormat.getString(MediaFormat.KEY_MIME)!!
            val durationUs = if (inputFormat.containsKey(MediaFormat.KEY_DURATION)) {
                inputFormat.getLong(MediaFormat.KEY_DURATION).coerceAtLeast(1_000_000L)
            } else {
                1_000_000L
            }

            var sourceSampleRate = if (inputFormat.containsKey(MediaFormat.KEY_SAMPLE_RATE)) {
                inputFormat.getInteger(MediaFormat.KEY_SAMPLE_RATE)
            } else {
                44100
            }
            var sourceChannels = if (inputFormat.containsKey(MediaFormat.KEY_CHANNEL_COUNT)) {
                inputFormat.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
            } else {
                2
            }
            var pcmEncoding = if (inputFormat.containsKey(MediaFormat.KEY_PCM_ENCODING)) {
                inputFormat.getInteger(MediaFormat.KEY_PCM_ENCODING)
            } else {
                android.media.AudioFormat.ENCODING_PCM_16BIT
            }

            // Configure decoder hints for 5.1 HE-AAC / multichannel downmixing and 16-bit PCM
            try {
                inputFormat.setInteger(MediaFormat.KEY_MAX_OUTPUT_CHANNEL_COUNT, 2)
                if (Build.VERSION.SDK_INT >= 28) {
                    inputFormat.setInteger("aac-max-output-channel_count", 2)
                }
                if (Build.VERSION.SDK_INT >= 24) {
                    inputFormat.setInteger(
                        MediaFormat.KEY_PCM_ENCODING,
                        android.media.AudioFormat.ENCODING_PCM_16BIT
                    )
                }
            } catch (_: Exception) {}

            codec = MediaCodec.createDecoderByType(mime)
            codec.configure(inputFormat, null, null, 0)
            codec.start()

            val fos = FileOutputStream(outputWavFile)
            // Write placeholder 44-byte WAV header
            WavUtils.writeWavHeader(fos, 0, 36, WavUtils.DEFAULT_SAMPLE_RATE, 1, 16)

            val bufferInfo = MediaCodec.BufferInfo()
            var isEos = false
            val kTimeOutUs = 5000L

            var lastReportTime = System.currentTimeMillis()

            while (!isEos) {
                // Feed input buffers
                val inIndex = codec.dequeueInputBuffer(kTimeOutUs)
                if (inIndex >= 0) {
                    val inputBuffer = codec.getInputBuffer(inIndex)
                    if (inputBuffer != null) {
                        val sampleSize = extractor.readSampleData(inputBuffer, 0)
                        if (sampleSize < 0) {
                            codec.queueInputBuffer(inIndex, 0, 0, 0, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
                        } else {
                            // Clamp negative start timestamps (e.g., MKV start: -0.105000) to >= 0L
                            val rawPtsUs = extractor.sampleTime
                            val presentationTimeUs = maxOf(0L, rawPtsUs)
                            codec.queueInputBuffer(inIndex, 0, sampleSize, presentationTimeUs, 0)
                            extractor.advance()

                            val now = System.currentTimeMillis()
                            if (now - lastReportTime > 250) {
                                val progress = (presentationTimeUs.toFloat() / durationUs).coerceIn(0f, 0.95f)
                                onProgress(progress)
                                lastReportTime = now
                            }
                        }
                    }
                }

                // Drain output buffers
                val outIndex = codec.dequeueOutputBuffer(bufferInfo, kTimeOutUs)
                if (outIndex == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) {
                    val newFormat = codec.outputFormat
                    if (newFormat.containsKey(MediaFormat.KEY_SAMPLE_RATE)) {
                        sourceSampleRate = newFormat.getInteger(MediaFormat.KEY_SAMPLE_RATE)
                    }
                    if (newFormat.containsKey(MediaFormat.KEY_CHANNEL_COUNT)) {
                        sourceChannels = newFormat.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
                    }
                    if (newFormat.containsKey(MediaFormat.KEY_PCM_ENCODING)) {
                        pcmEncoding = newFormat.getInteger(MediaFormat.KEY_PCM_ENCODING)
                    }
                    Log.i(TAG, "Decoder output format changed: sampleRate=$sourceSampleRate, channels=$sourceChannels, encoding=$pcmEncoding")
                } else if (outIndex >= 0) {
                    val outputBuffer = codec.getOutputBuffer(outIndex)
                    if (outputBuffer != null && bufferInfo.size > 0) {
                        outputBuffer.position(bufferInfo.offset)
                        outputBuffer.limit(bufferInfo.offset + bufferInfo.size)

                        // Convert decoded PCM (may be stereo/5.1/any sample rate/float) to 16kHz mono 16-bit PCM
                        writeResampledMonoPcm(
                            outputBuffer,
                            fos,
                            sourceSampleRate,
                            sourceChannels,
                            pcmEncoding,
                            WavUtils.DEFAULT_SAMPLE_RATE
                        )
                    }
                    codec.releaseOutputBuffer(outIndex, false)

                    if ((bufferInfo.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) {
                        isEos = true
                    }
                }
            }

            fos.flush()
            fos.close()

            // Update WAV header with actual recorded data length
            WavUtils.updateWavHeader(outputWavFile)
            onProgress(1.0f)
            Log.d(TAG, "Audio extracted successfully: ${outputWavFile.length()} bytes, path: ${outputWavFile.absolutePath}")

            Result.success(outputWavFile)
        } catch (e: Exception) {
            Log.e(TAG, "Error extracting audio from video", e)
            Result.failure(e)
        } finally {
            try {
                codec?.stop()
                codec?.release()
                extractor.release()
            } catch (e: Exception) {
                Log.w(TAG, "Error releasing audio codec/extractor", e)
            }
        }
    }

    /**
     * Resamples multi-channel PCM (mono, stereo, 5.1, 7.1) or float PCM to 16,000 Hz 16-bit Mono.
     */
    private fun writeResampledMonoPcm(
        sourceBuffer: ByteBuffer,
        fos: FileOutputStream,
        sourceSampleRate: Int,
        sourceChannels: Int,
        pcmEncoding: Int,
        targetSampleRate: Int = 16000
    ) {
        val bb = sourceBuffer.order(ByteOrder.LITTLE_ENDIAN)
        val isFloat = pcmEncoding == android.media.AudioFormat.ENCODING_PCM_FLOAT
        val bytesPerSample = if (isFloat) 4 else 2
        val bytesPerFrame = bytesPerSample * sourceChannels
        if (bytesPerFrame <= 0) return

        val numSourceFrames = bb.remaining() / bytesPerFrame
        if (numSourceFrames <= 0) return

        val step = sourceSampleRate.toDouble() / targetSampleRate.toDouble()
        val outSamples = (numSourceFrames / step).toInt().coerceAtLeast(1)
        val outBytes = ByteArray(outSamples * 2)
        val outBb = ByteBuffer.wrap(outBytes).order(ByteOrder.LITTLE_ENDIAN)

        var srcPos = 0.0
        while (srcPos < numSourceFrames && outBb.hasRemaining()) {
            val srcIdx = srcPos.toInt()
            val frameOffset = bb.position() + (srcIdx * bytesPerFrame)

            if (frameOffset + bytesPerFrame <= bb.limit()) {
                if (isFloat) {
                    var sum = 0.0f
                    for (ch in 0 until sourceChannels) {
                        val v = bb.getFloat(frameOffset + (ch * 4))
                        if (!v.isNaN() && !v.isInfinite()) sum += v
                    }
                    val avgFloat = (sum / sourceChannels).coerceIn(-1.0f, 1.0f)
                    val shortVal = (avgFloat * 32767.0f).toInt().coerceIn(-32768, 32767).toShort()
                    outBb.putShort(shortVal)
                } else {
                    var sum = 0
                    for (ch in 0 until sourceChannels) {
                        sum += bb.getShort(frameOffset + (ch * 2)).toInt()
                    }
                    val avgShort = (sum / sourceChannels).coerceIn(-32768, 32767).toShort()
                    outBb.putShort(avgShort)
                }
            }
            srcPos += step
        }

        fos.write(outBytes, 0, outBb.position())
    }
}

package com.example.audio

import android.content.Context
import android.util.Log
import com.example.database.TranscriptSegmentEntity
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.currentCoroutineContext
import kotlinx.coroutines.ensureActive

import java.io.File
import java.io.FileInputStream
import java.io.FileOutputStream
import java.nio.ByteBuffer
import java.nio.ByteOrder

class AudioSynchronizer(private val context: Context) {

    private val TAG = "AudioSynchronizer"

    companion object {
        private const val DUCKED_VOLUME = 0.18f
        private const val NORMAL_VOLUME = 0.95f
        private const val FADE_MS = 35L
    }

    private fun durationMsToEvenBytes(durationMs: Long, sampleRate: Int): Long {
        val samples = (durationMs.coerceAtLeast(0L) * sampleRate.toLong()) / 1000L
        return samples * 2L
    }

    /**
     * Builds the synchronized full-length dubbed audio track matching the video timeline.
     * Intelligently preserves original background sounds (cars, storms, engines, walking, foley, ambience, music)
     * by ducking original audio during speech and keeping it at full natural volume during silence/ambient scenes.
     * Guaranteed zero-hang, ultra-fast, 2-byte PCM aligned, and ultra-low memory (<1MB).
     */
    suspend fun synchronizeAndMux(
        segments: List<TranscriptSegmentEntity>,
        totalDurationMs: Long,
        outputFile: File,
        backgroundAudioFile: File? = null,
        onProgress: (Float) -> Unit
    ): Result<File> = withContext(Dispatchers.IO) {
        val targetWavFile = if (outputFile.name.endsWith(".wav", ignoreCase = true)) {
            outputFile
        } else {
            File(outputFile.parentFile ?: context.cacheDir, "dubbed_bn.wav")
        }

        try {
            targetWavFile.parentFile?.mkdirs()

            val firstValidSeg = segments.mapNotNull { it.audioSegmentPath?.let { path -> File(path) } }
                .firstOrNull { it.exists() && it.length() > 44 }
            val detectedMeta = firstValidSeg?.let { WavUtils.readWavMetadata(it) }
            val sampleRate = detectedMeta?.sampleRate ?: WavUtils.DEFAULT_SAMPLE_RATE

            val maxSegmentEnd = segments.maxOfOrNull { it.endMs } ?: 0L
            val safeTotalDurationMs = maxOf(totalDurationMs, maxSegmentEnd + 1000L, 1000L)
            val targetTotalBytes = durationMsToEvenBytes(safeTotalDurationMs, sampleRate)

            Log.i(TAG, "Audio synchronization starting: sampleRate=$sampleRate Hz, duration=$safeTotalDurationMs ms, bgAudio=${backgroundAudioFile?.exists()}")

            val fos = FileOutputStream(targetWavFile)
            WavUtils.writeWavHeader(fos, targetTotalBytes, targetTotalBytes + 36, sampleRate, 1, 16)

            // Prepare background audio reader if available
            val hasBgAudio = backgroundAudioFile != null && backgroundAudioFile.exists() && backgroundAudioFile.length() > 44
            val bgFis = if (hasBgAudio) {
                val fis = FileInputStream(backgroundAudioFile)
                val bgMeta = WavUtils.readWavMetadata(backgroundAudioFile!!)
                val skipOffset = (bgMeta?.dataOffset ?: 44).toLong()
                safeSkip(fis, skipOffset)
                fis
            } else {
                null
            }

            var timelineCursorMs = 0L
            val sortedSegments = segments.sortedBy { it.startMs }

            try {
                for (i in sortedSegments.indices) {
                    currentCoroutineContext().ensureActive()
                    val seg = sortedSegments[i]
                    val segFile = seg.audioSegmentPath?.let { File(it) }
                    val safeSegStartMs = seg.startMs.coerceAtLeast(0L)
                    val safeSegEndMs = maxOf(safeSegStartMs + 300L, seg.endMs)

                    // 1. Process gap before this segment (background ambience at full volume)
                    if (safeSegStartMs > timelineCursorMs) {
                        val gapDurationMs = safeSegStartMs - timelineCursorMs
                        writeBackgroundAudioOrSilence(
                            bgFis = bgFis,
                            fos = fos,
                            durationMs = gapDurationMs,
                            sampleRate = sampleRate,
                            volume = NORMAL_VOLUME
                        )
                        timelineCursorMs = safeSegStartMs
                    }

                    // 2. Process dialogue segment (Bangla TTS speech mixed with ducked background sound)
                    val targetSlotMs = (safeSegEndMs - timelineCursorMs).coerceAtLeast(300L)

                    if (segFile != null && segFile.exists() && segFile.length() > 44) {
                        val actualTtsDurationMs = WavUtils.getWavDurationMs(segFile, sampleRate).coerceAtLeast(300L)

                        if (actualTtsDurationMs > targetSlotMs && targetSlotMs > 0) {
                            // TTS is slightly longer than slot -> smooth time-stretch & mix
                            val speedFactor = (actualTtsDurationMs.toDouble() / targetSlotMs.toDouble()).coerceIn(1.0, 1.35)
                            val stretchedTtsBytes = getTimeStretchedPcmBytes(segFile, speedFactor, sampleRate)
                            val stretchedDurationMs = (actualTtsDurationMs / speedFactor).toLong().coerceAtLeast(100L)
                            writeMixedAudio(
                                ttsBytes = stretchedTtsBytes,
                                bgFis = bgFis,
                                fos = fos,
                                durationMs = stretchedDurationMs,
                                sampleRate = sampleRate
                            )
                            timelineCursorMs += stretchedDurationMs
                        } else {
                            // TTS fits comfortably -> mix TTS with ducked background ambience
                            val ttsBytes = getPcmBytes(segFile, sampleRate)
                            writeMixedAudio(
                                ttsBytes = ttsBytes,
                                bgFis = bgFis,
                                fos = fos,
                                durationMs = actualTtsDurationMs,
                                sampleRate = sampleRate
                            )
                            timelineCursorMs += actualTtsDurationMs

                            // Fill remaining slot with ambient background sound
                            if (timelineCursorMs < safeSegEndMs) {
                                val remainingSlot = safeSegEndMs - timelineCursorMs
                                writeBackgroundAudioOrSilence(
                                    bgFis = bgFis,
                                    fos = fos,
                                    durationMs = remainingSlot,
                                    sampleRate = sampleRate,
                                    volume = NORMAL_VOLUME
                                )
                                timelineCursorMs = safeSegEndMs
                            }
                        }
                    } else {
                        // Empty/blank speech -> preserve original background audio for entire slot
                        writeBackgroundAudioOrSilence(
                            bgFis = bgFis,
                            fos = fos,
                            durationMs = targetSlotMs,
                            sampleRate = sampleRate,
                            volume = NORMAL_VOLUME
                        )
                        timelineCursorMs += targetSlotMs
                    }

                    val progress = ((i + 1).toFloat() / sortedSegments.size.coerceAtLeast(1)) * 0.95f
                    onProgress(progress)
                }

                // 3. Trailing background audio until the end of the video
                if (timelineCursorMs < safeTotalDurationMs) {
                    val trailingMs = safeTotalDurationMs - timelineCursorMs
                    writeBackgroundAudioOrSilence(
                        bgFis = bgFis,
                        fos = fos,
                        durationMs = trailingMs,
                        sampleRate = sampleRate,
                        volume = NORMAL_VOLUME
                    )
                    timelineCursorMs = safeTotalDurationMs
                }
            } finally {
                bgFis?.close()
            }

            fos.flush()
            fos.close()
            WavUtils.updateWavHeader(targetWavFile)

            // If a different filename was requested, mirror it
            if (outputFile.absolutePath != targetWavFile.absolutePath) {
                try {
                    targetWavFile.copyTo(outputFile, overwrite = true)
                } catch (_: Exception) {}
            }

            onProgress(1.0f)
            Log.i(TAG, "Audio synchronization completed: ${targetWavFile.absolutePath} (${targetWavFile.length()} bytes)")
            Result.success(targetWavFile)
        } catch (e: Exception) {
            Log.e(TAG, "Failed to synchronize audio: ${e.message}", e)
            Result.failure(e)
        }
    }

    /**
     * Reads background audio from [bgFis] and writes to [fos] with [volume].
     * Strictly maintains 2-byte (16-bit PCM) alignment.
     */
    private fun writeBackgroundAudioOrSilence(
        bgFis: FileInputStream?,
        fos: FileOutputStream,
        durationMs: Long,
        sampleRate: Int,
        volume: Float
    ) {
        val totalBytes = durationMsToEvenBytes(durationMs, sampleRate)
        if (totalBytes <= 0L) return

        val buffer = ByteArray(8192)
        val silenceBuffer = ByteArray(8192)
        var remaining = totalBytes

        while (remaining > 0L) {
            val toRead = (remaining.coerceAtMost(buffer.size.toLong()).toInt() / 2) * 2
            if (toRead <= 0) break

            val rawRead = bgFis?.read(buffer, 0, toRead) ?: -1
            val evenBytesRead = if (rawRead > 0) (rawRead / 2) * 2 else -1

            if (evenBytesRead > 0) {
                if (volume != 1.0f) {
                    val shortCount = evenBytesRead / 2
                    val bb = ByteBuffer.wrap(buffer, 0, evenBytesRead).order(ByteOrder.LITTLE_ENDIAN)
                    val outBb = ByteBuffer.allocate(evenBytesRead).order(ByteOrder.LITTLE_ENDIAN)
                    for (i in 0 until shortCount) {
                        val s = (bb.short * volume).toInt().coerceIn(Short.MIN_VALUE.toInt(), Short.MAX_VALUE.toInt()).toShort()
                        outBb.putShort(s)
                    }
                    fos.write(outBb.array(), 0, evenBytesRead)
                } else {
                    fos.write(buffer, 0, evenBytesRead)
                }
                remaining -= evenBytesRead
            } else {
                fos.write(silenceBuffer, 0, toRead)
                remaining -= toRead
            }
        }
    }

    /**
     * Mixes Bangla TTS dialogue with ducked original audio.
     */
    private fun writeMixedAudio(
        ttsBytes: ByteArray,
        bgFis: FileInputStream?,
        fos: FileOutputStream,
        durationMs: Long,
        sampleRate: Int
    ) {
        if (ttsBytes.isEmpty()) {
            writeBackgroundAudioOrSilence(bgFis, fos, durationMs, sampleRate, DUCKED_VOLUME)
            return
        }

        val evenTtsSize = (ttsBytes.size / 2) * 2
        val targetBytes = durationMsToEvenBytes(durationMs, sampleRate).coerceAtLeast(evenTtsSize.toLong()).toInt()
        val ttsShortCount = evenTtsSize / 2
        val ttsBb = ByteBuffer.wrap(ttsBytes, 0, evenTtsSize).order(ByteOrder.LITTLE_ENDIAN)
        val ttsShorts = ShortArray(ttsShortCount)
        for (i in 0 until ttsShortCount) {
            ttsShorts[i] = ttsBb.short
        }

        val totalShortCount = targetBytes / 2
        val bgBuffer = ByteArray(targetBytes)
        val bgBytesRead = bgFis?.read(bgBuffer, 0, targetBytes) ?: -1

        val bgShorts = ShortArray(totalShortCount)
        if (bgBytesRead >= 2) {
            val evenBgRead = (bgBytesRead / 2) * 2
            val bgBb = ByteBuffer.wrap(bgBuffer, 0, evenBgRead).order(ByteOrder.LITTLE_ENDIAN)
            val bgCount = evenBgRead / 2
            for (i in 0 until bgCount) {
                bgShorts[i] = bgBb.short
            }
        }

        val fadeSamples = ((sampleRate * FADE_MS) / 1000L).toInt().coerceAtMost(totalShortCount / 4)
        val outBytes = ByteArray(totalShortCount * 2)
        val outBb = ByteBuffer.wrap(outBytes).order(ByteOrder.LITTLE_ENDIAN)

        for (i in 0 until totalShortCount) {
            val ttsSample = if (i < ttsShortCount) ttsShorts[i].toFloat() else 0f
            val bgSample = bgShorts[i].toFloat()

            val currentDuck = if (fadeSamples > 0) {
                when {
                    i < fadeSamples -> {
                        val progress = i.toFloat() / fadeSamples
                        NORMAL_VOLUME * (1f - progress) + DUCKED_VOLUME * progress
                    }
                    i >= totalShortCount - fadeSamples -> {
                        val progress = (totalShortCount - 1 - i).toFloat() / fadeSamples
                        NORMAL_VOLUME * (1f - progress) + DUCKED_VOLUME * progress
                    }
                    else -> DUCKED_VOLUME
                }
            } else {
                DUCKED_VOLUME
            }

            val mixed = (ttsSample * 1.0f) + (bgSample * currentDuck)
            val clamped = mixed.toInt().coerceIn(Short.MIN_VALUE.toInt(), Short.MAX_VALUE.toInt()).toShort()
            outBb.putShort(clamped)
        }

        fos.write(outBytes)
    }

    private fun readDataChunkBytes(segFile: File): ByteArray {
        val meta = WavUtils.readWavMetadata(segFile)
        val dataOffset = (meta?.dataOffset ?: 44).toLong()
        val maxDataBytes = (meta?.dataSize ?: (segFile.length() - dataOffset)).coerceAtLeast(0L)
        FileInputStream(segFile).use { fis ->
            safeSkip(fis, dataOffset)
            val raw = fis.readBytes()
            val validLen = minOf(raw.size.toLong(), maxDataBytes).toInt()
            val evenLen = (validLen / 2) * 2
            if (evenLen < 4) return ByteArray(0)
            return if (evenLen == raw.size) raw else raw.copyOf(evenLen)
        }
    }

    private fun getPcmBytes(segFile: File, sampleRate: Int): ByteArray {
        val raw = readDataChunkBytes(segFile)
        if (raw.size < 4) return ByteArray(0)

        val shortCount = raw.size / 2
        val srcBuffer = ByteBuffer.wrap(raw).order(ByteOrder.LITTLE_ENDIAN)
        val fadeSamples = ((sampleRate * 0.005).toInt()).coerceAtMost(shortCount / 4)
        val outBytes = ByteArray(shortCount * 2)
        val outBuffer = ByteBuffer.wrap(outBytes).order(ByteOrder.LITTLE_ENDIAN)

        for (i in 0 until shortCount) {
            var s = srcBuffer.short.toFloat()
            if (fadeSamples > 0) {
                if (i < fadeSamples) {
                    s *= (i.toFloat() / fadeSamples)
                } else if (i >= shortCount - fadeSamples) {
                    s *= ((shortCount - 1 - i).toFloat() / fadeSamples)
                }
            }
            outBuffer.putShort(s.toInt().coerceIn(Short.MIN_VALUE.toInt(), Short.MAX_VALUE.toInt()).toShort())
        }
        return outBytes
    }

    private fun getTimeStretchedPcmBytes(segFile: File, speedFactor: Double, sampleRate: Int): ByteArray {
        val rawBytes = readDataChunkBytes(segFile)
        if (rawBytes.size < 4) return ByteArray(0)

        val shortCount = rawBytes.size / 2
        val srcBuffer = ByteBuffer.wrap(rawBytes).order(ByteOrder.LITTLE_ENDIAN)
        val shorts = ShortArray(shortCount)
        for (i in 0 until shortCount) {
            shorts[i] = srcBuffer.short
        }

        val outLength = (shortCount / speedFactor).toInt().coerceAtLeast(1)
        val outBytes = ByteArray(outLength * 2)
        val outBuffer = ByteBuffer.wrap(outBytes).order(ByteOrder.LITTLE_ENDIAN)

        val fadeSamples = ((sampleRate * 0.005).toInt()).coerceAtMost(outLength / 4)
        for (i in 0 until outLength) {
            val srcPos = i * speedFactor
            val idx0 = srcPos.toInt().coerceIn(0, shortCount - 1)
            val idx1 = (idx0 + 1).coerceIn(0, shortCount - 1)
            val frac = (srcPos - idx0).toFloat()
            var sample = (shorts[idx0] * (1.0f - frac) + shorts[idx1] * frac)

            if (fadeSamples > 0) {
                if (i < fadeSamples) {
                    sample *= (i.toFloat() / fadeSamples)
                } else if (i >= outLength - fadeSamples) {
                    sample *= ((outLength - 1 - i).toFloat() / fadeSamples)
                }
            }
            outBuffer.putShort(sample.toInt().coerceIn(Short.MIN_VALUE.toInt(), Short.MAX_VALUE.toInt()).toShort())
        }
        return outBytes
    }

    private fun safeSkip(fis: FileInputStream, bytesToSkip: Long) {
        var remaining = bytesToSkip
        while (remaining > 0L) {
            val skipped = fis.skip(remaining)
            if (skipped <= 0L) {
                if (fis.read() == -1) break
                remaining -= 1L
            } else {
                remaining -= skipped
            }
        }
    }
}

class DubbingPipeline(

    private val TAG = "DubbingPipeline"

    private val _pipelineState = MutableStateFlow<PipelineProgress?>(null)

    private var activeJobCancelled = false

    private fun queryContentSizeBytes(uri: Uri): Long {

        }

                    if (idx != -1 && !cursor.isNull(idx)) cursor.getLong(idx) else -1L

        videoTitle: String,
        isReDubOnly: Boolean = false,
        providedSubtitleCues: List<com.example.subtitle.SubtitleCue>? = null,
        onProgressUpdate: ((PipelineProgress) -> Unit)? = null
    ): Result<DubbingProject> = withContext(Dispatchers.Default) {
        activeJobCancelled = false
        val projectDir = storageManager.getProjectDir(projectId)
        val segmentsDir = storageManager.getProjectSegmentsDir(projectId)

        val rawAudioFile = File(projectDir, "source_audio.wav")
        val transcriptJsonFile = File(projectDir, "transcript.json")
        val srtFile = File(projectDir, "subtitle_bn.srt")
        val finalDubbedAudioFile = File(projectDir, "dubbed_bn.wav")

        VideoUriResolver.takePersistablePermission(context, videoUri)

        val rawExt = videoTitle.substringAfterLast('.', "mp4").lowercase()
        val videoExt = if (rawExt.isEmpty() || rawExt.length > 5 || rawExt.contains("/")) "mp4" else rawExt
        val localVideoFile = File(projectDir, "source_video.$videoExt")

        var workingVideoUri = videoUri
        if (localVideoFile.exists() && localVideoFile.length() > 1024) {
            workingVideoUri = Uri.fromFile(localVideoFile)
        } else {
            // Only cache small videos (< 60 MB) to avoid blocking startup or exhausting storage on 1 GB+ MKV movies
            val sourceSize = queryContentSizeBytes(videoUri)
            val maxCacheSizeBytes = 60L * 1024L * 1024L
            if (sourceSize in 1025L..maxCacheSizeBytes) {
                val tempCopyFile = File(projectDir, "source_video.tmp")
                try {
                    withContext(Dispatchers.IO) {
                        if (videoUri.scheme == "file") {
                            val srcFile = File(videoUri.path ?: "")
                            if (srcFile.exists() && srcFile.canonicalPath != localVideoFile.canonicalPath) {
                                srcFile.copyTo(tempCopyFile, overwrite = true)
                            }
                        } else {
                            context.contentResolver.openInputStream(videoUri)?.use { input ->
                                tempCopyFile.outputStream().use { output ->
                                    val buffer = ByteArray(128 * 1024)
                                    var bytesRead: Int
                                    while (input.read(buffer).also { bytesRead = it } != -1) {
                                        output.write(buffer, 0, bytesRead)
                                    }
                                    output.flush()
                                }
                            }
                        }
                        if (tempCopyFile.exists() && tempCopyFile.length() > 1024) {
                            if (localVideoFile.exists()) {
                                localVideoFile.delete()
                            }
                            val renamed = tempCopyFile.renameTo(localVideoFile)
                            if (!renamed) {
                                tempCopyFile.copyTo(localVideoFile, overwrite = true)
                                tempCopyFile.delete()
                            }
                        }
                    }
                    if (localVideoFile.exists() && localVideoFile.length() > 1024) {
                        workingVideoUri = Uri.fromFile(localVideoFile)
                        Log.i(TAG, "Cached small source video to project storage: ${localVideoFile.absolutePath}")
                    }
                } catch (e: Exception) {
                    try { if (tempCopyFile.exists()) tempCopyFile.delete() } catch (_: Exception) {}
                    Log.w(TAG, "Skipping local video cache (${e.message}), using persistent source URI directly")
                }
            } else {
                Log.i(TAG, "Source video is large (${sourceSize / (1024 * 1024)} MB); streaming directly from persistent URI without copying")
            }
        }

        val resolvedInitial = VideoUriResolver.resolvePlayableVideoUri(
            context = context,
            projectId = projectId,
            videoTitle = videoTitle,
            savedUriString = workingVideoUri.toString(),
            projectDirPath = projectDir.absolutePath
        )
        if (resolvedInitial != null) {
            workingVideoUri = resolvedInitial
        }

        var project = repository.findProject(projectId) ?: DubbingProject(
            id = projectId,
            title = videoTitle,
            videoUriString = workingVideoUri.toString(),
            projectDirPath = projectDir.absolutePath,
            originalAudioPath = rawAudioFile.absolutePath,
            transcriptJsonPath = transcriptJsonFile.absolutePath,
            subtitleSrtPath = srtFile.absolutePath,
            dubbedAudioPath = finalDubbedAudioFile.absolutePath
        )
        if (project.videoUriString != workingVideoUri.toString()) {
            project = project.copy(videoUriString = workingVideoUri.toString())
        }
        repository.saveProject(project)

        try {
            val mode = settingsManager.processingMode.value
            Log.d(TAG, "Starting pipeline for $projectId in mode: $mode, reDubOnly=$isReDubOnly")

            val startingStage = if (isReDubOnly) {
                ProcessingStage.TRANSLATE
            } else {
                determineResumeStage(project, rawAudioFile, transcriptJsonFile, srtFile, finalDubbedAudioFile)
            }

            // STAGE 1: Extract Audio (0% - 15%) - BYPASSED when subtitle cues are already provided
            val hasProvidedSubtitles = providedSubtitleCues != null && providedSubtitleCues.isNotEmpty()
            if (hasProvidedSubtitles) {
                checkCancelled()
                reportStage(
                    projectId,
                    ProcessingStage.EXTRACT_AUDIO,
                    100,
                    15,
                    "⚡ MKV Subtitle Track Ready (Instant)",
                    onProgressUpdate
                )

                val retrievedDurationMs = try {
                    val retriever = android.media.MediaMetadataRetriever()
                    retriever.setDataSource(context, workingVideoUri)
                    val durStr = retriever.extractMetadata(android.media.MediaMetadataRetriever.METADATA_KEY_DURATION)
                    retriever.release()
                    durStr?.toLongOrNull() ?: 0L
                } catch (_: Exception) { 0L }

                val maxCueEndMs = providedSubtitleCues!!.maxOfOrNull { it.endMs } ?: 0L
                val calculatedDuration = maxOf(retrievedDurationMs, maxCueEndMs + 1000L)

                project = project.copy(
                    durationMs = calculatedDuration
                )
                repository.saveProject(project)
                Log.i(TAG, "Fast dubbing mode: skipped audio extraction, detected video duration = $calculatedDuration ms")
            } else if (startingStage.ordinal <= ProcessingStage.EXTRACT_AUDIO.ordinal) {
                checkCancelled()
                reportStage(projectId, ProcessingStage.EXTRACT_AUDIO, 0, 0, "Extracting audio from video...", onProgressUpdate)

                val extractor = MediaCodecAudioExtractor(context)
                val extractResult = extractor.extractAudio(workingVideoUri, rawAudioFile) { progress ->
                    val overall = (progress * 15).toInt()
                    reportStageSync(projectId, ProcessingStage.EXTRACT_AUDIO, (progress * 100).toInt(), overall, "Extracting audio: ${(progress * 100).toInt()}%", onProgressUpdate)
                }

                if (extractResult.isFailure) {
                    throw extractResult.exceptionOrNull() ?: IllegalStateException("Audio extraction failed")
                }
                project = project.copy(
                    originalAudioPath = rawAudioFile.absolutePath,
                    durationMs = WavUtils.getWavDurationMs(rawAudioFile)
                )
                repository.saveProject(project)
            }

            val totalDurationMs = if (project.durationMs > 0) {
                project.durationMs
            } else if (hasProvidedSubtitles) {
                (providedSubtitleCues?.maxOfOrNull { it.endMs } ?: 0L) + 1000L
            } else {
                WavUtils.getWavDurationMs(rawAudioFile)
            }

            // STAGE 2: Transcribe English Speech (15% - 40%) OR Use Extracted Subtitle Cues (Instant)
            var segments: List<TranscriptSegmentEntity>
            if (providedSubtitleCues != null && providedSubtitleCues.isNotEmpty()) {
                checkCancelled()
                reportStage(
                    projectId,
                    ProcessingStage.TRANSCRIBE,
                    100,
                    40,
                    "⚡ Extracted ${providedSubtitleCues.size} subtitle cues (ASR skipped)",
                    onProgressUpdate
                )
                segments = providedSubtitleCues.mapIndexed { index, cue ->
                    TranscriptSegmentEntity(
                        projectId = projectId,
                        index = index,
                        startMs = cue.startMs,
                        endMs = cue.endMs,
                        sourceText = cue.text,
                        confidence = 1.0f
                    )
                }
                repository.deleteSegmentsForProject(projectId)
                repository.saveSegments(segments)

                val sourceSrtFile = File(projectDir, "subtitle_en_source.srt")
                SubtitleExtractor.exportToSrt(providedSubtitleCues, sourceSrtFile)

                val baseVideoTitle = videoTitle.substringBeforeLast(".")
                SubtitleExtractor.saveSrtToPublicDownloads(
                    context,
                    "${baseVideoTitle}_English",
                    sourceSrtFile.readText()
                )
                Log.i(TAG, "Successfully bypassed ASR using ${providedSubtitleCues.size} extracted subtitle cues")
            } else if (startingStage.ordinal <= ProcessingStage.TRANSCRIBE.ordinal) {
                checkCancelled()
                reportStage(projectId, ProcessingStage.TRANSCRIBE, 0, 15, "English Speech Recognition (ASR)...", onProgressUpdate)

                val asrEngine = EnglishAsrEngine(context) { seg, chunkProgress ->
                    val overall = 15 + (chunkProgress * 25).toInt()
                    reportStageSync(projectId, ProcessingStage.TRANSCRIBE, (chunkProgress * 100).toInt(), overall, "Transcribing: \"${seg.sourceText.take(24)}...\"", onProgressUpdate)
                }
                val rawSegments = asrEngine.transcribe(rawAudioFile)
                asrEngine.close()

                segments = rawSegments.mapIndexed { index, seg ->
                    TranscriptSegmentEntity(
                        projectId = projectId,
                        index = index,
                        startMs = seg.startMs,
                        endMs = seg.endMs,
                        sourceText = seg.sourceText,
                        confidence = seg.confidence
                    )
                }
                repository.deleteSegmentsForProject(projectId)
                repository.saveSegments(segments)

                com.example.models.MemoryDiagnostics.logHeapSnapshot("PIPELINE", "ASR Stage Complete, released ASR session")
                System.gc()
            } else {
                segments = repository.findSegments(projectId)
            }

            if (segments.isEmpty()) {
                throw IllegalStateException("Transcription failed: No English speech detected in the audio track.")
            }

            // STAGE 3: Translate to Bangla (40% - 55%) via GTX API with Offline ML Kit Fallback & Normalization
            if (startingStage.ordinal <= ProcessingStage.TRANSLATE.ordinal) {
                checkCancelled()

                val isAlreadyBangla = segments.any { seg -> seg.sourceText.any { ch -> ch in '\u0980'..'\u09FF' } }
                if (isAlreadyBangla) {
                    reportStage(projectId, ProcessingStage.TRANSLATE, 100, 55, "✓ সাবটাইটেল ইতোমধ্যে বাংলায় রয়েছে, নর্মালাইজেশন সম্পন্ন...", onProgressUpdate)
                    segments = segments.map { seg -> seg.copy(translatedText = BanglaNaturalizer.naturalize(seg.sourceText)) }
                    repository.saveSegments(segments)
                } else {
                    var gtxSuccess = false

                    // Tier 1: Try Fast Online Google GTX API
                    try {
                        reportStage(projectId, ProcessingStage.TRANSLATE, 0, 40, "অনলাইন Google Translate API দিয়ে বাংলায় অনুবাদ শুরু হচ্ছে...", onProgressUpdate)

                        val translationStartTime = System.currentTimeMillis()
                        val gtxItems = segments.map { seg ->
                            val sTime = SubtitleExtractor.formatSrtTimestamp(seg.startMs)
                            val eTime = SubtitleExtractor.formatSrtTimestamp(seg.endMs)
                            com.example.translation.SubtitleItem(
                                index = seg.index + 1,
                                startTimeStr = sTime,
                                endTimeStr = eTime,
                                startTimeMs = seg.startMs,
                                endTimeMs = seg.endMs,
                                originalText = seg.sourceText
                            )
                        }

                        var lastReportedOverall = 40
                        com.example.translation.GtxSubtitleTranslator.translateSubtitlesToBangla(gtxItems) { current, total ->
                            checkCancelled()
                            val stageProg = (current.toFloat() / total.toFloat())
                            val overall = maxOf(lastReportedOverall, 40 + (stageProg * 15).toInt())
                            lastReportedOverall = overall
                            val sample = gtxItems.getOrNull(current - 1)?.translatedText?.take(22) ?: ""
                            reportStage(
                                projectId,
                                ProcessingStage.TRANSLATE,
                                (stageProg * 100).toInt(),
                                overall,
                                "অনলাইন অনুবাদ (GTX): $current/$total (\"$sample...\")",
                                onProgressUpdate
                            )
                        }

                        val banglaCount = gtxItems.count { it.translatedText.any { ch -> ch in '\u0980'..'\u09FF' } }
                        if (banglaCount > 0) {
                            gtxSuccess = true
                            segments = segments.mapIndexed { idx, seg ->
                                val rawTrans = gtxItems.getOrNull(idx)?.translatedText?.trim()?.takeIf { it.isNotEmpty() } ?: seg.sourceText
                                val natural = BanglaNaturalizer.naturalize(rawTrans)
                                seg.copy(translatedText = natural)
                            }
                            repository.saveSegments(segments)

                            val transElapsed = System.currentTimeMillis() - translationStartTime
                            Log.i(TAG, "GTX Translation & Normalization Complete in ${transElapsed}ms for ${segments.size} cues")
                            reportStage(
                                projectId,
                                ProcessingStage.TRANSLATE,
                                100,
                                55,
                                "✓ অনলাইন অনুবাদ ও নর্মালাইজেশন সম্পন্ন (${segments.size} সংলাপ - GTX API)",
                                onProgressUpdate
                            )
                        } else {
                            Log.w(TAG, "GTX API did not return Bengali text, falling back to offline ML Kit")
                        }
                    } catch (e: Exception) {
                        Log.w(TAG, "GTX online translation failed: ${e.message}, switching to offline translation")
                    }

                    // Tier 2: Offline ML Kit Fallback with Rule-Based Normalization
                    if (!gtxSuccess) {
                        checkCancelled()
                        reportStage(projectId, ProcessingStage.TRANSLATE, 0, 40, "অনলাইন API কাজ করেনি, অফলাইন ML Kit মডেলে অনুবাদ ও নর্মালাইজেশন শুরু হচ্ছে...", onProgressUpdate)

                        val offlineTranslator = SubtitleBatchTranslator(context)
                        try {
                            val prepared = offlineTranslator.prepareModel()
                            if (prepared || offlineTranslator.isModelDownloaded()) {
                                var lastReportedOverall = 40
                                segments = offlineTranslator.translateSegments(segments) { current, total, sample ->
                                    checkCancelled()
                                    val stageProg = (current.toFloat() / total.toFloat())
                                    val overall = maxOf(lastReportedOverall, 40 + (stageProg * 12).toInt())
                                    lastReportedOverall = overall
                                    reportStage(
                                        projectId,
                                        ProcessingStage.TRANSLATE,
                                        (stageProg * 100).toInt(),
                                        overall,
                                        "অফলাইন অনুবাদ (ML Kit + নর্মালাইজেশন): $current/$total (\"${sample.take(20)}...\")",
                                        onProgressUpdate
                                    )
                                }
                                repository.saveSegments(segments)
                                Log.i(TAG, "Offline ML Kit translation pass complete")
                            }
                        } catch (e: Exception) {
                            Log.w(TAG, "Offline ML Kit translation note: ${e.message}")
                        } finally {
                            offlineTranslator.close()
                        }
                    }

                    // Tier 3: Guaranteed Offline Dialogue Engine + BanglaNaturalizer
                    val unTranslatedCount = segments.count { it.translatedText.isNullOrBlank() }
                    if (unTranslatedCount > 0) {
                        checkCancelled()
                        reportStage(projectId, ProcessingStage.TRANSLATE, 0, 48, "অফলাইন ডায়ালগ ইঞ্জিন ও নর্মালাইজেশন বিকল্প কার্যকর করা হচ্ছে...", onProgressUpdate)
                        segments = segments.map { seg ->
                            checkCancelled()
                            val text = if (!seg.translatedText.isNullOrBlank()) {
                                seg.translatedText!!
                            } else {
                                OfflineDialogueTranslator.translate(seg.sourceText)
                            }
                            val natural = BanglaNaturalizer.naturalize(text)
                            seg.copy(translatedText = natural)
                        }
                        repository.saveSegments(segments)
                        reportStage(
                            projectId,
                            ProcessingStage.TRANSLATE,
                            100,
                            55,
                            "✓ অফলাইন ডায়ালগ ইঞ্জিন ও নর্মালাইজেশন সম্পন্ন (${segments.size} সংলাপ)",
                            onProgressUpdate
                        )
                        Log.i(TAG, "OfflineDialogueTranslator finished for $unTranslatedCount segments")
                    } else {
                        reportStage(
                            projectId,
                            ProcessingStage.TRANSLATE,
                            100,
                            55,
                            "✓ অনুবাদ ও নর্মালাইজেশন সম্পন্ন (${segments.size} সংলাপ)",
                            onProgressUpdate
                        )
                    }
                }
            }

            // STAGE 4: Generate Bangla Subtitles & JSON transcript (55% - 65%)
            if (startingStage.ordinal <= ProcessingStage.GENERATE_SUBTITLE.ordinal) {
                checkCancelled()
                reportStage(projectId, ProcessingStage.GENERATE_SUBTITLE, 0, 55, "Generating Bangla SRT and structured transcript...", onProgressUpdate)

                SubtitleGenerator.generateSrt(segments, srtFile)
                SubtitleGenerator.generateTranscriptJson("en", "bn", segments, transcriptJsonFile)

                val baseVideoTitle = videoTitle.substringBeforeLast(".")
                val bnSrtContent = srtFile.readText()
                val downloadedFile = SubtitleExtractor.saveSrtToPublicDownloads(
                    context,
                    "${baseVideoTitle}_bangla",
                    bnSrtContent
                )
                val statusMsg = if (downloadedFile != null) {
                    "✓ Bangla SRT saved to Download/FileTranslato (${downloadedFile.name})"
                } else {
                    "Bangla subtitles generated."
                }
                reportStage(projectId, ProcessingStage.GENERATE_SUBTITLE, 100, 65, statusMsg, onProgressUpdate)
                Log.i(TAG, "Translated Bangla subtitle generated and exported to FileTranslato: ${downloadedFile?.absolutePath}")
            }

            // STAGE 5: Bangla Voice Synthesis (TTS) (65% - 85%)
            if (startingStage.ordinal <= ProcessingStage.GENERATE_TTS.ordinal) {
                checkCancelled()
                reportStage(projectId, ProcessingStage.GENERATE_TTS, 0, 65, "Preparing Android Bengali TextToSpeech...", onProgressUpdate)

                val ttsStartTime = System.currentTimeMillis()
                val ttsEngine = BanglaTtsEngine(context)
                try {
                    val initResult = ttsEngine.ensureInitialized()
                    if (initResult.isFailure) {
                        val reason = initResult.exceptionOrNull()?.message ?: BanglaTtsEngine.ERROR_VOICE_NOT_INSTALLED
                        reportStage(projectId, ProcessingStage.GENERATE_TTS, 0, 65, "Failed: $reason", onProgressUpdate)
                        throw IllegalStateException(reason)
                    }

                    val ttsSegments = mutableListOf<TranscriptSegmentEntity>()
                    var successfulTtsSegments = 0
                    var failedTtsSegments = 0

                    for (i in segments.indices) {
                        checkCancelled()
                        val seg = segments[i]
                        val segAudioFile = File(segmentsDir, "tts_seg_${seg.index}.wav")

                        // Resume capability: skip if already synthesized on disk from earlier run
                        if (segAudioFile.exists() && segAudioFile.length() > 44L) {
                            ttsSegments.add(seg.copy(audioSegmentPath = segAudioFile.absolutePath))
                            successfulTtsSegments++
                            val stageProg = ((i + 1).toFloat() / segments.size)
                            val overall = 65 + (stageProg * 20).toInt()
                            reportStageSync(projectId, ProcessingStage.GENERATE_TTS, (stageProg * 100).toInt(), overall, "Synthesized (${i + 1}/${segments.size})", onProgressUpdate)
                            continue
                        }

                        val rawText = seg.translatedText?.trim()?.ifEmpty { null } ?: ""
                        val textToSpeak = rawText.filter { it != '\u0000' && !it.isISOControl() || it == '\n' || it == '\t' }

                        if (textToSpeak.isBlank()) {
                            val segDuration = maxOf(300L, seg.endMs - seg.startMs)
                            WavUtils.createSilenceWav(segAudioFile, segDuration)
                        } else {
                            try {
                                ttsEngine.synthesize(textToSpeak, segAudioFile)
                                successfulTtsSegments++
                            } catch (e: Exception) {
                                failedTtsSegments++
                                Log.w(TAG, "Bangla TTS failed for segment ${seg.index + 1}: ${e.message}")
                                val segDuration = maxOf(300L, seg.endMs - seg.startMs)
                                WavUtils.createSilenceWav(segAudioFile, segDuration)
                            }
                        }
                        ttsSegments.add(seg.copy(audioSegmentPath = segAudioFile.absolutePath))

                        // Periodically save segment audio paths to database every 50 items to minimize SQLite I/O on 2,000+ cue movies
                        if ((i + 1) % 50 == 0 || i == segments.size - 1) {
                            repository.saveSegments(ttsSegments)
                        }

                        if ((i + 1) % 25 == 0 || i == segments.size - 1) {
                            val elapsed = System.currentTimeMillis() - ttsStartTime
                            val avg = elapsed.toDouble() / (i + 1)
                            Log.i(TAG, "TTS Synthesis Progress: ${i + 1}/${segments.size}. Elapsed: ${elapsed}ms, Avg: ${String.format("%.1f", avg)}ms/seg, Failures: $failedTtsSegments")
                        }

                        val stageProg = ((i + 1).toFloat() / segments.size)
                        val overall = 65 + (stageProg * 20).toInt()
                        reportStageSync(projectId, ProcessingStage.GENERATE_TTS, (stageProg * 100).toInt(), overall, "Synthesizing (${i + 1}/${segments.size})", onProgressUpdate)
                    }

                    segments = ttsSegments
                    repository.saveSegments(segments)
                    val ttsTotalElapsed = System.currentTimeMillis() - ttsStartTime
                    Log.i(TAG, "TTS Stage Complete in ${ttsTotalElapsed}ms. Synthesized ${successfulTtsSegments}/${segments.size} segments (Failures: $failedTtsSegments)")
                    reportStage(
                        projectId,
                        ProcessingStage.GENERATE_TTS,
                        100,
                        85,
                        "✓ Voice synthesis complete: ${segments.size} dialogue segments",
                        onProgressUpdate
                    )
                } finally {
                    ttsEngine.close()
                }
                com.example.models.MemoryDiagnostics.logHeapSnapshot("PIPELINE", "TTS Stage Complete")
                System.gc()
            }

            // STAGE 6: Audio Synchronization (85% - 95%)
            var effectiveDubbedAudioFile = finalDubbedAudioFile
            if (startingStage.ordinal <= ProcessingStage.SYNC_AUDIO.ordinal) {
                checkCancelled()
                reportStage(projectId, ProcessingStage.SYNC_AUDIO, 0, 85, "Synchronizing dubbed audio with video...", onProgressUpdate)

                val syncStartTime = System.currentTimeMillis()
                val synchronizer = AudioSynchronizer(context)
                val syncResult = synchronizer.synchronizeAndMux(
                    segments = segments,
                    totalDurationMs = totalDurationMs,
                    outputFile = finalDubbedAudioFile,
                    backgroundAudioFile = rawAudioFile.takeIf { it.exists() && it.length() > 44 }
                ) { prog ->
                    val overall = 85 + (prog * 10).toInt()
                    reportStageSync(projectId, ProcessingStage.SYNC_AUDIO, (prog * 100).toInt(), overall, "Synchronizing audio: ${(prog * 100).toInt()}%", onProgressUpdate)
                }

                if (syncResult.isSuccess) {
                    effectiveDubbedAudioFile = syncResult.getOrThrow()
                    try {
                        val baseVideoTitle = videoTitle.substringBeforeLast(".")
                        val publicDownloads = android.os.Environment.getExternalStoragePublicDirectory(android.os.Environment.DIRECTORY_DOWNLOADS)
                        val fileTranslatoDir = File(publicDownloads, "FileTranslato").apply { if (!exists()) mkdirs() }
                        val publicDubbedAudio = File(fileTranslatoDir, "${baseVideoTitle}_bangla_dubbed_audio.wav")
                        effectiveDubbedAudioFile.copyTo(publicDubbedAudio, overwrite = true)
                        Log.i(TAG, "Exported dubbed audio to Download/FileTranslato: ${publicDubbedAudio.absolutePath}")

                        if (rawAudioFile.exists() && rawAudioFile.length() > 44) {
                            val publicOriginalAudio = File(fileTranslatoDir, "${baseVideoTitle}_original_audio.wav")
                            rawAudioFile.copyTo(publicOriginalAudio, overwrite = true)
                        }
                    } catch (e: Exception) {
                        Log.w(TAG, "Could not mirror audio to FileTranslato: ${e.message}")
                    }
                } else {
                    WavUtils.createSilenceWav(finalDubbedAudioFile, maxOf(1000L, totalDurationMs))
                    effectiveDubbedAudioFile = finalDubbedAudioFile
                }
                val syncElapsed = System.currentTimeMillis() - syncStartTime
                Log.i(TAG, "Audio Synchronization Complete in ${syncElapsed}ms, output: ${effectiveDubbedAudioFile.length()} bytes")
                reportStage(projectId, ProcessingStage.SYNC_AUDIO, 100, 95, "✓ Synchronized audio ready", onProgressUpdate)
            }

            // STAGE 7: Finalize & Complete (95% - 100%)
            reportStage(projectId, ProcessingStage.FINALIZE, 50, 95, "Verifying output files...", onProgressUpdate)

            if (!transcriptJsonFile.exists() || transcriptJsonFile.length() == 0L) {
                throw IllegalStateException("Output verification failed: English/Bangla transcript JSON file is missing or empty (Stage: Subtitle & Transcript Generation).")
            }

            if (!srtFile.exists() || srtFile.length() == 0L) {
                throw IllegalStateException("Output verification failed: Bangla subtitle SRT file is missing or empty (Stage: Subtitle Generation).")
            }

            val srtContent = srtFile.readText()
            if (!srtContent.contains("-->") || srtContent.isBlank()) {
                throw IllegalStateException("Output verification failed: Bangla subtitle SRT file does not contain valid subtitle timestamp blocks.")
            }

            val dubbedAudioToUse = when {
                effectiveDubbedAudioFile.exists() && effectiveDubbedAudioFile.length() >= 1000L -> effectiveDubbedAudioFile
                finalDubbedAudioFile.exists() && finalDubbedAudioFile.length() >= 1000L -> finalDubbedAudioFile
                else -> {
                    val fallbackWav = File(projectDir, "dubbed_bn.wav")
                    if (fallbackWav.exists() && fallbackWav.length() >= 1000L) fallbackWav
                    else throw IllegalStateException("Output verification failed: Bangla dubbed audio file is missing or contains insufficient data (Stage: Audio Synchronization).")
                }
            }

            val resolvedFinalVideoUri = when {
                localVideoFile.exists() && localVideoFile.length() > 1024 -> Uri.fromFile(localVideoFile)
                workingVideoUri.toString().isNotBlank() -> workingVideoUri
                else -> VideoUriResolver.resolvePlayableVideoUri(
                    context = context,
                    projectId = projectId,
                    videoTitle = videoTitle,
                    savedUriString = workingVideoUri.toString(),
                    projectDirPath = projectDir.absolutePath
                ) ?: workingVideoUri
            }

            project = project.copy(
                videoUriString = resolvedFinalVideoUri.toString(),
                currentStage = ProcessingStage.COMPLETE,
                progressPercent = 100,
                statusMessage = "Offline AI Dubbing Complete",
                originalAudioPath = if (rawAudioFile.exists()) rawAudioFile.absolutePath else null,
                transcriptJsonPath = transcriptJsonFile.absolutePath,
                subtitleSrtPath = srtFile.absolutePath,
                dubbedAudioPath = dubbedAudioToUse.absolutePath,
                durationMs = totalDurationMs,
                updatedAt = System.currentTimeMillis()
            )
            repository.saveProject(project)

            if (settingsManager.autoCleanTempFiles.value) {
                storageManager.clearTemporaryFiles()
            }

            reportStage(projectId, ProcessingStage.COMPLETE, 100, 100, "Dubbing ready to play!", onProgressUpdate)
            Result.success(project)

        } catch (e: CancellationException) {
            Log.w(TAG, "Dubbing cancelled by user")
            val cancelledState = project.copy(
                currentStage = ProcessingStage.CANCELLED,
                statusMessage = "Dubbing cancelled by user"
            )
            repository.saveProject(cancelledState)
            reportStage(projectId, ProcessingStage.CANCELLED, 0, 0, "Cancelled", onProgressUpdate)
            Result.failure(e)
        } catch (e: Throwable) {
            Log.e(TAG, "Dubbing pipeline failed", e)
            val failedState = project.copy(
                currentStage = ProcessingStage.FAILED,
                statusMessage = "Error: ${e.localizedMessage ?: "Unknown error"}",
                errorMessage = e.localizedMessage
            )
            repository.saveProject(failedState)
            reportStage(projectId, ProcessingStage.FAILED, 0, 0, e.localizedMessage ?: "Processing error", onProgressUpdate)
            Result.failure(e)
        }
    }

    private suspend fun determineResumeStage(
        project: DubbingProject,
        rawAudio: File,
        transcriptJson: File,
        srtFile: File,
        dubbedAudio: File
    ): ProcessingStage {
        val existingSegments = repository.findSegments(project.id)
        if (existingSegments.isNotEmpty()) {
            val allTranslated = existingSegments.all { !it.translatedText.isNullOrBlank() }
            if (dubbedAudio.exists() && dubbedAudio.length() > 1000) return ProcessingStage.COMPLETE
            if (srtFile.exists() && srtFile.length() > 0) return ProcessingStage.GENERATE_TTS
            if (allTranslated) return ProcessingStage.GENERATE_SUBTITLE
            return ProcessingStage.TRANSLATE
        }

        return when {
            dubbedAudio.exists() && dubbedAudio.length() > 1000 -> ProcessingStage.COMPLETE
            srtFile.exists() && srtFile.length() > 0 -> ProcessingStage.GENERATE_TTS
            transcriptJson.exists() && transcriptJson.length() > 0 -> ProcessingStage.TRANSLATE
            rawAudio.exists() && rawAudio.length() > 44 -> ProcessingStage.TRANSCRIBE
            else -> ProcessingStage.EXTRACT_AUDIO
        }
    }

    private fun checkCancelled() {
        if (activeJobCancelled) {
            throw CancellationException("Pipeline cancelled by user")
        }
    }

    private fun reportStageSync(
        projectId: String,
        stage: ProcessingStage,
        stageProg: Int,
        overallProg: Int,
        message: String,
        callback: ((PipelineProgress) -> Unit)?
    ) {
        val safeOverall = if (stage == ProcessingStage.FAILED || stage == ProcessingStage.CANCELLED) {
            overallProg
        } else {
            maxOf(_pipelineState.value?.overallProgressPercent ?: 0, overallProg)
        }

        val progress = PipelineProgress(
            projectId = projectId,
            stage = stage,
            stageProgressPercent = stageProg,
            overallProgressPercent = safeOverall,
            statusMessage = message,
            isRunning = stage != ProcessingStage.COMPLETE && stage != ProcessingStage.FAILED && stage != ProcessingStage.CANCELLED
        )
        _pipelineState.value = progress
        callback?.invoke(progress)
    }

    private suspend fun reportStage(
        projectId: String,
        stage: ProcessingStage,
        stageProg: Int,
        overallProg: Int,
        message: String,
        callback: ((PipelineProgress) -> Unit)?
    ) {
        reportStageSync(projectId, stage, stageProg, overallProg, message, callback)
        repository.updateProjectProgress(projectId, stage, overallProg, message)
    }
}

    @Test
    fun testUniversalStereoAudioProcessorHandles5Point1FloatAndStereo() {
        val processor = UniversalStereoAudioProcessor()

        // 1. Standard 16-bit Stereo should pass through (isActive == false) without throwing
        val stereo16Format = AudioProcessor.AudioFormat(48000, 2, C.ENCODING_PCM_16BIT)
        val outStereo16 = processor.configure(stereo16Format)
        assertEquals(AudioProcessor.AudioFormat.NOT_SET, outStereo16)
        assertFalse(processor.isActive)

        // 2. 5.1 Surround (6 channels) Float PCM (fltp, like HE-AAC 5.1 in MKV) must configure to 2ch 16-bit PCM
        val surround51Float = AudioProcessor.AudioFormat(48000, 6, C.ENCODING_PCM_FLOAT)
        val outSurround = processor.configure(surround51Float)
        processor.flush()
        assertTrue(processor.isActive)
        assertEquals(48000, outSurround.sampleRate)
        assertEquals(2, outSurround.channelCount)
        assertEquals(C.ENCODING_PCM_16BIT, outSurround.encoding)

        // Feed 2 frames of 6-channel float PCM (FL, FR, FC, LFE, BL, BR)
        val inBuf = ByteBuffer.allocateDirect(2 * 6 * 4).order(ByteOrder.nativeOrder())
        repeat(2) {
            inBuf.putFloat(0.4f) // FL
            inBuf.putFloat(0.4f) // FR
            inBuf.putFloat(0.5f) // FC (Center dialogue)
            inBuf.putFloat(0.1f) // LFE
            inBuf.putFloat(0.2f) // BL
            inBuf.putFloat(0.2f) // BR
        }
        inBuf.flip()
        processor.queueInput(inBuf)
        val output = processor.output
        // 2 frames * 2 channels * 2 bytes = 8 bytes
        assertEquals(8, output.remaining())
    }
